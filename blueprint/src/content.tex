\chapter{Approximation schemes}
\label{ch:schemes}

This blueprint follows the formalisation of approximation spaces in the sense of
Pietsch~\cite{Pietsch}, of Besov and Sobolev spaces defined through them, and of the
embedding theorems between these spaces. An approximation scheme is an increasing family of
sets $A_0 \subseteq A_1 \subseteq \cdots$ in a normed space, and $f$ belongs to the
approximation space $A^s_q$ when its errors of best approximation
$E_n(f) = \dist(f, A_n)$ decay like $n^{-s}$ in the $\ell_q$ sense. For a scheme of
bounded continuous functions on a finite measure space, the approximation spaces of $L_p$
are the Besov spaces, and the embedding theorems follow from the abstract theory together
with a Nikolskii inequality for the scheme. The torus $\T^d$ with the trigonometric
polynomials is the concrete instance, and there the spaces are compared with the classical
definitions through moduli of smoothness and weak derivatives.

Each statement names the Lean declaration that formalises it. Every statement and every
proof is formalised: the project compiles without \texttt{sorry}, so each result is
verified by the Lean kernel.

Throughout, $X$ and $Y$ are real normed spaces. Complex spaces, such as $L_p$ spaces of
complex-valued functions, are regarded as real ones.

\begin{definition}[Approximation scheme]
\label{def:scheme}
\lean{ApproximationScheme}
\leanok
An \emph{approximation scheme} in $X$ is a sequence of subsets
$A_0 \subseteq A_1 \subseteq A_2 \subseteq \cdots$ of $X$ such that
\begin{enumerate}
  \item $A_0 = \{0\}$,
  \item $c\,a \in A_n$ for all $c \in \R$, $n \in \N$ and $a \in A_n$,
  \item $a + b \in A_{m+n}$ for all $m, n \in \N$, $a \in A_m$ and $b \in A_n$.
\end{enumerate}
\end{definition}

The elements of $A_n$ are the approximants of complexity $n$. The sets need not be linear
subspaces: in the $n$-term scheme of \Cref{def:nterm} they are unions of subspaces. A family
indexed by a degree enters with a strict inequality, $A_n = \{\text{degree} < n\}$, so that
$A_0 = \{0\}$ holds.

\begin{lemma}[Elementary properties]
\label{lem:scheme-basic}
\lean{ApproximationScheme.zero_mem, ApproximationScheme.nonempty,
  ApproximationScheme.neg_mem, ApproximationScheme.sub_mem, ApproximationScheme.sum_mem,
  ApproximationScheme.smul_set_eq, ApproximationScheme.sub_mem_pow_add_two}
\leanok
\uses{def:scheme}
Let $A$ be an approximation scheme in $X$. Then for all $m, n, k \in \N$:
\begin{enumerate}
  \item $0 \in A_n$; in particular $A_n$ is nonempty;
  \item $a - b \in A_{m+n}$ for all $a \in A_m$ and $b \in A_n$;
  \item $\sum_{i \in I} a_i \in A_{\sum_{i \in I} n_i}$ for every finite family with
    $a_i \in A_{n_i}$;
  \item $c\,A_n = A_n$ for every $c \in \R \setminus \{0\}$;
  \item $a - b \in A_{2^{k+2}}$ for all $a \in A_{2^{k+1}}$ and $b \in A_{2^k}$.
\end{enumerate}
\end{lemma}

\begin{proof}
\leanok
The first item follows from $A_0 = \{0\} \subseteq A_n$, the second from $-b = (-1)\,b$, and
the third by induction over $I$. For the fourth, $c^{-1} a \in A_n$ for $a \in A_n$. The last
follows from the second and $2^{k+1} + 2^k \leq 2^{k+2}$.
\end{proof}

\begin{definition}[Preimage and image of a scheme]
\label{def:comap}
\lean{ApproximationScheme.comap, ApproximationScheme.map, ApproximationScheme.ofSubmodules}
\leanok
\uses{def:scheme}
Let $A$ be an approximation scheme in $X$. For an injective linear map $\iota : Y \to X$, the
sets $\iota^{-1}(A_n)$ form an approximation scheme in $Y$. For a linear map $T : X \to Y$, the
sets $T(A_n)$ form an approximation scheme in $Y$. An increasing sequence of linear subspaces
$V_0 \subseteq V_1 \subseteq \cdots$ of $X$ with $V_0 = \{0\}$ is an approximation scheme.
\end{definition}

Injectivity of $\iota$ is what gives $\iota^{-1}(\{0\}) = \{0\}$.

\section{The error of best approximation}

\begin{definition}[Error of best approximation]
\label{def:error}
\lean{ApproximationScheme.approxError}
\leanok
\uses{def:scheme}
Let $A$ be an approximation scheme in $X$. For $n \in \N$ and $f \in X$,
\[
  E_n(f) = \dist(f, A_n) = \inf_{a \in A_n} \|f - a\|.
\]
\end{definition}

\begin{lemma}[Calculus of the error]
\label{lem:error-basic}
\lean{ApproximationScheme.approxError_zero_left, ApproximationScheme.approxError_le_norm,
  ApproximationScheme.approxError_anti, ApproximationScheme.approxError_smul,
  ApproximationScheme.approxError_add_le, ApproximationScheme.abs_approxError_sub_le}
\leanok
\uses{def:error, lem:scheme-basic}
Let $A$ be an approximation scheme in $X$. For all $f, g \in X$, $c \in \R$ and
$m, n \in \N$:
\begin{enumerate}
  \item $E_0(f) = \|f\|$ and $E_n(f) \leq \|f\|$;
  \item $E_n(f) \leq E_m(f)$ whenever $m \leq n$;
  \item $E_n(c f) = |c| \, E_n(f)$;
  \item $E_{m+n}(f + g) \leq E_m(f) + E_n(g)$;
  \item $|E_n(f) - E_n(g)| \leq \|f - g\|$.
\end{enumerate}
\end{lemma}

\begin{proof}
\leanok
The first item holds because $A_0 = \{0\}$ and $0 \in A_n$, the second because the sets
increase, the third because $c A_n = A_n$ for $c \neq 0$. For the fourth, given
$\varepsilon > 0$ choose $a \in A_m$ and $b \in A_n$ with $\|f - a\| \leq E_m(f) +
\varepsilon$ and $\|g - b\| \leq E_n(g) + \varepsilon$; then $a + b \in A_{m+n}$ and the
triangle inequality gives the bound up to $2\varepsilon$. The last item is the
$1$-Lipschitz property of the distance to a set.
\end{proof}

A best approximant, an $a \in A_n$ with $\|f - a\| = E_n(f)$, need not exist, since the sets
$A_n$ need not be closed. The proofs work with near-best approximants as in the proof of the
fourth item.

\begin{lemma}[Convergence of the errors]
\label{lem:error-tendsto}
\lean{ApproximationScheme.tendsto_approxError_zero_iff,
  ApproximationScheme.tendsto_approxError_zero_of_pow}
\leanok
\uses{lem:error-basic}
Let $A$ be an approximation scheme in $X$ and $f \in X$. Then $E_n(f) \to 0$ as
$n \to \infty$ if and only if $f$ lies in the closure of $\bigcup_n A_n$, and this holds as
soon as $E_{2^k}(f) \to 0$ as $k \to \infty$.
\end{lemma}

\begin{proof}
\leanok
If $\|f - a\| < \varepsilon$ for some $a \in A_m$, then $E_n(f) < \varepsilon$ for all
$n \geq m$ since the errors decrease. Conversely $E_n(f) < \varepsilon$ gives such an $a$.
The errors decrease, so convergence along $2^k$ gives convergence.
\end{proof}

\begin{lemma}[Functoriality of the error]
\label{lem:error-map}
\lean{ApproximationScheme.approxError_map_le}
\leanok
\uses{def:error}
Let $A$ and $B$ be approximation schemes in $X$ and $Y$, and let $T : X \to Y$ be a bounded
linear map with $T(A_n) \subseteq B_n$ for all $n$. Then for all $n \in \N$ and $f \in X$,
\[
  E^B_n(T f) \leq \|T\| \, E^A_n(f).
\]
\end{lemma}

\begin{proof}
\leanok
For $a \in A_n$, $E^B_n(Tf) \leq \|Tf - Ta\| \leq \|T\| \, \|f - a\|$; take the infimum
over $a$.
\end{proof}

\section{Two schemes}

\begin{definition}[Linear approximation]
\label{def:ofspan}
\lean{ApproximationScheme.ofSpan}
\leanok
\uses{def:scheme}
For a sequence $(e_i)_{i \in \N}$ in $X$, the sets $A_n = \spn\{e_i : i < n\}$ form an
approximation scheme.
\end{definition}

\begin{lemma}
\label{lem:ofspan}
\lean{ApproximationScheme.isClosed_ofSpan, ApproximationScheme.iUnion_ofSpan,
  ApproximationScheme.tendsto_approxError_ofSpan_iff}
\leanok
\uses{def:ofspan, lem:error-tendsto}
For the scheme of \Cref{def:ofspan}, every $A_n$ is closed, $\bigcup_n A_n = \spn\{e_i : i
\in \N\}$, and $E_n(f) \to 0$ if and only if $f$ lies in the closed span of $(e_i)$.
\end{lemma}

\begin{proof}
\leanok
The sets are finite-dimensional subspaces and therefore closed. The union of an increasing
family of spans is the span of the union. The last statement is \Cref{lem:error-tendsto}.
\end{proof}

\begin{definition}[$n$-term approximation]
\label{def:nterm}
\lean{ApproximationScheme.nTerm}
\leanok
\uses{def:scheme}
For a set $D \subseteq X$, the \emph{dictionary}, let $A_n$ be the set of linear combinations
of at most $n$ elements of $D$. These sets form an approximation scheme.
\end{definition}

The third axiom holds because a combination of $m$ and one of $n$ elements of $D$ together
use at most $m + n$ elements.

\begin{lemma}
\label{lem:nterm-ofspan}
\lean{ApproximationScheme.ofSpan_subset_nTerm, ApproximationScheme.approxError_nTerm_le_ofSpan}
\leanok
\uses{def:ofspan, def:nterm, lem:error-basic}
For a sequence $(e_i)$ in $X$, the linear scheme of $(e_i)$ is contained in the $n$-term
scheme of the dictionary $\{e_i : i \in \N\}$, and hence the $n$-term errors are at most the
linear ones.
\end{lemma}

\begin{proof}
\leanok
An element of $\spn\{e_i : i < n\}$ uses the $n$ elements $e_0, \dots, e_{n-1}$.
\end{proof}

\chapter{Sequence spaces}
\label{ch:sequences}

The quasi-norms of this blueprint take values in $[0, \infty]$, and a function belongs to a
space when its quasi-norm is finite. Every series of elements of $[0, \infty]$ converges, so
no summability conditions are needed. Arithmetic in $[0, \infty]$ uses the conventions
$0 \cdot \infty = 0$, $0^{-1} = \infty$, $\infty^{-1} = 0$, and $x - y = 0$ for $x \leq y$.
With these, a constant $(1 - 2^{-\alpha})^{-1}$ is finite exactly for $\alpha > 0$.

\begin{definition}[$\ell_q$ norm and dyadic weights]
\label{def:lq-norm}
\lean{ENNReal.lqNorm, ENNReal.lqNorm_top, ENNReal.lqNorm_eq_tsum, ENNReal.dyadicWeight}
\leanok
For $q \in (0, \infty)$ and a sequence $a = (a_n)_{n \in \N}$ in $[0, \infty]$,
\[
  \|a\|_{\ell_q} = \Bigl(\sum_{n=0}^\infty a_n^q\Bigr)^{1/q},
  \qquad
  \|a\|_{\ell_\infty} = \sup_{n \in \N} a_n .
\]
For $s \in \R$ and $k \in \N$, the \emph{dyadic weight} is $2^{ks}$.
\end{definition}

In Lean, $\|a\|_{\ell_q}$ is the $L_q$ norm of $a$ for the counting measure on $\N$.

\begin{lemma}[Properties of the $\ell_q$ norm]
\label{lem:lq-basic}
\lean{ENNReal.lqNorm_mono, ENNReal.le_lqNorm, ENNReal.lqNorm_const_mul,
  ENNReal.lqNorm_add_le, ENNReal.lqNorm_le_lqNorm_of_le, ENNReal.lqNorm_le_head_add_shift,
  ENNReal.lqNorm_le_lqNorm_one, ENNReal.lqNorm_geometric_le}
\leanok
\uses{def:lq-norm}
Let $1 \leq q \leq r \leq \infty$. For all sequences $a, b$ in $[0, \infty]$, all
$c, \rho \in [0, \infty]$ and all $n \in \N$:
\begin{enumerate}
  \item if $a_k \leq b_k$ for every $k$, then $\|a\|_{\ell_q} \leq \|b\|_{\ell_q}$;
  \item $a_n \leq \|a\|_{\ell_q}$ and $\|c\,a\|_{\ell_q} = c \, \|a\|_{\ell_q}$;
  \item $\|a + b\|_{\ell_q} \leq \|a\|_{\ell_q} + \|b\|_{\ell_q}$;
  \item $\|a\|_{\ell_r} \leq \|a\|_{\ell_q} \leq \|a\|_{\ell_1}$;
  \item $\|a\|_{\ell_q} \leq a_0 + \|(a_{k+1})_k\|_{\ell_q}$;
  \item $\|(\rho^k)_k\|_{\ell_q} \leq (1 - \rho)^{-1}$.
\end{enumerate}
\end{lemma}

\begin{proof}
\leanok
The first two items are immediate. The third is Minkowski's inequality for the finite
partial sums, followed by the supremum over them. For the fourth put $M = \|a\|_{\ell_q}$;
then $a_k^r = a_k^{r-q} a_k^q \leq M^{r-q} a_k^q$, and summation gives
$\|a\|_{\ell_r}^r \leq M^r$. The fifth is the third applied to $a_0 \delta_0$ and the rest.
The last is the fourth and the geometric series.
\end{proof}

\begin{lemma}[Countable Minkowski inequality]
\label{lem:countable-minkowski}
\lean{ENNReal.lqNorm_sum_le, ENNReal.lqNorm_iSup, ENNReal.lqNorm_tsum_le}
\leanok
\uses{lem:lq-basic}
Let $1 \leq q \leq \infty$. For every family $(f_{i,k})_{i, k \in \N}$ in $[0, \infty]$,
\[
  \Bigl\| \Bigl(\sum_{i=0}^\infty f_{i,k}\Bigr)_k \Bigr\|_{\ell_q}
  \leq \sum_{i=0}^\infty \|(f_{i,k})_k\|_{\ell_q} .
\]
\end{lemma}

\begin{proof}
\leanok
For finite sums this follows from \Cref{lem:lq-basic} by induction. The partial sums increase
to the series, and the $\ell_q$ norm of an increasing family of sequences is the supremum of
their norms, by monotone convergence.
\end{proof}

The following two inequalities are the discrete Hardy inequalities: a tail sum and a partial
sum are controlled in a dyadically weighted $\ell_q$ norm.

\begin{lemma}[Hardy inequalities]
\label{lem:hardy}
\lean{ENNReal.hardy_tail_le, ENNReal.hardy_sum_le, ENNReal.lqNorm_shift_le',
  ENNReal.lqNorm_shiftRight_le, ENNReal.tsum_dyadicWeight_neg}
\leanok
\uses{lem:countable-minkowski}
Let $\alpha \in \R$ and $1 \leq q \leq \infty$. For every sequence $b$ in $[0, \infty]$,
\[
  \Bigl\| \Bigl(2^{k\alpha} \sum_{i=0}^\infty b_{k+i}\Bigr)_k \Bigr\|_{\ell_q}
  \leq (1 - 2^{-\alpha})^{-1} \, \|(2^{k\alpha} b_k)_k\|_{\ell_q}
\]
and
\[
  \Bigl\| \Bigl(2^{-k\alpha} \sum_{j=0}^{k} b_j\Bigr)_k \Bigr\|_{\ell_q}
  \leq (1 - 2^{-\alpha})^{-1} \, \|(2^{-k\alpha} b_k)_k\|_{\ell_q} .
\]
\end{lemma}

\begin{proof}
\leanok
Write $2^{k\alpha} \sum_i b_{k+i} = \sum_i 2^{-i\alpha} \, 2^{(k+i)\alpha} b_{k+i}$. By
\Cref{lem:countable-minkowski} the left side is at most
$\sum_i 2^{-i\alpha} \|(2^{(k+i)\alpha} b_{k+i})_k\|_{\ell_q}$, a shift does not increase the
norm, and $\sum_i 2^{-i\alpha} = (1 - 2^{-\alpha})^{-1}$. The second inequality is the same
argument with $2^{-k\alpha} \sum_{j \leq k} b_j = \sum_{i \leq k} 2^{-i\alpha} \,
2^{-(k-i)\alpha} b_{k-i}$ and the shift to the right.
\end{proof}

The comparison of Pietsch's quasi-norm with its dyadic form rests on Cauchy's condensation
test with power weights.

\begin{lemma}[Cauchy condensation with power weights]
\label{lem:condensation}
\lean{ENNReal.tsum_pow_mul_le_dyadic, ENNReal.dyadic_le_tsum_pow_mul,
  ENNReal.iSup_pow_mul_le_dyadic, ENNReal.dyadic_le_iSup_pow_mul,
  ENNReal.sum_range_two_pow_eq}
\leanok
\uses{def:lq-norm}
Let $b$ be a decreasing sequence in $[0, \infty]$, let $\gamma \in \R$, and put
$\gamma_+ = \max(\gamma, 0)$ and $\gamma_- = \max(-\gamma, 0)$. Then
\[
  \sum_{n=0}^\infty (n+1)^\gamma b_n \leq b_0 + 2^{\gamma_+} \sum_{k=0}^\infty
    2^{k(\gamma+1)} b_{2^k}
\]
and
\[
  \sum_{k=0}^\infty 2^{k(\gamma+1)} b_{2^{k+1}} \leq 2^{\gamma_-} \sum_{n=0}^\infty
    (n+1)^\gamma b_n ,
\]
and for every $s \geq 0$,
\[
  \sup_n (n+1)^s b_n \leq b_0 + 2^s \sup_k 2^{ks} b_{2^k},
  \qquad
  \sup_k 2^{ks} b_{2^k} \leq \sup_n (n+1)^s b_n .
\]
\end{lemma}

\begin{proof}
\leanok
Split $\N \setminus \{0\}$ into the blocks $2^k \leq n < 2^{k+1}$ of $2^k$ elements each. On
the block $k$, $(n+1)^\gamma$ lies between $2^{-\gamma_-} 2^{k\gamma}$ and
$2^{\gamma_+} 2^{k\gamma}$, and $b_n$ lies between $b_{2^{k+1}}$ and $b_{2^k}$.
\end{proof}

\chapter{Approximation spaces}
\label{ch:approximation-spaces}

Throughout, $A$ is an approximation scheme in $X$. The following definition is
Pietsch's~\cite{Pietsch}, with his sequence $(n^{s-1/q} E_{n-1}(f))_{n \geq 1}$ indexed
from $0$.

\begin{definition}[Approximation space]
\label{def:approx-space}
\lean{ApproximationScheme.approxSeq, ApproximationScheme.approxNorm,
  ApproximationScheme.MemApprox, ApproximationScheme.approxNorm_top,
  ApproximationScheme.approxNorm_rpow}
\leanok
\uses{def:error, def:lq-norm}
Let $s \in \R$ and $1 \leq q \leq \infty$. For $f \in X$,
\[
  \|f\|_{A^s_q} = \bigl\| \bigl((n+1)^{s - 1/q} E_n(f)\bigr)_{n \in \N} \bigr\|_{\ell_q}
  \in [0, \infty],
\]
with $1/\infty = 0$. The \emph{approximation space} $A^s_q$ consists of the $f \in X$ with
$\|f\|_{A^s_q} < \infty$.
\end{definition}

\begin{definition}[Dyadic quasi-norm]
\label{def:dyadic-norm}
\lean{ApproximationScheme.approxSeqDyadic, ApproximationScheme.approxNormDyadic}
\leanok
\uses{def:error, def:lq-norm}
Let $s \in \R$ and $1 \leq q \leq \infty$. For $f \in X$,
\[
  \|f\|^{\mathrm{dy}}_{A^s_q} = \|f\| + \bigl\| \bigl(2^{ks} E_{2^k}(f)\bigr)_{k \in \N}
  \bigr\|_{\ell_q} .
\]
\end{definition}

The term $\|f\| = E_0(f)$ is needed because the dyadic sequence starts at $E_1$. The dyadic
form is the working norm: every later theorem is stated for it, and the next theorem
transports it to Pietsch's definition.

\begin{theorem}[Dyadic form of the quasi-norm]
\label{thm:dyadic-norm}
\lean{ApproximationScheme.approxNorm_le_approxNormDyadic,
  ApproximationScheme.approxNormDyadic_le_approxNorm,
  ApproximationScheme.memApprox_iff_approxNormDyadic_lt_top}
\leanok
\uses{def:approx-space, def:dyadic-norm, lem:condensation, lem:lq-basic}
Let $s \geq 0$ and $1 \leq q \leq \infty$. For every $f \in X$,
\[
  \|f\|_{A^s_q} \leq 2^s \, \|f\|^{\mathrm{dy}}_{A^s_q}
  \qquad \text{and} \qquad
  \|f\|^{\mathrm{dy}}_{A^s_q} \leq 4 \cdot 2^s \, \|f\|_{A^s_q} .
\]
In particular $f \in A^s_q$ if and only if $\|f\|^{\mathrm{dy}}_{A^s_q} < \infty$.
\end{theorem}

\begin{proof}
\leanok
\uses{lem:error-basic}
For $q < \infty$ raise both sides to the power $q$: $\|f\|_{A^s_q}^q = \sum_n (n+1)^{sq-1}
E_n(f)^q$. The errors decrease, so \Cref{lem:condensation} with $\gamma = sq - 1$ and
$b_n = E_n(f)^q$ compares this series with $E_0(f)^q + \sum_k 2^{ksq} E_{2^k}(f)^q$ in both
directions, and $E_0(f) = \|f\|$. Taking $q$-th roots costs at most the factors stated,
using $(x + y)^{1/q} \leq x^{1/q} + y^{1/q}$ and, for the second bound,
$2^{(k+1)s} = 2^s 2^{ks}$ and $E_1(f) \leq \|f\|$. For $q = \infty$ use the supremum form of
\Cref{lem:condensation}.
\end{proof}

\begin{proposition}[Linear structure]
\label{prop:approx-linear}
\lean{ApproximationScheme.approxNorm_smul, ApproximationScheme.enorm_le_approxNorm,
  ApproximationScheme.approxNorm_eq_zero_iff, ApproximationScheme.approxNormDyadic_add_le,
  ApproximationScheme.approxNorm_add_le, ApproximationScheme.memApprox_of_mem,
  ApproximationScheme.approxSpace, ApproximationScheme.mem_approxSpace}
\leanok
\uses{thm:dyadic-norm, lem:error-basic}
Let $s \geq 0$ and $1 \leq q \leq \infty$. For all $f, g \in X$, $c \in \R$ and $m \in \N$:
\begin{enumerate}
  \item $\|c f\|_{A^s_q} = |c| \, \|f\|_{A^s_q}$;
  \item $\|f\| \leq \|f\|_{A^s_q}$, and $\|f\|_{A^s_q} = 0$ only for $f = 0$;
  \item $\|f + g\|^{\mathrm{dy}}_{A^s_q} \leq 2 \cdot 2^s \bigl(\|f\|^{\mathrm{dy}}_{A^s_q} +
    \|g\|^{\mathrm{dy}}_{A^s_q}\bigr)$ and
    $\|f + g\|_{A^s_q} \leq 8 \cdot 8^s \bigl(\|f\|_{A^s_q} + \|g\|_{A^s_q}\bigr)$;
  \item $A_m \subseteq A^s_q$.
\end{enumerate}
In particular $A^s_q$ is a linear subspace of $X$.
\end{proposition}

\begin{proof}
\leanok
The first item follows from $E_n(cf) = |c| E_n(f)$, the second from the term $n = 0$, which
is $E_0(f) = \|f\|$. For the third, $E_{2^{k+1}}(f + g) \leq E_{2^k}(f) + E_{2^k}(g)$ by
\Cref{lem:error-basic}, so $2^{(k+1)s} E_{2^{k+1}}(f+g) \leq 2^s (2^{ks} E_{2^k}(f) +
2^{ks} E_{2^k}(g))$; the term $k = 0$ is at most $\|f + g\|$, and Minkowski's inequality
gives the dyadic bound. Pietsch's form follows by \Cref{thm:dyadic-norm}. For the fourth,
$E_n(a) = 0$ for $a \in A_m$ and $n \geq m$, so only finitely many terms are nonzero.
\end{proof}

The constants are not optimised. The quasi-triangle inequality is all that is needed: Mathlib
has no class of quasi-normed spaces, so $A^s_q$ is a linear subspace of $X$ together with a
function into $[0, \infty]$, in the way Mathlib treats $L_p$ through \texttt{eLpNorm}.

\begin{lemma}[Approximation spaces lie in the closure]
\label{lem:approx-closure}
\lean{ApproximationScheme.approxSeq_pow_le,
  ApproximationScheme.tendsto_approxError_pow_zero_of_memApprox,
  ApproximationScheme.mem_closure_iUnion_of_memApprox,
  ApproximationScheme.approxSpace_subset_closure}
\leanok
\uses{def:dyadic-norm, lem:error-tendsto}
Let $s > 0$ and $1 \leq q \leq \infty$. For every $f \in A^s_q$ and $k \in \N$,
$E_{2^k}(f) \leq 2^{-ks} \|f\|^{\mathrm{dy}}_{A^s_q}$, and $f$ lies in the closure of
$\bigcup_n A_n$.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:dyadic-norm}
The bound is $2^{ks} E_{2^k}(f) \leq \|(2^{js} E_{2^j}(f))_j\|_{\ell_q}$. It tends to zero,
and \Cref{lem:error-tendsto} applies.
\end{proof}

\begin{theorem}[Embeddings between approximation spaces]
\label{thm:approx-embedding}
\lean{ApproximationScheme.approxNormDyadic_mono_exponent,
  ApproximationScheme.approxNormDyadic_le_of_lt, ApproximationScheme.MemApprox.mono,
  ApproximationScheme.approxSpace_mono, ApproximationScheme.MemApprox.lqNorm_one_lt_top}
\leanok
\uses{def:dyadic-norm, lem:lq-basic, thm:dyadic-norm}
Let $1 \leq q, r \leq \infty$, let $s, t \in \R$ and $f \in X$.
\begin{enumerate}
  \item If $q \leq r$, then $\|f\|^{\mathrm{dy}}_{A^s_r} \leq \|f\|^{\mathrm{dy}}_{A^s_q}$.
  \item If $t < s$, then $\|f\|^{\mathrm{dy}}_{A^t_r} \leq (1 - 2^{-(s-t)})^{-1}
    \|f\|^{\mathrm{dy}}_{A^s_q}$.
  \item If $0 \leq t$, and $t < s$ or $t = s$ and $q \leq r$, then $A^s_q \subseteq A^t_r$.
\end{enumerate}
\end{theorem}

\begin{proof}
\leanok
The first item is the monotonicity of the $\ell_q$ norms in $q$, since the dyadic weights do
not depend on $q$; Pietsch's weights do, which is why the dyadic form is used. For the
second, $2^{kt} E_{2^k}(f) = 2^{-k(s-t)} \, 2^{ks} E_{2^k}(f) \leq 2^{-k(s-t)}
\|(2^{js} E_{2^j}(f))_j\|_{\ell_q}$, and $\|(2^{-k(s-t)})_k\|_{\ell_r} \leq (1 -
2^{-(s-t)})^{-1}$. The third follows by \Cref{thm:dyadic-norm}.
\end{proof}

\begin{lemma}[Functoriality]
\label{lem:approx-functorial}
\lean{ApproximationScheme.approxSeq_map_le, ApproximationScheme.approxNorm_map_le,
  ApproximationScheme.approxNormDyadic_map_le, ApproximationScheme.MemApprox.map,
  ApproximationScheme.approxSpace_map_le}
\leanok
\uses{lem:error-map, def:approx-space, def:dyadic-norm}
Let $A$ and $B$ be approximation schemes in $X$ and $Y$, and let $T : X \to Y$ be a bounded
linear map with $T(A_n) \subseteq B_n$ for all $n$. Then for all $s \in \R$,
$1 \leq q \leq \infty$ and $f \in X$,
\[
  \|Tf\|_{B^s_q} \leq \|T\| \, \|f\|_{A^s_q}
  \qquad \text{and} \qquad
  \|Tf\|^{\mathrm{dy}}_{B^s_q} \leq \|T\| \, \|f\|^{\mathrm{dy}}_{A^s_q} .
\]
\end{lemma}

\begin{proof}
\leanok
Apply \Cref{lem:error-map} to every term, and $\|Tf\| \leq \|T\| \|f\|$.
\end{proof}

\chapter{Lifting and the Nikolskii transfer theorem}
\label{ch:lifting}

Throughout, $A$ is an approximation scheme in $X$, and $\iota : Y \to X$ is a bounded linear
map. One hypothesis covers two classical inequalities. If $\iota$ is the embedding of a
space of smooth functions, it is a Bernstein inequality; if $\iota$ is the inclusion
$L_r \hookrightarrow L_p$, it is a Nikolskii inequality.

\begin{definition}[Lifting bound]
\label{def:lift-bound}
\lean{ApproximationScheme.LiftBound}
\leanok
\uses{def:scheme}
Let $\beta, C \in \R$. The map $\iota$ satisfies the \emph{lifting bound} of exponent $\beta$
and constant $C$ for $A$ if for every $n \geq 1$ and every $a \in A_n$ there is $y \in Y$ with
$\iota y = a$ and $\|y\| \leq C \, n^\beta \, \|a\|$.
\end{definition}

\begin{lemma}[Near-best dyadic approximants]
\label{lem:near-best}
\lean{ApproximationScheme.DyadicApprox, ApproximationScheme.exists_dyadicApprox,
  ApproximationScheme.DyadicApprox.sub_mem,
  ApproximationScheme.DyadicApprox.enorm_sub_succ_le,
  ApproximationScheme.DyadicApprox.tendsto}
\leanok
\uses{lem:error-basic, lem:scheme-basic}
Let $f \in X$ and let $(t_k)_k$ be a sequence in $(0, \infty)$. There are $a_k \in A_{2^k}$
with $\|f - a_k\| \leq E_{2^k}(f) + t_k$ for every $k$. Every such sequence satisfies
$a_{k+1} - a_k \in A_{2^{k+2}}$; if $(t_k)$ decreases, then
$\|a_{k+1} - a_k\| \leq 2 (E_{2^k}(f) + t_k)$; and if $E_{2^k}(f) \to 0$ and $t_k \to 0$, then
$a_k \to f$.
\end{lemma}

\begin{proof}
\leanok
The approximants exist by the definition of the infimum. The rest is
\Cref{lem:scheme-basic}(5), the triangle inequality and the monotonicity of the errors.
\end{proof}

The tolerances $t_k$ replace best approximants, which need not exist. The proofs below take
$t_k = \|f\| \, 2^{-k(\sigma+1)}$ for a suitable $\sigma$, so that
$\sum_k 2^{k\sigma} t_k \leq 2 \|f\|$.

\begin{theorem}[Lifting theorem]
\label{thm:lift}
\lean{ApproximationScheme.exists_preimage_of_liftBound, ApproximationScheme.liftConst,
  ApproximationScheme.DyadicApprox.exists_lift,
  ApproximationScheme.DyadicApprox.exists_preimage, ApproximationScheme.tolerance}
\leanok
\uses{def:lift-bound, lem:near-best, def:dyadic-norm}
Let $Y$ be a Banach space, let $\beta \geq 0$ and $C \geq 0$, and let $\iota$ satisfy the
lifting bound of exponent $\beta$ and constant $C$ for $A$. For every $f \in X$ with
$\sum_k 2^{k\beta} E_{2^k}(f) < \infty$ there is $y \in Y$ with $\iota y = f$ and
\[
  \|y\| \leq (3C + 6C \cdot 4^\beta) \Bigl(\|f\| + \sum_{k=0}^\infty 2^{k\beta}
    E_{2^k}(f)\Bigr) .
\]
\end{theorem}

\begin{proof}
\leanok
For $f = 0$ take $y = 0$. Otherwise choose near-best approximants $a_k$ with tolerances
$t_k = \|f\| 2^{-k(\beta+1)}$ by \Cref{lem:near-best}. Lift $a_0 \in A_1$ to $y_0$ and
$a_{k+1} - a_k \in A_{2^{k+2}}$ to $d_k$, so that $\|y_0\| \leq C \|a_0\| \leq 3C\|f\|$
and $\|d_k\| \leq C \, 2^{(k+2)\beta} \cdot 2(E_{2^k}(f) + t_k) = 2C \, 4^\beta \,
2^{k\beta} (E_{2^k}(f) + t_k)$. The series $\sum_k \|d_k\|$ converges, so
$y = y_0 + \sum_k d_k$ exists in $Y$. By continuity of $\iota$, $\iota y$ is the limit of
$\iota(y_0 + \sum_{j<k} d_j) = a_k$, which is $f$ since $E_{2^k}(f) \to 0$. Adding the bounds
gives the constant.
\end{proof}

The same construction gives the next theorem, the transfer of approximation spaces along an
injective map with a Nikolskii inequality.

\begin{theorem}[Nikolskii transfer theorem]
\label{thm:nikolskii-transfer}
\lean{ApproximationScheme.exists_preimage_of_nikolskii,
  ApproximationScheme.MemApprox.exists_preimage_of_nikolskii,
  ApproximationScheme.nikolskiiConst}
\leanok
\uses{thm:lift, lem:hardy, thm:approx-embedding, def:comap}
Let $Y$ be a Banach space, let $\iota$ be injective, let $C \geq 0$ and $0 \leq \alpha < s$,
and let $\iota$ satisfy the lifting bound of exponent $\alpha$ and constant $C$ for $A$. Let
$B$ be the scheme $B_n = \iota^{-1}(A_n)$ in $Y$, and let $1 \leq q \leq \infty$. For every
$f \in A^s_q$ there is $y \in Y$ with $\iota y = f$ and
\[
  \|y\|^{\mathrm{dy}}_{B^{s-\alpha}_q} \leq K \, \|f\|^{\mathrm{dy}}_{A^s_q},
  \qquad
  K = 3C + 6C \cdot 4^\alpha (1 - 2^{-(s-\alpha)})^{-1} + 4C \cdot 4^\alpha .
\]
\end{theorem}

\begin{proof}
\leanok
By \Cref{thm:approx-embedding}, $\sum_k 2^{k\alpha} E_{2^k}(f) < \infty$, so the
construction of \Cref{thm:lift} applies, now with tolerances $t_k = \|f\| 2^{-k(s+1)}$. The
partial lifts $z_k = y_0 + \sum_{j<k} d_j$ satisfy $\iota z_k = a_k \in A_{2^k}$, so
$z_k \in B_{2^k}$ and
\[
  E^B_{2^k}(y) \leq \|y - z_k\| \leq \sum_{i=0}^\infty \|d_{k+i}\|
  \leq 2C \, 4^\alpha \sum_{i=0}^\infty 2^{(k+i)\alpha} \bigl(E_{2^{k+i}}(f) + t_{k+i}\bigr).
\]
Multiply by $2^{k(s-\alpha)}$ and apply the first Hardy inequality (\Cref{lem:hardy}) with
exponent $s - \alpha > 0$: the weights combine to $2^{k(s-\alpha)} 2^{k\alpha} = 2^{ks}$,
and $\|(2^{ks} t_k)_k\|_{\ell_q} \leq 2\|f\|$.
\end{proof}

\section{Jackson and Bernstein inequalities}

\begin{definition}[Jackson inequality]
\label{def:jackson}
\lean{ApproximationScheme.JacksonBound}
\leanok
\uses{def:error}
Let $r, C \in \R$. The map $\iota$ satisfies \emph{Jackson's inequality} of order $r$ and
constant $C$ for $A$ if $E_n(\iota y) \leq C \, n^{-r} \|y\|$ for every $n \geq 1$ and every
$y \in Y$.
\end{definition}

\begin{lemma}[Jackson's inequality gives $Y \subseteq A^r_\infty$]
\label{lem:jackson-top}
\lean{ApproximationScheme.approxNormDyadic_top_le_of_jackson,
  ApproximationScheme.memApprox_top_of_jackson}
\leanok
\uses{def:jackson, def:dyadic-norm, thm:dyadic-norm}
Let $\iota$ satisfy Jackson's inequality of order $r \geq 0$ and constant $C \geq 0$. Then
$\|\iota y\|^{\mathrm{dy}}_{A^r_\infty} \leq (\|\iota\| + C) \|y\|$ for every $y \in Y$; in
particular $\iota y \in A^r_\infty$.
\end{lemma}

\begin{proof}
\leanok
$2^{kr} E_{2^k}(\iota y) \leq C \|y\|$ for every $k$, and $\|\iota y\| \leq \|\iota\| \|y\|$.
\end{proof}

\begin{definition}[Smoothness space]
\label{def:smoothness}
\lean{ApproximationScheme.SmoothnessSpace}
\leanok
\uses{def:jackson, def:lift-bound}
A \emph{smoothness space} of order $r > 0$ for $A$ is a normed space $Y$ with an injective
bounded linear map $\iota : Y \to X$ that satisfies Jackson's inequality of order $r$ and the
lifting bound of exponent $r$ for $A$, both with nonnegative constants.
\end{definition}

The lifting bound is then Bernstein's inequality $\|\iota^{-1} a\|_Y \leq C n^r \|a\|$ for
$a \in A_n$. Sobolev spaces of order $r$ inside $L_p$ are the motivating example.

\begin{proposition}[Smoothness spaces between $A^r_1$ and $A^r_\infty$]
\label{prop:smoothness}
\lean{ApproximationScheme.SmoothnessSpace.memApprox_top,
  ApproximationScheme.SmoothnessSpace.exists_eq_of_memApprox_one,
  ApproximationScheme.SmoothnessSpace.exists_eq_of_lt}
\leanok
\uses{def:smoothness, lem:jackson-top, thm:lift, thm:approx-embedding}
Let $(Y, \iota)$ be a smoothness space of order $r$ for $A$. Then $\iota(Y) \subseteq
A^r_\infty$. If $Y$ is a Banach space, then $A^r_1 \subseteq \iota(Y)$, and
$A^s_q \subseteq \iota(Y)$ for every $s > r$ and $1 \leq q \leq \infty$.
\end{proposition}

\begin{proof}
\leanok
The first inclusion is \Cref{lem:jackson-top}, the second \Cref{thm:lift} with $\beta = r$.
The third follows from the second and $A^s_q \subseteq A^r_1$ (\Cref{thm:approx-embedding}).
\end{proof}

\chapter{Interpolation and the representation theorem}
\label{ch:interpolation}

Throughout, $\iota : Y \to X$ is a bounded linear map. The $K$-functional of Peetre measures
how well $f \in X$ is approximated by elements of $Y$ of controlled norm; the real
interpolation spaces are defined from it~\cite{BerghLofstrom}.

\begin{definition}[$K$-functional]
\label{def:k-functional}
\lean{ApproximationScheme.kFunctional}
\leanok
For $t \in [0, \infty]$ and $f \in X$,
\[
  K(t, f) = \inf_{y \in Y} \bigl(\|f - \iota y\| + t \|y\|\bigr) .
\]
\end{definition}

\begin{lemma}[Properties of the $K$-functional]
\label{lem:k-functional}
\lean{ApproximationScheme.kFunctional_le_enorm, ApproximationScheme.kFunctional_mono,
  ApproximationScheme.kFunctional_add_le, ApproximationScheme.kFunctional_smul_le}
\leanok
\uses{def:k-functional}
For all $t \leq u$ in $[0, \infty]$, $f, g \in X$ and $c \in \R$: $K(t, f) \leq \|f\|$,
$K(t, f) \leq K(u, f)$, $K(t, f + g) \leq K(t, f) + K(t, g)$ and
$K(t, c f) \leq |c| \, K(t, f)$.
\end{lemma}

\begin{proof}
\leanok
Take $y = 0$ for the first bound; the others hold for each $y$ before the infimum.
\end{proof}

\begin{definition}[Real interpolation space]
\label{def:interpolation}
\lean{ApproximationScheme.interpSeq, ApproximationScheme.interpNorm,
  ApproximationScheme.MemInterp, ApproximationScheme.interpSpace,
  ApproximationScheme.interpNorm_add_le, ApproximationScheme.enorm_le_interpNorm}
\leanok
\uses{def:k-functional, def:lq-norm, lem:k-functional}
Let $\theta, \rho \in \R$ and $1 \leq q \leq \infty$. For $f \in X$,
\[
  \|f\|_{\theta, q} = \|f\| + \bigl\| \bigl(2^{k\rho\theta} K(2^{-k\rho}, f)\bigr)_{k \in \N}
  \bigr\|_{\ell_q} .
\]
The \emph{real interpolation space} $(X, Y)_{\theta, q}$ consists of the $f \in X$ with
$\|f\|_{\theta, q} < \infty$. It is a linear subspace of $X$, and $\|\cdot\|_{\theta, q}$
satisfies the triangle inequality.
\end{definition}

The textbook definition samples $K(t, f)$ on the grid $t = 2^{-k}$, which is $\rho = 1$;
the representation theorem produces the grid $t = 2^{-kr}$. Grids for different $\rho > 0$
give equivalent norms, since $K(t, f)$ increases and $K(t, f)/t$ decreases in $t$; this is
not formalised. The term $\|f\|$ replaces the negative indices $k < 0$ of the textbook
definition.

\begin{lemma}[Errors and the $K$-functional]
\label{lem:error-k-functional}
\lean{ApproximationScheme.approxSeq_le_kFunctional_of_jackson}
\leanok
\uses{def:k-functional, def:jackson}
Let $A$ be an approximation scheme in $X$, and let $\iota$ satisfy Jackson's inequality of
order $r$ and constant $C \geq 0$. For every $n \geq 1$ and $f \in X$,
$E_n(f) \leq \max(1, C) \, K(n^{-r}, f)$.
\end{lemma}

\begin{proof}
\leanok
\uses{lem:error-basic}
For every $y \in Y$, $E_n(f) \leq \|f - \iota y\| + E_n(\iota y) \leq \|f - \iota y\| + C
n^{-r} \|y\|$.
\end{proof}

The representation theorem identifies approximation spaces with interpolation spaces, as in
DeVore and Lorentz~\cite[Chapter~7]{DeVoreLorentz}.

\begin{theorem}[Representation theorem]
\label{thm:representation}
\lean{ApproximationScheme.approxNormDyadic_le_interpNorm_of_jackson,
  ApproximationScheme.interpNorm_le_approxNormDyadic_of_bernstein,
  ApproximationScheme.reprConst, ApproximationScheme.memApprox_iff_memInterp,
  ApproximationScheme.approxSpace_eq_interpSpace,
  ApproximationScheme.SmoothnessSpace.memApprox_iff_memInterp}
\leanok
\uses{lem:error-k-functional, def:interpolation, lem:near-best, lem:hardy,
  def:lift-bound, def:dyadic-norm}
Let $A$ be an approximation scheme in $X$, let $r \in \R$ and $C_1, C_2 \geq 0$, and let
$\iota$ satisfy Jackson's inequality of order $r$ and constant $C_1$ and the lifting bound of
exponent $r$ and constant $C_2$. Let $0 < s < r$ and $1 \leq q \leq \infty$. Then for every
$f \in X$,
\[
  \|f\|^{\mathrm{dy}}_{A^s_q} \leq \max(1, C_1) \, \|f\|_{s/r, q}
  \qquad \text{and} \qquad
  \|f\|_{s/r, q} \leq K \, \|f\|^{\mathrm{dy}}_{A^s_q},
\]
where $K = 3 + (3C_2 + 4C_2 \cdot 4^r)(1 - 2^{-(r-s)})^{-1}$ and the interpolation norm is
taken on the grid $\rho = r$. In particular $A^s_q = (X, Y)_{s/r, q}$.
\end{theorem}

\begin{proof}
\leanok
For the first inequality apply \Cref{lem:error-k-functional} with $n = 2^k$: then
$n^{-r} = 2^{-kr}$ and $2^{ks} = 2^{kr \cdot s/r}$. For the second choose near-best
approximants $a_k$ (\Cref{lem:near-best}) and lift them as in \Cref{thm:lift}: the partial
lifts $z_k = y_0 + \sum_{j<k} d_j$ satisfy $\iota z_k = a_k$ and
$\|z_k\| \leq 3C_2 \|f\| + 2C_2 4^r \sum_{j<k} 2^{jr} (E_{2^j}(f) + t_j)$. Hence
\[
  2^{ks} K(2^{-kr}, f) \leq 2^{ks} \|f - a_k\| + 2^{-k(r-s)} \|z_k\| ,
\]
and the second Hardy inequality (\Cref{lem:hardy}) with exponent $r - s > 0$ bounds the
partial sums. The tolerances are made arbitrarily small at the end. Only finitely many
lifts enter each $z_k$, so $Y$ need not be complete.
\end{proof}

\begin{corollary}[Schemes with the same Jackson and Bernstein inequalities]
\label{cor:equal-spaces}
\lean{ApproximationScheme.approxSpace_eq_of_jackson_bernstein}
\leanok
\uses{thm:representation}
Let $A$ and $B$ be approximation schemes in $X$, let $r \in \R$, and let $\iota$ satisfy
Jackson's inequality of order $r$ and the lifting bound of exponent $r$ for $A$ and for $B$,
with nonnegative constants. Then $A^s_q = B^s_q$ for all $0 < s < r$ and
$1 \leq q \leq \infty$.
\end{corollary}

\begin{proof}
\leanok
Both are $(X, Y)_{s/r, q}$ by \Cref{thm:representation}.
\end{proof}

This is how approximation by trigonometric polynomials, splines and wavelets are seen to
define the same spaces.

\chapter{Block decompositions and projections}
\label{ch:decompositions}

Throughout, $A$ is an approximation scheme in $X$. Approximation spaces can be described by
block decompositions $f = \sum_k g_k$ into blocks $g_k \in A_{2^k}$~\cite[Chapter~7]{DeVoreLorentz}.
Each block is a whole level; atomic decompositions refine the blocks into localised atoms with
coefficients, and wavelet decompositions expand them in a basis.

\begin{definition}[Block decomposition norm]
\label{def:block-decomposition}
\lean{ApproximationScheme.blockDecompNorm}
\leanok
\uses{def:scheme, def:lq-norm}
Let $s \in \R$ and $1 \leq q \leq \infty$. For $f \in X$, $\|f\|^{\mathrm{dec}}_{s,q}$ is
the infimum of $\|(2^{ks} \|g_k\|)_k\|_{\ell_q}$ over all sequences $(g_k)_k$ with
$g_k \in A_{2^k}$ for every $k$ and $\sum_k g_k = f$, the series converging in $X$.
\end{definition}

\begin{theorem}[Block decompositions]
\label{thm:block-decomposition}
\lean{ApproximationScheme.approxNormDyadic_le_of_hasSum,
  ApproximationScheme.approxNormDyadic_le_blockDecompNorm,
  ApproximationScheme.blockDecompNorm_le_approxNormDyadic,
  ApproximationScheme.memApprox_iff_blockDecompNorm_lt_top,
  ApproximationScheme.DyadicApprox.blocks}
\leanok
\uses{def:block-decomposition, def:dyadic-norm, lem:hardy, lem:near-best, lem:approx-closure}
Let $s > 0$ and $1 \leq q \leq \infty$. For every $f \in X$,
\[
  \|f\|^{\mathrm{dy}}_{A^s_q} \leq 2 (1 - 2^{-s})^{-1} \, \|f\|^{\mathrm{dec}}_{s,q} .
\]
If $X$ is a Banach space, then for every $f \in X$,
\[
  \|f\|^{\mathrm{dec}}_{s,q} \leq (3 \cdot 2^s + 4 \cdot 4^s) \, \|f\|^{\mathrm{dy}}_{A^s_q},
\]
and $f \in A^s_q$ if and only if $\|f\|^{\mathrm{dec}}_{s,q} < \infty$.
\end{theorem}

\begin{proof}
\leanok
\uses{lem:scheme-basic}
Let $f = \sum_k g_k$ with $g_k \in A_{2^k}$. Since $\sum_{j<k} 2^j < 2^k$, the partial sum
$\sum_{j<k} g_j$ lies in $A_{2^k}$, so $E_{2^k}(f) \leq T_k = \sum_{i \geq 0} \|g_{k+i}\|$,
and also $\|f\| \leq T_0$. The first Hardy inequality bounds $\|(2^{ks} T_k)_k\|_{\ell_q}$,
which gives the first inequality. Conversely, take near-best approximants $a_k$ with
tolerances $\|f\| 2^{-k(s+1)}$ and the blocks $g_0 = 0$, $g_1 = a_0$, $g_{k+2} =
a_{k+1} - a_k \in A_{2^{k+2}}$. Their partial sums are the $a_k$, which converge to $f$ by
\Cref{lem:approx-closure}, and $\|g_{k+2}\| \leq 2(E_{2^k}(f) + t_k)$ gives the bound.
\end{proof}

\section{Projection schemes}

\begin{definition}[Projection scheme]
\label{def:projection-scheme}
\lean{ApproximationScheme.ProjectionScheme}
\leanok
\uses{def:scheme}
A \emph{projection scheme} for $A$ is a sequence of bounded linear maps $P_n : X \to X$ with
$P_n(X) \subseteq A_n$, $P_n a = a$ for every $a \in A_n$, and $\|P_n\| \leq M$ for all $n$,
for some $M \geq 0$.
\end{definition}

\begin{lemma}[Lebesgue inequality]
\label{lem:lebesgue}
\lean{ApproximationScheme.ProjectionScheme.approxSeq_le_enorm_sub,
  ApproximationScheme.ProjectionScheme.enorm_sub_le_approxSeq,
  ApproximationScheme.ProjectionScheme.tendsto_of_dense}
\leanok
\uses{def:projection-scheme, def:error, lem:error-tendsto}
Let $(P_n)$ be a projection scheme for $A$ with bound $M$. For every $n \in \N$ and
$f \in X$,
\[
  E_n(f) \leq \|f - P_n f\| \leq (1 + M) \, E_n(f) .
\]
If $\bigcup_n A_n$ is dense in $X$, then $P_n f \to f$ for every $f \in X$.
\end{lemma}

\begin{proof}
\leanok
For $a \in A_n$, $f - P_n f = (f - a) - P_n(f - a)$. The last statement follows from
$E_n(f) \to 0$.
\end{proof}

\begin{definition}[Dyadic blocks]
\label{def:blocks}
\lean{ApproximationScheme.ProjectionScheme.block,
  ApproximationScheme.ProjectionScheme.blockNorm}
\leanok
\uses{def:projection-scheme, def:lq-norm}
For a projection scheme $(P_n)$, $k \in \N$ and $f \in X$, the \emph{dyadic block} is
$\Delta_k f = P_{2^{k+1}} f - P_{2^k} f$. For $s \in \R$ and $1 \leq q \leq \infty$,
\[
  \|f\|^{\mathrm{bl}}_{s,q} = \|P_1 f\| + \bigl\| \bigl(2^{ks} \|\Delta_k f\|\bigr)_k
  \bigr\|_{\ell_q} .
\]
\end{definition}

\begin{theorem}[Characterisation by dyadic blocks]
\label{thm:blocks}
\lean{ApproximationScheme.ProjectionScheme.enorm_block_le,
  ApproximationScheme.ProjectionScheme.blockNorm_le_approxNormDyadic,
  ApproximationScheme.ProjectionScheme.enorm_sub_le_tsum_block,
  ApproximationScheme.ProjectionScheme.approxNormDyadic_le_blockNorm,
  ApproximationScheme.ProjectionScheme.memApprox_iff_blockNorm_lt_top}
\leanok
\uses{def:blocks, lem:lebesgue, lem:hardy, def:dyadic-norm}
Let $(P_n)$ be a projection scheme for $A$ with bound $M$, let $s \in \R$ and
$1 \leq q \leq \infty$. For every $f \in X$,
\[
  \|f\|^{\mathrm{bl}}_{s,q} \leq (M + 2(1 + M)) \, \|f\|^{\mathrm{dy}}_{A^s_q} .
\]
If $P_n f \to f$, then
\[
  \|f\|^{\mathrm{dy}}_{A^s_q} \leq \bigl(1 + 2(1 - 2^{-s})^{-1}\bigr) \,
  \|f\|^{\mathrm{bl}}_{s,q} .
\]
In particular, if $\bigcup_n A_n$ is dense in $X$ and $s > 0$, then $f \in A^s_q$ if and
only if $\|f\|^{\mathrm{bl}}_{s,q} < \infty$.
\end{theorem}

\begin{proof}
\leanok
By \Cref{lem:lebesgue}, $\|\Delta_k f\| \leq \|f - P_{2^{k+1}} f\| + \|f - P_{2^k} f\| \leq
2(1 + M) E_{2^k}(f)$, and $\|P_1 f\| \leq M \|f\|$. Conversely, if $P_n f \to f$, then
$f - P_{2^k} f = \sum_{i \geq 0} \Delta_{k+i} f$, so $E_{2^k}(f) \leq \sum_{i \geq 0}
\|\Delta_{k+i} f\|$ and $\|f\| \leq \|P_1 f\| + \sum_{i \geq 0} \|\Delta_i f\|$; the first
Hardy inequality concludes.
\end{proof}

\section{Synthesis with decay and atoms}

Atoms of a function space lie in no fixed set of approximants. Their blocks satisfy a weaker
condition: they are small and approximable at a rate $\sigma$ from their own level on.

\begin{definition}[Linear scheme]
\label{def:linear-scheme}
\lean{ApproximationScheme.IsLinear, ApproximationScheme.isLinear_ofSubmodules,
  ApproximationScheme.approxSeq_add_le_of_isLinear,
  ApproximationScheme.approxSeq_sum_le_of_isLinear}
\leanok
\uses{def:scheme, def:error}
An approximation scheme is \emph{linear} if $a + b \in A_n$ for all $n \in \N$ and
$a, b \in A_n$. For a linear scheme, $E_n(f + g) \leq E_n(f) + E_n(g)$ for all $n$, $f$, $g$.
\end{definition}

\begin{theorem}[Synthesis with decay]
\label{thm:synthesis-decay}
\lean{ApproximationScheme.approxNormDyadic_le_of_hasSum_decay,
  ApproximationScheme.dyadicWeight_mul_dyadicWeight_sub,
  ApproximationScheme.approxSeq_le_approxSeq_add_enorm}
\leanok
\uses{def:linear-scheme, def:dyadic-norm, lem:hardy}
Let $A$ be a linear scheme, $\sigma \in \R$, $C \in [0, \infty]$ and $1 \leq q \leq \infty$.
Let $f = \sum_j g_j$ in $X$, and let $(\varepsilon_j)_j$ in $[0, \infty]$ satisfy
$\|g_j\| \leq \varepsilon_j$ for every $j$ and $E_{2^k}(g_j) \leq C \, 2^{-(k-j)\sigma}
\varepsilon_j$ for all $j \leq k$. Then for every $s \in \R$,
\[
  \|f\|^{\mathrm{dy}}_{A^s_q} \leq \bigl(2 (1 - 2^{-s})^{-1} + C (1 - 2^{-(\sigma - s)})^{-1}\bigr)
  \bigl\|(2^{js} \varepsilon_j)_j\bigr\|_{\ell_q} ,
\]
a finite constant for $0 < s < \sigma$.
\end{theorem}

\begin{proof}
\leanok
Split $f$ at level $k$: $E_{2^k}(f) \leq \sum_{j \leq k} E_{2^k}(g_j) + \|\sum_{j > k} g_j\|$
by linearity. The first sum is at most $C \, 2^{-k\sigma} \sum_{j \leq k} 2^{j\sigma}
\varepsilon_j$, and after multiplication by $2^{ks}$ the second Hardy inequality with exponent
$\sigma - s$ applies. The second term is at most $\sum_{i \geq 0} \varepsilon_{k+i}$, and the
first Hardy inequality with exponent $s$ applies; it also bounds $\|f\| \leq \sum_j
\varepsilon_j$.
\end{proof}

This is the pattern of the proof of \Cref{thm:moduli-coincidence}. For atoms the size
$\varepsilon_j$ is that of a whole level, measured by the coefficients.

\begin{definition}[Atomic system and the sequence space $b^s_{p,q}$]
\label{def:atomic-system}
\lean{ApproximationScheme.AtomicSystem, ApproximationScheme.bNorm, ENNReal.finLqNorm}
\leanok
\uses{def:scheme, def:error, def:lq-norm}
Let $M_j$ be finite sets, $1 \leq p \leq \infty$ and $\delta, \sigma \in \R$. An \emph{atomic
system} for $A$ consists of sets $\mathcal{A}_{j,m} \subseteq X$ of admissible atoms, $j \in
\N$, $m \in M_j$, each containing $0$, and a constant $C < \infty$ such that for every $j$,
every choice $a_m \in \mathcal{A}_{j,m}$ and all real coefficients $c = (c_m)_{m \in M_j}$,
\[
  \Bigl\| \sum_m c_m a_m \Bigr\| \leq C \, 2^{-j\delta} \|c\|_{\ell_p}
\]
and, for every $k \geq j$,
\[
  E_{2^k}\Bigl(\sum_m c_m a_m\Bigr) \leq 2^{-(k-j)\sigma} \, C \, 2^{-j\delta} \|c\|_{\ell_p} .
\]
For coefficients $\lambda = (\lambda_{j,m})$ and $s, q$ as before,
$\|\lambda\|_{b^s_{p,q}} = \|(2^{j(s-\delta)} \|\lambda_{j,\cdot}\|_{\ell_p})_j\|_{\ell_q}$.
\end{definition}

On a domain of dimension $d$ in $L_p$, the positions are the dyadic cubes of side $2^{-j}$ and
$\delta = d/p$. The size bound expresses the normalisation and the bounded overlap of the
supports, the decay bound the smoothness $\sigma$ of the atoms relative to the scheme, and
$b^s_{p,q}$ is the sequence space of Schneider and Vyb\'{\i}ral~\cite{SchneiderVybiral}.

\begin{theorem}[Atomic synthesis]
\label{thm:atomic-synthesis}
\lean{ApproximationScheme.AtomicSystem.approxNormDyadic_le_bNorm,
  ApproximationScheme.AtomicSystem.memApprox_of_bNorm_lt_top,
  ApproximationScheme.AtomicSystem.synthConst}
\leanok
\uses{def:atomic-system, thm:synthesis-decay}
Let $A$ be a linear scheme with an atomic system as in \Cref{def:atomic-system}, let
$0 < s < \sigma$ and $1 \leq q \leq \infty$. Let $a_{j,m} \in \mathcal{A}_{j,m}$, let
$\lambda_{j,m} \in \R$, and let $f = \sum_j \sum_m \lambda_{j,m} a_{j,m}$, the series over $j$
converging in $X$. Then
\[
  \|f\|^{\mathrm{dy}}_{A^s_q} \leq \bigl(2 (1 - 2^{-s})^{-1} + (1 - 2^{-(\sigma - s)})^{-1}\bigr)
  \, C \, \|\lambda\|_{b^s_{p,q}} .
\]
\end{theorem}

\begin{proof}
\leanok
Apply \Cref{thm:synthesis-decay} to the levels $g_j = \sum_m \lambda_{j,m} a_{j,m}$ with
$\varepsilon_j = C \, 2^{-j\delta} \|\lambda_{j,\cdot}\|_{\ell_p}$; then
$2^{js} \varepsilon_j = C \, 2^{j(s - \delta)} \|\lambda_{j,\cdot}\|_{\ell_p}$.
\end{proof}

\begin{lemma}[Block decompositions with bounds]
\label{lem:exists-blocks}
\lean{ApproximationScheme.exists_blocks, ApproximationScheme.DyadicApprox.hasSum_blocks,
  ApproximationScheme.DyadicApprox.lqNorm_blocks_le,
  ApproximationScheme.DyadicApprox.tsum_enorm_blocks_ne_top}
\leanok
\uses{thm:block-decomposition}
Let $X$ be a Banach space, $s > 0$, $1 \leq q \leq \infty$ and $f \in A^s_q$. There are
$g_k \in A_{2^k}$ with $f = \sum_k g_k$, $\sum_k \|g_k\| < \infty$ and
$\|(2^{ks} \|g_k\|)_k\|_{\ell_q} \leq (3 \cdot 2^s + 4 \cdot 4^s) \|f\|^{\mathrm{dy}}_{A^s_q}$.
\end{lemma}

\begin{proof}
\leanok
These are the blocks of near-best approximants from the proof of
\Cref{thm:block-decomposition}.
\end{proof}

\begin{theorem}[Atomic analysis]
\label{thm:atomic-analysis}
\lean{ApproximationScheme.AtomicSystem.StableExpansion,
  ApproximationScheme.AtomicSystem.exists_atomic_decomposition,
  ApproximationScheme.AtomicSystem.memApprox_iff_exists_atomic}
\leanok
\uses{def:atomic-system, lem:exists-blocks, thm:atomic-synthesis}
Let $X$ be a Banach space with an atomic system for $A$, and let $C' \geq 0$ be such that every
$x \in A_{2^j}$ is $\sum_m c_m a_m$ with $a_m \in \mathcal{A}_{j,m}$ and
$\|c\|_{\ell_p} \leq C' \, 2^{j\delta} \|x\|$. Let $s > 0$ and $1 \leq q \leq \infty$. Then
every $f \in A^s_q$ is $\sum_j \sum_m \lambda_{j,m} a_{j,m}$ with $a_{j,m} \in
\mathcal{A}_{j,m}$ and
\[
  \|\lambda\|_{b^s_{p,q}} \leq C' (3 \cdot 2^s + 4 \cdot 4^s) \|f\|^{\mathrm{dy}}_{A^s_q} .
\]
If moreover $A$ is linear, $C' < \infty$ and $s < \sigma$, then $f \in A^s_q$ if and only if
$f$ is such an atomic sum with $\|\lambda\|_{b^s_{p,q}} < \infty$.
\end{theorem}

\begin{proof}
\leanok
Expand every block of \Cref{lem:exists-blocks} in atoms of its level; then
$2^{j(s-\delta)} \|\lambda_{j,\cdot}\|_{\ell_p} \leq C' \, 2^{js} \|g_j\|$. The converse is
\Cref{thm:atomic-synthesis}.
\end{proof}

Local bases such as Haar functions and hat functions on dyadic grids have the expansion property
of \Cref{thm:atomic-analysis}.

\chapter{Traces and intrinsic norms}
\label{ch:traces-abstract}

Throughout, $A$ is an approximation scheme in $X$ and $B$ one in $Y$. Two questions about
function spaces on domains become statements about approximation spaces: the restriction of
functions to a hypersurface, a map defined only on smooth functions, and the comparison of the
intrinsic norm of a domain with the norm of restrictions.

\begin{definition}[Trace bound]
\label{def:trace-bound}
\lean{ApproximationScheme.TraceBound, ApproximationScheme.TraceBound.ofSubmodule,
  ApproximationScheme.TraceBound.ofSubmodule_T}
\leanok
\uses{def:scheme}
Let $\beta \in \R$. A \emph{trace bound} of exponent $\beta$ is a linear map $T : X \to Y$ with
$T(A_n) \subseteq B_n$ for every $n$ and a constant $C < \infty$ with
$\|T a\| \leq C n^\beta \|a\|$ for all $n \geq 1$ and $a \in A_n$.
\end{definition}

Only the values of $T$ on $\bigcup_n A_n$ matter, and a linear map defined on the span of the
approximants extends linearly to $X$. The model is the restriction of functions to a
hyperplane, defined on continuous functions and unbounded on $L_p$, with $\beta = 1/p$.

\begin{lemma}[The trace]
\label{lem:trace-ext}
\lean{ApproximationScheme.TraceBound.ext, ApproximationScheme.canonicalApprox,
  ApproximationScheme.TraceBound.tendsto_ext, ApproximationScheme.TraceBound.tendsto_edist_apply,
  ApproximationScheme.TraceBound.cauchySeq_apply, ApproximationScheme.TraceBound.ext_eq_of_mem,
  ApproximationScheme.TraceBound.ext_add, ApproximationScheme.TraceBound.ext_smul,
  ApproximationScheme.TraceBound.extLinear, ApproximationScheme.TraceBound.enorm_ext_le}
\leanok
\uses{def:trace-bound, lem:near-best, def:dyadic-norm}
Let $T$ be a trace bound of exponent $\beta \geq 0$ and let $Y$ be a Banach space. For every
$f \in A^\beta_1$ there is $\operatorname{Tr} f \in Y$ such that $T c_k \to \operatorname{Tr} f$
for all $\gamma \in \N$ and all $c_k \in A_{2^{k+\gamma}}$ with
$2^{k\beta} \|f - c_k\| \to 0$. The map $\operatorname{Tr}$ is linear on $A^\beta_1$, agrees
with $T$ on every $A_m$, and
$\|\operatorname{Tr} f\| \leq (3C + 6C \cdot 4^\beta) \|f\|^{\mathrm{dy}}_{A^\beta_1}$.
\end{lemma}

\begin{proof}
\leanok
For near-best approximants $a_k$, $\|T a_{k+1} - T a_k\| \leq C \, 2^{(k+2)\beta} \cdot
2 (E_{2^k}(f) + t_k)$, which is summable; so $T a_k$ is a Cauchy sequence. Two admissible
sequences differ by $c_k - c'_k \in A_{2^{k+\gamma+\gamma'+1}}$ with
$\|T(c_k - c'_k)\| \lesssim 2^{k\beta} (\|f - c_k\| + \|f - c'_k\|) \to 0$. Linearity follows
by applying this to $a_k(f) + a_k(g)$ and $r a_k(f)$; on $A_m$ take $c_k = a$. The bound is the
telescoping sum, as in \Cref{thm:lift}.
\end{proof}

\begin{theorem}[Trace transfer]
\label{thm:trace-transfer}
\lean{ApproximationScheme.TraceBound.approxNormDyadic_ext_le,
  ApproximationScheme.TraceBound.memApprox_ext, ApproximationScheme.traceTransferConst,
  ApproximationScheme.TraceBound.edist_apply_ext_le}
\leanok
\uses{lem:trace-ext, lem:hardy}
Let $T$ be a trace bound of exponent $\beta$ with constant $C$, let $Y$ be a Banach space, and
let $0 \leq \beta < s$ and $1 \leq q \leq \infty$. For every $f \in A^s_q$,
\[
  \|\operatorname{Tr} f\|^{\mathrm{dy}}_{B^{s-\beta}_q} \leq
  \bigl(3C + 6C \cdot 4^\beta (1 - 2^{-(s-\beta)})^{-1} + 4C \cdot 4^\beta\bigr)
  \|f\|^{\mathrm{dy}}_{A^s_q} .
\]
\end{theorem}

\begin{proof}
\leanok
Since $T a_k \in B_{2^k}$, $E^B_{2^k}(\operatorname{Tr} f) \leq \sum_{i \geq 0}
\|T a_{k+i+1} - T a_{k+i}\|$. The proof of \Cref{thm:nikolskii-transfer} then applies word for
word; that theorem is the case $T = \iota^{-1}$ on the approximants.
\end{proof}

\begin{theorem}[Surjectivity of the trace]
\label{thm:trace-surjective}
\lean{ApproximationScheme.TraceBound.RightInverseBound,
  ApproximationScheme.TraceBound.exists_ext_eq, ApproximationScheme.traceSurjConst}
\leanok
\uses{thm:trace-transfer, lem:exists-blocks, thm:block-decomposition}
Let $X$ and $Y$ be Banach spaces, $T$ a trace bound of exponent $\beta \geq 0$, and let
$\gamma \in \N$ and $C' < \infty$ be such that every $b \in B_{2^k}$ is $T a$ for some
$a \in A_{2^{k+\gamma}}$ with $\|a\| \leq C' \, 2^{-k\beta} \|b\|$. Let $s > 0$ and
$1 \leq q \leq \infty$. For every $g \in B^s_q$ there is $f \in A^{s+\beta}_q$ with
$\operatorname{Tr} f = g$ and
\[
  \|f\|^{\mathrm{dy}}_{A^{s+\beta}_q} \leq 2 (1 - 2^{-(s+\beta)})^{-1} \, 2^{\gamma(s+\beta)}
  \, C' \, (3 \cdot 2^s + 4 \cdot 4^s) \, \|g\|^{\mathrm{dy}}_{B^s_q} .
\]
\end{theorem}

\begin{proof}
\leanok
Lift the blocks $b_j \in B_{2^j}$ of \Cref{lem:exists-blocks} for $g$ to $a_j \in
A_{2^{j+\gamma}}$ and put $f = \sum_j a_j$. The blocks $a_j$, placed at level $j + \gamma$,
form a block decomposition of $f$, and $2^{(j+\gamma)(s+\beta)} \|a_j\| \leq
2^{\gamma(s+\beta)} C' \, 2^{js} \|b_j\|$, so \Cref{thm:block-decomposition} gives the bound.
The partial sums $c_K = \sum_{j < K} a_j$ satisfy $2^{K\beta} \|f - c_K\| \leq C' \sum_{m \geq 0}
\|b_{m+K}\| \to 0$, so $T c_K \to \operatorname{Tr} f$ by \Cref{lem:trace-ext}, while
$T c_K = \sum_{j < K} b_j \to g$.
\end{proof}

This is where block decompositions are used: they turn the right inverse on approximants into
a right inverse of the trace.

\begin{definition}[Approximation-preserving extension]
\label{def:approx-extension}
\lean{ApproximationScheme.ApproxExtension, ApproximationScheme.restrictionNorm}
\leanok
\uses{def:error}
Let $R : X \to Y$ be a bounded linear map. An \emph{approximation-preserving extension} is a
map $E : Y \to X$, not necessarily linear, with $R(E g) = g$, $\|E g\| \leq C \|g\|$ and
$E^A_{2^{k+\gamma}}(E g) \leq C \, E^B_{2^k}(g)$ for all $g \in Y$ and $k \in \N$, for some
$C < \infty$ and $\gamma \in \N$. The \emph{restriction quasi-norm} of $g \in Y$ is
$\inf\{\|f\|^{\mathrm{dy}}_{A^s_q} : R f = g\}$.
\end{definition}

\begin{theorem}[Intrinsic and restriction quasi-norms]
\label{thm:retract}
\lean{ApproximationScheme.approxNormDyadic_extension_le, ApproximationScheme.extensionConst,
  ApproximationScheme.approxNormDyadic_le_iInf_restrict,
  ApproximationScheme.restrictionNorm_le_approxNormDyadic,
  ApproximationScheme.memApprox_iff_exists_restrict}
\leanok
\uses{def:approx-extension, lem:approx-functorial}
Let $R : X \to Y$ be a bounded linear map with $R(A_n) \subseteq B_n$ for every $n$, with an
approximation-preserving extension $E$. Let $s \geq 0$ and $1 \leq q \leq \infty$. Then for
every $g \in Y$,
\[
  \|g\|^{\mathrm{dy}}_{B^s_q} \leq \inf_{R f = g} \|R\| \, \|f\|^{\mathrm{dy}}_{A^s_q}
  \qquad \text{and} \qquad
  \|E g\|^{\mathrm{dy}}_{A^s_q} \leq C \bigl(1 + (\gamma + 1) 2^{\gamma s}\bigr)
  \|g\|^{\mathrm{dy}}_{B^s_q} ,
\]
and $g \in B^s_q$ if and only if $g = R f$ for some $f \in A^s_q$.
\end{theorem}

\begin{proof}
\leanok
The first inequality is \Cref{lem:approx-functorial}. For the second, the terms $k < \gamma$
are at most $2^{\gamma s} C \|g\|$, and for $k = j + \gamma$,
$2^{ks} E^A_{2^k}(E g) \leq 2^{\gamma s} C \, 2^{js} E^B_{2^j}(g)$.
\end{proof}

For a domain $\Omega \subseteq \R^d$, $X = L_p(\R^d)$, $Y = L_p(\Omega)$ and $R$ the restriction,
the quasi-norm of $B^s_q$ uses only $g$ on $\Omega$: it is intrinsic. The theorem reduces the
existence of an intrinsic norm to an approximation-preserving extension, which DeVore and
Sharpley~\cite{DeVoreSharpley} construct on Lipschitz domains from near-best local
polynomials. The extension may be
nonlinear on purpose: in parts of the quasi-Banach range no linear extension exists.

\chapter{Moduli of smoothness and approximation}
\label{ch:moduli-abstract}

Classical function spaces are defined by moduli of smoothness, approximation spaces by errors
of approximation. A Jackson inequality and an inverse inequality make them coincide. This
chapter states this once for every scheme; the examples are the trigonometric polynomials, the
piecewise polynomials, splines and wavelets.

\begin{definition}[Modulus compatible with a scheme]
\label{def:modulus-data}
\lean{ApproximationScheme.ModulusData, ApproximationScheme.ModulusData.modNorm,
  ApproximationScheme.ModulusData.modulus_zero, ApproximationScheme.ModulusData.modulus_sum_le}
\leanok
\uses{def:scheme, def:error}
Let $\theta \in \R$. A \emph{modulus compatible with $A$ of inverse order $\theta$} is a family
$\omega(f, k) \in [0, \infty]$, $f \in X$, $k \in \N$ (the modulus of smoothness at the scale
$2^{-k}$), together with a level $k_0 \in \N$ and constants $C_0, C_J, C_B < \infty$ such that for
all $f, g \in X$ and $j, k \in \N$
\[
  \omega(f + g, k) \leq \omega(f, k) + \omega(g, k), \qquad \omega(f, k) \leq C_0 \|f\|,
\]
\[
  E_{2^k}(f) \leq C_J \, \omega(f, k) \quad (k \geq k_0), \qquad
  \omega(a, k) \leq C_B \, 2^{-(k - j)_+ \theta} \|a\| \quad (a \in A_{2^j}) .
\]
The \emph{classical quasi-norm} is
$\|f\|^{\omega}_{s,q} = \|f\| + \|(2^{ks} \omega(f, k))_k\|_{\ell_q}$.
\end{definition}

\begin{theorem}[Coincidence of approximation and moduli]
\label{thm:modulus-coincidence}
\lean{ApproximationScheme.ModulusData.directConst,
  ApproximationScheme.ModulusData.directConst_ne_top,
  ApproximationScheme.ModulusData.approxNormDyadic_le_modNorm,
  ApproximationScheme.ModulusData.dyadicWeight_neg_sub_two_le,
  ApproximationScheme.ModulusData.inverseConst,
  ApproximationScheme.ModulusData.inverseConst_ne_top,
  ApproximationScheme.ModulusData.modNorm_le_approxNormDyadic,
  ApproximationScheme.ModulusData.memApprox_iff_modNorm_lt_top}
\leanok
\uses{def:modulus-data, def:dyadic-norm, lem:hardy}
Let $\omega$ be a modulus compatible with $A$ of inverse order $\theta$ and $1 \leq q \leq
\infty$. Then for every $f \in X$ and $s \geq 0$,
$\|f\|^{\mathrm{dy}}_{A^s_q} \leq (1 + \sum_{k < k_0} 2^{ks} + C_J) \|f\|^{\omega}_{s,q}$, and
for $0 \leq s < \theta$,
\[
  \|f\|^{\omega}_{s,q} \leq \bigl(1 + 3 C_0 + C_B (3 + 6 \cdot 4^{\theta})
  (1 - 2^{-(\theta - s)})^{-1}\bigr) \|f\|^{\mathrm{dy}}_{A^s_q} .
\]
In particular $f \in A^s_q$ if and only if $\|f\|^{\omega}_{s,q} < \infty$, for $0 \leq s < \theta$.
\end{theorem}

\begin{proof}
\leanok
The first bound is Jackson's inequality for $k \geq k_0$ and $E_{2^k}(f) \leq \|f\|$ for
$k < k_0$. For the second,
let $a_k \in A_{2^k}$ be near-best approximants and write
$f = (f - a_k) + a_0 + \sum_{j < k} (a_{j+1} - a_j)$, where $a_{j+1} - a_j \in A_{2^{j+2}}$. Then
\[
  2^{ks} \omega(f, k) \leq C_0 \, 2^{ks} \|f - a_k\| + C_B \, 2^{-k(\theta - s)} \|a_0\|
  + 2 C_B 4^{\theta} \, 2^{-k(\theta - s)} \sum_{j \leq k} 2^{j(\theta - s)} \,
  2^{js} (E_{2^j}(f) + t_j),
\]
and the last sum is bounded in $\ell_q$ by Hardy's inequality (\Cref{lem:hardy}) with exponent
$\theta - s > 0$.
\end{proof}

\chapter{Besov spaces and their embeddings}
\label{ch:besov}

Throughout, $\Omega$ is a topological space with its Borel $\sigma$-algebra and $\mu$ is a
finite measure on it. For $1 \leq p \leq \infty$, $L_p = L_p(\mu)$ is the space of
complex-valued $p$-integrable functions, and $C_b(\Omega)$ is the space of bounded continuous
complex-valued functions with the supremum norm.

\section{Three facts on $L_p$}

\begin{lemma}[The inclusion $L_r \hookrightarrow L_p$]
\label{lem:inclusion}
\lean{MeasureTheory.Lp.inclusionOfExponentLe, MeasureTheory.Lp.inclusionConst,
  MeasureTheory.Lp.coeFn_inclusionOfExponentLe,
  MeasureTheory.Lp.norm_inclusionOfExponentLe_le,
  MeasureTheory.Lp.norm_inclusionOfExponentLe_apply_le_of_prob,
  MeasureTheory.Lp.inclusionOfExponentLe_injective, MeasureTheory.Lp.inclusionOfExponentLe_comp}
\leanok
Let $1 \leq p \leq r \leq \infty$. The identity of functions defines an injective bounded
linear map $\iota_{r,p} : L_r \to L_p$ with $\|\iota_{r,p}\| \leq
\mu(\Omega)^{1/p - 1/r}$, and $\iota_{t,p} \circ \iota_{r,t} = \iota_{r,p}$ for every
$p \leq t \leq r$.
\end{lemma}

\begin{proof}
\leanok
H\"older's inequality gives $\|f\|_p \leq \mu(\Omega)^{1/p - 1/r} \|f\|_r$. A function that
vanishes almost everywhere does so in both spaces.
\end{proof}

\begin{lemma}[Interpolation between $L_p$ and $L_\infty$]
\label{lem:lp-interpolation}
\lean{MeasureTheory.eLpNorm_le_rpow_mul_eLpNorm_rpow, MeasureTheory.toReal_eLpNorm_le_rpow_mul}
\leanok
Let $0 < p \leq r < \infty$ and $M \geq 0$. For every measurable $f$ with $|f| \leq M$
almost everywhere,
\[
  \|f\|_r \leq M^{1 - p/r} \, \|f\|_p^{p/r} .
\]
\end{lemma}

\begin{proof}
\leanok
Integrate $|f|^r \leq M^{r-p} |f|^p$.
\end{proof}

\begin{lemma}[Bounded continuous functions in $L_\infty$]
\label{lem:bc-linfty}
\lean{BoundedContinuousFunction.norm_toLp_top, BoundedContinuousFunction.isometry_toLp_top,
  BoundedContinuousFunction.isClosed_range_toLp_top}
\leanok
Assume that $\mu(U) > 0$ for every nonempty open $U \subseteq \Omega$. Then
$\|a\|_{L_\infty} = \sup_{x \in \Omega} |a(x)|$ for every $a \in C_b(\Omega)$, and the classes
of bounded continuous functions form a closed subspace of $L_\infty$.
\end{lemma}

\begin{proof}
\leanok
If $|a(x)| > c$ for some $x$, the open set $\{|a| > c\}$ has positive measure, so
$\|a\|_{L_\infty} \geq c$. The map $C_b(\Omega) \to L_\infty$ is therefore an isometry, and
its range is complete, hence closed.
\end{proof}

\section{Schemes of functions and Besov spaces}

\begin{definition}[Scheme of functions]
\label{def:function-scheme}
\lean{FunctionScheme, FunctionScheme.toLp, FunctionScheme.mem_toLp}
\leanok
\uses{def:scheme, def:comap}
A \emph{scheme of functions} on $\Omega$ is an approximation scheme $S$ in $C_b(\Omega)$. For
$1 \leq p \leq \infty$, the classes in $L_p$ of the functions in $S_n$ form an approximation
scheme $S^{(p)}$ in $L_p$.
\end{definition}

Taking the approximants continuous and bounded makes them elements of every $L_p$ at once,
so that one scheme serves all exponents.

\begin{lemma}[Compatibility with the inclusions]
\label{lem:function-scheme-inclusion}
\lean{FunctionScheme.image_toLp_of_le, FunctionScheme.comap_toLp_of_le,
  BoundedContinuousFunction.inclusionOfExponentLe_toLp}
\leanok
\uses{def:function-scheme, lem:inclusion}
Let $S$ be a scheme of functions and $1 \leq p \leq r \leq \infty$. For every $n \in \N$,
$\iota_{r,p}(S^{(r)}_n) = S^{(p)}_n$ and $\iota_{r,p}^{-1}(S^{(p)}_n) = S^{(r)}_n$.
\end{lemma}

\begin{proof}
\leanok
The map $\iota_{r,p}$ sends the class of $a$ in $L_r$ to its class in $L_p$, and it is
injective.
\end{proof}

\begin{definition}[Besov space]
\label{def:besov}
\lean{FunctionScheme.besovError, FunctionScheme.besovNorm, FunctionScheme.MemBesov,
  FunctionScheme.besovSpace, FunctionScheme.besovSpaceSup}
\leanok
\uses{def:function-scheme, def:approx-space, prop:approx-linear}
Let $S$ be a scheme of functions, $s \geq 0$ and $1 \leq p, q \leq \infty$. The
\emph{Besov space} is $\Besov{s}{p}{q}(\Omega, S) = A^s_q(L_p, S^{(p)})$, the approximation
space of $L_p$ for the scheme $S^{(p)}$.
\end{definition}

On the torus $\T^d$ with $S_n$ the trigonometric polynomials of degree $< n$, these are the
classical periodic Besov spaces (\Cref{thm:moduli-coincidence}).

\begin{proposition}[Elementary embeddings]
\label{prop:besov-elementary}
\lean{FunctionScheme.MemBesov.mono, FunctionScheme.besovNorm_inclusion_le,
  FunctionScheme.MemBesov.inclusion, FunctionScheme.MemBesov.of_sup}
\leanok
\uses{def:besov, thm:approx-embedding, lem:approx-functorial,
  lem:function-scheme-inclusion}
Let $S$ be a scheme of functions and $1 \leq p, q, q' \leq \infty$.
\begin{enumerate}
  \item If $0 \leq t$, and $t < s$ or $t = s$ and $q \leq q'$, then
    $\Besov{s}{p}{q} \subseteq \Besov{t}{p}{q'}$.
  \item If $p \leq r$, then $\iota_{r,p} f \in \Besov{s}{p}{q}$ and
    $\|\iota_{r,p} f\|_{\Besov{s}{p}{q}} \leq \|\iota_{r,p}\| \, \|f\|_{\Besov{s}{r}{q}}$ for
    every $f \in \Besov{s}{r}{q}$.
  \item The class in $L_p$ of every $a \in A^s_q(C_b(\Omega), S)$ lies in
    $\Besov{s}{p}{q}$.
\end{enumerate}
\end{proposition}

\begin{proof}
\leanok
The first item is \Cref{thm:approx-embedding}. The others are \Cref{lem:approx-functorial}
for $T = \iota_{r,p}$, using \Cref{lem:function-scheme-inclusion}, and for the map
$C_b(\Omega) \to L_p$, of norm at most $\mu(\Omega)^{1/p}$.
\end{proof}

\section{The Nikolskii inequality and the Besov embeddings}

\begin{definition}[Nikolskii property]
\label{def:nikolskii-property}
\lean{FunctionScheme.HasNikolskii, nikolskiiExponent, nikolskiiExponent_nonneg}
\leanok
\uses{def:function-scheme}
Let $S$ be a scheme of functions and $D \in \R$. The scheme has the \emph{Nikolskii property}
of dimension $D$ if for every $1 \leq p \leq \infty$ there is $C \geq 0$ with
\[
  \sup_{x \in \Omega} |a(x)| \leq C \, n^{D/p} \, \|a\|_{L_p}
  \qquad \text{for all } n \geq 1 \text{ and } a \in S_n .
\]
For $1 \leq p \leq r \leq \infty$ put $\alpha(D, p, r) = D(1/p - 1/r)$.
\end{definition}

\begin{lemma}[Nikolskii inequality]
\label{lem:nikolskii-inequality}
\lean{FunctionScheme.HasNikolskii.eLpNorm_le, FunctionScheme.HasNikolskii.liftBound}
\leanok
\uses{def:nikolskii-property, lem:lp-interpolation, def:lift-bound,
  lem:function-scheme-inclusion}
Let $S$ have the Nikolskii property of dimension $D$, and let $1 \leq p \leq r \leq \infty$.
There is $C \geq 0$ with $\|a\|_{L_r} \leq C \, n^{\alpha(D,p,r)} \|a\|_{L_p}$ for all
$n \geq 1$ and $a \in S_n$. In particular $\iota_{r,p}$ satisfies the lifting bound of
exponent $\alpha(D, p, r)$ for $S^{(p)}$.
\end{lemma}

\begin{proof}
\leanok
For $r = \infty$, $\|a\|_{L_\infty} \leq \sup |a|$. For $r < \infty$, \Cref{lem:lp-interpolation}
with $M = \sup|a| \leq C n^{D/p} \|a\|_p$ gives $\|a\|_r \leq C^{1-p/r} n^{(D/p)(1-p/r)}
\|a\|_p$, and $(D/p)(1 - p/r) = \alpha(D, p, r)$. The class of $a$ in $L_r$ lifts the class
in $L_p$.
\end{proof}

\begin{theorem}[Besov embeddings]
\label{thm:besov-embedding}
\lean{FunctionScheme.exists_inclusion_eq_of_nikolskii,
  FunctionScheme.MemBesov.exists_inclusion_eq_of_nikolskii,
  FunctionScheme.MemBesov.exists_inclusion_eq_top,
  FunctionScheme.MemBesov.exists_boundedContinuous}
\leanok
\uses{lem:nikolskii-inequality, thm:nikolskii-transfer, lem:function-scheme-inclusion,
  lem:approx-closure, lem:bc-linfty}
Let $S$ have the Nikolskii property of dimension $D \geq 0$, let $1 \leq p \leq r \leq
\infty$ and $1 \leq q \leq \infty$, and let $s > \alpha = \alpha(D, p, r)$.
\begin{enumerate}
  \item There is $K < \infty$ such that for every $f \in \Besov{s}{p}{q}$ there is $g \in L_r$
    with $\iota_{r,p} g = f$ and $\|g\|^{\mathrm{dy}}_{\Besov{s-\alpha}{r}{q}} \leq K
    \|f\|^{\mathrm{dy}}_{\Besov{s}{p}{q}}$.
  \item If $\mu(U) > 0$ for every nonempty open $U \subseteq \Omega$ and $s > D/p$, then every
    $f \in \Besov{s}{p}{q}$ is the class of a bounded continuous function.
\end{enumerate}
\end{theorem}

\begin{proof}
\leanok
For the first item apply \Cref{thm:nikolskii-transfer} to $\iota = \iota_{r,p}$, which is
injective and satisfies the lifting bound of exponent $\alpha$ by
\Cref{lem:nikolskii-inequality}; the preimage scheme is $S^{(r)}$ by
\Cref{lem:function-scheme-inclusion}, and $L_r$ is complete. For the second take
$r = \infty$: then $g \in \Besov{s - D/p}{\infty}{q}$ with $s - D/p > 0$, so $g$ lies in the
closure of $\bigcup_n S^{(\infty)}_n$ (\Cref{lem:approx-closure}), which is contained in the
closed set of classes of bounded continuous functions (\Cref{lem:bc-linfty}).
\end{proof}

The limiting case $s = \alpha$ is out of reach of this method, since the Nikolskii transfer
theorem needs $\alpha < s$.

\chapter{Sobolev spaces and their embeddings}
\label{ch:sobolev}

The setting is that of \Cref{ch:besov}. A Sobolev space enters through its Jackson and
Bernstein inequalities, as a smoothness space for the scheme in $L_p$.

\begin{definition}[Sobolev space for a scheme]
\label{def:sobolev-scheme}
\lean{FunctionScheme.SobolevSpace}
\leanok
\uses{def:smoothness, def:function-scheme}
Let $S$ be a scheme of functions and $1 \leq p \leq \infty$. A \emph{Sobolev space} of order
$s > 0$ for $S$ in $L_p$ is a smoothness space of order $s$ for $S^{(p)}$: a normed space $V$
with an injective bounded linear map $\iota : V \to L_p$, Jackson's inequality
$E_n(\iota v)_p \leq C n^{-s} \|v\|$ and Bernstein's inequality, every $a \in S^{(p)}_n$ being
$\iota v$ with $\|v\| \leq C n^s \|a\|_p$.
\end{definition}

\begin{theorem}[Sobolev embeddings]
\label{thm:sobolev-embedding}
\lean{FunctionScheme.SobolevSpace.exists_eq_inclusion,
  FunctionScheme.SobolevSpace.exists_eq_inclusion',
  FunctionScheme.SobolevSpace.exists_boundedContinuous}
\leanok
\uses{def:sobolev-scheme, thm:besov-embedding, prop:smoothness}
Let $S$ have the Nikolskii property of dimension $D \geq 0$, and let $1 \leq p \leq r \leq
\infty$. Let $(V, \iota)$ be a Sobolev space of order $s$ for $S$ in $L_p$.
\begin{enumerate}
  \item Let $(V', \iota')$ be a Sobolev space of order $t$ for $S$ in $L_r$ with $V'$ a Banach
    space, and let $t - D/r < s - D/p$. For every $v \in V$ there is $v' \in V'$ with
    $\iota_{r,p}(\iota' v') = \iota v$.
  \item If $\mu(U) > 0$ for every nonempty open $U \subseteq \Omega$ and $s > D/p$, then
    $\iota v$ is the class of a bounded continuous function for every $v \in V$.
\end{enumerate}
\end{theorem}

\begin{proof}
\leanok
The condition $t - D/r < s - D/p$ is $t < s - \alpha(D, p, r)$, and it implies $s > \alpha$
since $t > 0$. By \Cref{prop:smoothness}, $\iota v \in \Besov{s}{p}{\infty}$. By
\Cref{thm:besov-embedding}, $\iota v = \iota_{r,p} g$ with $g \in
\Besov{s-\alpha}{r}{\infty}$, and $g \in \iota'(V')$ by \Cref{prop:smoothness}, as
$s - \alpha > t$. The second item is \Cref{thm:besov-embedding}(2) with $q = \infty$.
\end{proof}

The limiting cases $t - D/r = s - D/p$ and $s = D/p$ are not reached: the passage from $V$
to $\Besov{s}{p}{\infty}$ and back from $\Besov{t}{r}{1}$ to $V'$ loses the fine index at
both ends.

\chapter{Spectral schemes}
\label{ch:spectral}

Throughout, $H$ is a complex Hilbert space with a Hilbert basis $(e_i)_{i \in I}$, and
$\hat f(i) = \langle e_i, f \rangle$ are the coefficients of $f \in H$.

\begin{definition}[Spectral scheme]
\label{def:spectral}
\lean{SpectralScheme, SpectralScheme.span, SpectralScheme.toScheme, SpectralScheme.weight}
\leanok
\uses{def:comap}
A \emph{level function} is a map $\lambda : I \to \N$ such that $\{i : \lambda(i) < n\}$ is
finite for every $n$. The sets $A_n = \spn\{e_i : \lambda(i) < n\}$ form an approximation
scheme in $H$, the \emph{spectral scheme} of $(e_i, \lambda)$.
\end{definition}

On the torus, $e_k$ is the Fourier basis and $\lambda(k) = \max_j |k_j|$. On a compact
manifold, $e_i$ would be the eigenfunctions of the Laplacian and $\lambda$ a function of the
eigenvalue.

\begin{definition}[Sobolev embedding of $\ell_2$]
\label{def:sobolev-embed}
\lean{SpectralScheme.sobolevEmbed, SpectralScheme.repr_sobolevEmbed,
  SpectralScheme.sobolevEmbed_injective, SpectralScheme.diagInvWeight}
\leanok
\uses{def:spectral}
For $s \geq 0$ let $J_s : \ell_2(I) \to H$ be the bounded injective linear map
\[
  J_s c = \sum_{i \in I} (1 + \lambda(i))^{-s} c_i \, e_i .
\]
\end{definition}

\begin{lemma}[Range of the Sobolev embedding]
\label{lem:sobolev-range}
\lean{SpectralScheme.mem_range_sobolevEmbed_iff, SpectralScheme.sobolevEmbed_weighted}
\leanok
\uses{def:sobolev-embed}
Let $s \geq 0$. An element $f \in H$ lies in $J_s(\ell_2(I))$ if and only if
$\sum_i (1 + \lambda(i))^{2s} |\hat f(i)|^2 < \infty$.
\end{lemma}

\begin{proof}
\leanok
The preimage is $c_i = (1 + \lambda(i))^s \hat f(i)$.
\end{proof}

\begin{proposition}[Jackson and Bernstein inequalities]
\label{prop:spectral-jackson-bernstein}
\lean{SpectralScheme.jacksonBound, SpectralScheme.liftBound, SpectralScheme.toSmoothnessSpace,
  SpectralScheme.truncate, SpectralScheme.repr_eq_zero_of_mem_span}
\leanok
\uses{def:sobolev-embed, def:jackson, def:lift-bound, def:smoothness}
Let $s \geq 0$. For every $n \geq 1$ and $c \in \ell_2(I)$, $E_n(J_s c) \leq n^{-s} \|c\|$;
and every $a \in A_n$ is $J_s c$ for some $c$ with $\|c\| \leq n^s \|a\|$. For $s > 0$,
$(\ell_2(I), J_s)$ is a smoothness space of order $s$ for the spectral scheme, with
constants $1$.
\end{proposition}

\begin{proof}
\leanok
The truncation $P_n f = \sum_{\lambda(i) < n} \hat f(i) e_i$ lies in $A_n$, and by
Parseval's identity $\|J_s c - P_n J_s c\|^2 = \sum_{\lambda(i) \geq n} (1 +
\lambda(i))^{-2s} |c_i|^2 \leq n^{-2s} \|c\|^2$. An element $a \in A_n$ has $\hat a(i) = 0$
for $\lambda(i) \geq n$, and $c_i = (1 + \lambda(i))^s \hat a(i)$ satisfies
$|c_i| \leq n^s |\hat a(i)|$.
\end{proof}

\begin{proposition}[Spectral projections]
\label{prop:spectral-projections}
\lean{SpectralScheme.projectionScheme, SpectralScheme.truncateCLM,
  SpectralScheme.closure_iUnion_toScheme, SpectralScheme.repr_block,
  SpectralScheme.enorm_block_sq}
\leanok
\uses{def:spectral, def:projection-scheme, def:blocks}
The truncations $P_n$ form a projection scheme with bound $1$ for the spectral scheme, the
union $\bigcup_n A_n$ is dense in $H$, and for every $k \in \N$ and $f \in H$
\[
  \Delta_k f = \sum_{2^k \leq \lambda(i) < 2^{k+1}} \hat f(i) \, e_i ,
  \qquad
  \|\Delta_k f\|^2 = \sum_{2^k \leq \lambda(i) < 2^{k+1}} |\hat f(i)|^2 .
\]
\end{proposition}

\begin{proof}
\leanok
Truncation is the orthogonal projection onto $A_n$. The union contains the span of the
basis, which is dense. The formulas are Parseval's identity.
\end{proof}

\begin{theorem}[Coefficient characterisation]
\label{thm:spectral-coefficients}
\lean{SpectralScheme.memApprox_iff_blockNorm_lt_top, SpectralScheme.memApprox_two_iff,
  SpectralScheme.mem_l2_iff, SpectralScheme.tsum_ite_block_eq}
\leanok
\uses{prop:spectral-projections, thm:blocks, lem:sobolev-range}
Let $s > 0$ and $1 \leq q \leq \infty$. An element $f \in H$ lies in $A^s_q$ if and only if
\[
  \Bigl(\sum_{\lambda(i) = 0} |\hat f(i)|^2\Bigr)^{1/2} + \Bigl\| \Bigl(2^{ks}
    \Bigl(\sum_{2^k \leq \lambda(i) < 2^{k+1}} |\hat f(i)|^2\Bigr)^{1/2}\Bigr)_k
    \Bigr\|_{\ell_q} < \infty .
\]
It lies in $A^s_2$ if and only if $\sum_i (1 + \lambda(i))^{2s} |\hat f(i)|^2 < \infty$; that
is, $A^s_2 = J_s(\ell_2(I))$.
\end{theorem}

\begin{proof}
\leanok
The first statement is \Cref{thm:blocks} with the formulas of
\Cref{prop:spectral-projections}, since $P_1 f = \sum_{\lambda(i) = 0} \hat f(i) e_i$. For
$q = 2$ the square of the block norm is comparable with $\sum_i \omega_i |\hat f(i)|^2$,
where $\omega_i = 2^{2ks}$ on the block $2^k \leq \lambda(i) < 2^{k+1}$ and $\omega_i = 1$
for $\lambda(i) = 0$; and $\omega_i \leq (1 + \lambda(i))^{2s} \leq 4^s \omega_i$.
\end{proof}

\chapter{The torus}
\label{ch:torus}

Let $\T = \R / \Z$ with its normalised Haar measure, let $d$ be a finite index set, and let
$\T^d$ carry the product measure, a probability measure charging every nonempty open set.
For $k \in \Z^d$ the trigonometric monomial is $e_k(x) = \exp(2\pi i \sum_j k_j x_j)$, and
$\hat f(k) = \int_{\T^d} f \, \overline{e_k}$ is the Fourier coefficient of $f \in
L_1(\T^d)$. The norm of $h \in \T^d$ is $\|h\| = \max_j \dist(h_j, \Z)$.

In Mathlib the Haar measure on $\T$, over which the Fourier basis of $L_2(\T^d)$ is stated, is
a local instance of one file; the project re-declares it as a scoped instance.

\begin{definition}[Trigonometric polynomials]
\label{def:trig-poly}
\lean{Torus.deg, Torus.boxLt, Torus.card_boxLt_le, Torus.trigMonomial, Torus.trigPoly,
  Torus.trigScheme, TorusHaar.volume_eq_haarAddCircle}
\leanok
\uses{def:function-scheme}
The degree of $k \in \Z^d$ is $\deg k = \max_j |k_j|$, with $\deg k = 0$ for $d = \emptyset$.
The space $\mathcal{T}_n$ of trigonometric polynomials of degree $< n$ is the span of
$\{e_k : \deg k < n\}$ in $C(\T^d)$, a space of dimension at most $(2n)^d$. The spaces
$\mathcal{T}_n$ form a scheme of functions on $\T^d$, the \emph{trigonometric scheme}.
\end{definition}

\begin{lemma}[Products of trigonometric polynomials]
\label{lem:trig-products}
\lean{Torus.trigPoly_mul_le, Torus.mul_mem_trigPoly, Torus.pow_mem_trigPoly,
  Torus.trigMonomial_add}
\leanok
\uses{def:trig-poly}
For all $m, n \in \N$, $S \in \mathcal{T}_m$ and $T \in \mathcal{T}_n$: $S T \in
\mathcal{T}_{m+n}$ and $T^{m+1} \in \mathcal{T}_{(m+1)n}$.
\end{lemma}

\begin{proof}
\leanok
$e_k e_l = e_{k+l}$ and $\deg(k + l) \leq \deg k + \deg l$.
\end{proof}

\begin{lemma}[Supremum norm and $L_2$ norm]
\label{lem:trig-sup-l2}
\lean{Torus.norm_sq_le_card_mul_norm_toLp_sq, Torus.exists_sum_of_mem_trigPoly,
  Torus.norm_toLp_sum_sq}
\leanok
\uses{def:trig-poly}
For every $n \in \N$ and $T \in \mathcal{T}_n$, $\sup |T|^2 \leq (2n)^d \, \|T\|_{L_2}^2$.
\end{lemma}

\begin{proof}
\leanok
Write $T = \sum_{\deg k < n} c_k e_k$. Then $|T(x)| \leq \sum |c_k|$, the Cauchy--Schwarz
inequality bounds this by $(\#\{k : \deg k < n\})^{1/2} (\sum |c_k|^2)^{1/2}$, and
$\sum |c_k|^2 = \|T\|_2^2$ by orthonormality.
\end{proof}

The Nikolskii inequality for trigonometric polynomials~\cite{Nikolskii, Temlyakov} is
usually proved with the Dirichlet or de la Vall\'ee Poussin kernel. The proof below avoids
kernels: it passes to a power of the polynomial and uses \Cref{lem:trig-sup-l2} for it.

\begin{theorem}[Nikolskii inequality on the torus]
\label{thm:torus-nikolskii}
\lean{Torus.norm_le_rpow_mul_eLpNorm, Torus.norm_rpow_le_card_mul_eLpNorm_rpow,
  Torus.norm_pow_le_norm_pow, Torus.norm_toLp_two_pow, Torus.trigScheme_hasNikolskii}
\leanok
\uses{lem:trig-products, lem:trig-sup-l2, def:nikolskii-property, lem:bc-linfty}
For all $n \geq 1$, $T \in \mathcal{T}_n$ and $1 \leq p < \infty$,
\[
  \sup |T| \leq \bigl(2 \lceil p/2 \rceil\bigr)^{d/p} \, n^{d/p} \, \|T\|_{L_p} .
\]
The trigonometric scheme has the Nikolskii property of dimension $d$.
\end{theorem}

\begin{proof}
\leanok
Let $m = \lceil p/2 \rceil$, so $2m \geq p$, and $T \neq 0$. By \Cref{lem:trig-products},
$T^m \in \mathcal{T}_{mn}$, and \Cref{lem:trig-sup-l2} gives
\[
  (\sup|T|)^{2m} \leq (\sup|T^m|)^2 \leq (2mn)^d \int |T|^{2m} ,
\]
and $\int |T|^{2m} \leq (\sup|T|)^{2m-p} \int |T|^p$.
Divide by $(\sup|T|)^{2m-p}$ and take $p$-th roots. For $p = \infty$ the constant is $1$ by
\Cref{lem:bc-linfty}.
\end{proof}

\begin{definition}[Sobolev space $H^s(\T^d)$]
\label{def:torus-sobolev}
\lean{Torus.fourierScheme, Torus.toLp_trigScheme_eq, Torus.sobolevSpace,
  Torus.mem_range_sobolev_iff}
\leanok
\uses{def:spectral, def:sobolev-embed, prop:spectral-jackson-bernstein, def:sobolev-scheme}
The Fourier basis $(e_k)_{k \in \Z^d}$ of $L_2(\T^d)$ with the level $\deg$ is a spectral
scheme whose approximants are the classes of $\mathcal{T}_n$. For $s > 0$, the Sobolev space
$H^s(\T^d)$ consists of the $f \in L_2(\T^d)$ with
\[
  \sum_{k \in \Z^d} (1 + \deg k)^{2s} |\hat f(k)|^2 < \infty ,
\]
the range of $J_s$. With $J_s$ it is a Sobolev space of order $s$ for the trigonometric
scheme in $L_2$, with constants $1$.
\end{definition}

The weight $(1 + \max_j |k_j|)^{2s}$ is equivalent to the more common $(1 + |k|^2)^s$, up to
constants depending on $d$ and $s$; the degree is used because it indexes the trigonometric
polynomials.

\begin{theorem}[Embeddings on the torus]
\label{thm:torus-embeddings}
\lean{Torus.memBesov_exists_inclusion_eq, Torus.memBesov_exists_continuous,
  Torus.sobolev_memBesov_top, Torus.exists_sobolev_of_memBesov,
  Torus.sobolev_exists_inclusion_eq, Torus.sobolev_exists_continuous}
\leanok
\uses{thm:torus-nikolskii, thm:besov-embedding, def:torus-sobolev, prop:smoothness}
Let $1 \leq p \leq r \leq \infty$ and $1 \leq q \leq \infty$.
\begin{enumerate}
  \item If $s > d(1/p - 1/r)$, then every $f \in \Besov{s}{p}{q}(\T^d)$ is $\iota_{r,p} g$ for
    some $g \in \Besov{s - d(1/p - 1/r)}{r}{q}(\T^d)$.
  \item If $s > d/p$, then every $f \in \Besov{s}{p}{q}(\T^d)$ has a continuous
    representative.
  \item If $0 < s < t$, then $H^s(\T^d) \subseteq \Besov{s}{2}{\infty}(\T^d)$ and
    $\Besov{t}{2}{q}(\T^d) \subseteq H^s(\T^d)$.
  \item If $2 \leq r$ and $s > d(1/2 - 1/r)$, then every $f \in H^s(\T^d)$ is
    $\iota_{r,2} g$ for some $g \in \Besov{s - d(1/2 - 1/r)}{r}{\infty}(\T^d)$.
  \item If $s > d/2$, then every $f \in H^s(\T^d)$ has a continuous representative.
\end{enumerate}
\end{theorem}

\begin{proof}
\leanok
The first two items are \Cref{thm:besov-embedding} with \Cref{thm:torus-nikolskii}, the
third is \Cref{prop:smoothness} for $H^s(\T^d)$, and the last two combine both.
\end{proof}

The next theorem is the Littlewood--Paley characterisation of $\Besov{s}{2}{q}(\T^d)$ with
sharp dyadic cut-offs. For $p \neq 2$ it needs smooth cut-offs and a Fourier multiplier
theorem, which Mathlib does not have.

\begin{theorem}[Fourier coefficients]
\label{thm:torus-coefficients}
\lean{Torus.memBesov_two_two_iff, Torus.fourierBlock, Torus.enorm_fourierBlock_sq,
  Torus.memBesov_two_iff_blockNorm_lt_top}
\leanok
\uses{thm:spectral-coefficients, def:torus-sobolev}
Let $s > 0$, $1 \leq q \leq \infty$ and $f \in L_2(\T^d)$. Then $f \in
\Besov{s}{2}{q}(\T^d)$ if and only if
\[
  |\hat f(0)| + \Bigl\| \Bigl(2^{ks} \Bigl(\sum_{2^k \leq \deg m < 2^{k+1}} |\hat f(m)|^2
    \Bigr)^{1/2}\Bigr)_k \Bigr\|_{\ell_q} < \infty ,
\]
and $\Besov{s}{2}{2}(\T^d) = H^s(\T^d)$.
\end{theorem}

\begin{proof}
\leanok
\Cref{thm:spectral-coefficients} for the Fourier basis; the only index of level $0$ is
$k = 0$.
\end{proof}

\section{Traces on the torus}

The trace of a function on $\T^{n+1}$ is its restriction to the face $\{x_0 = 0\} \cong \T^n$,
$f \mapsto f(0, \cdot)$. It is defined on continuous functions and not on $L_p$; the abstract
trace theorems of \Cref{ch:traces-abstract} extend it to Besov spaces.

\begin{definition}[Restriction to the face]
\label{def:face-restriction}
\lean{Torus.face, Torus.restrictFace, Torus.compLinear, Torus.lineThrough,
  Torus.restrictFace_trigMonomial, Torus.restrictFace_mem_trigPoly,
  Torus.compLinear_lineThrough_mem_trigPoly}
\leanok
\uses{def:trig-poly}
For a bounded continuous $a$ on $\T^{n+1}$, its restriction to the face is
$a(0, \cdot) : x' \mapsto a(0, x')$ on $\T^n$. It maps $\mathcal{T}_m$ on $\T^{n+1}$ into
$\mathcal{T}_m$ on $\T^n$, since $e_k(0, x') = e_{(k_1, \dots, k_n)}(x')$; so does the
restriction to every line $t \mapsto a(t, x')$, onto $\mathcal{T}_m$ on $\T^1$.
\end{definition}

\begin{lemma}[Fubini on $\T^{n+1} = \T \times \T^n$]
\label{lem:fubini-cons}
\lean{Torus.lintegral_eq_lintegral_cons}
\leanok
For every measurable $F : \T^{n+1} \to [0, \infty]$,
$\int_{\T^{n+1}} F = \int_{\T^n} \int_{\T} F(t, x') \, dt \, dx'$.
\end{lemma}

\begin{proof}
\leanok
The map $(t, x') \mapsto (t, x')$ is measure preserving from the product measure, and Tonelli's
theorem applies.
\end{proof}

\begin{theorem}[Trace inequality on the torus]
\label{thm:trace-inequality-torus}
\lean{Torus.eLpNorm_restrictFace_le, Torus.enorm_rpow_le_lintegral}
\leanok
\uses{def:face-restriction, lem:fubini-cons, thm:torus-nikolskii}
For all $m \in \N$, $a \in \mathcal{T}_m$ on $\T^{n+1}$ and $1 \leq p < \infty$,
\[
  \|a(0, \cdot)\|_{L_p(\T^n)} \leq \bigl(2 \lceil p/2 \rceil m\bigr)^{1/p} \,
  \|a\|_{L_p(\T^{n+1})} .
\]
\end{theorem}

\begin{proof}
\leanok
For fixed $x'$ the line $t \mapsto a(t, x')$ is a trigonometric polynomial of degree $< m$ in
one variable, so \Cref{thm:torus-nikolskii} with $d = 1$ gives
$|a(0, x')|^p \leq 2 \lceil p/2 \rceil m \int_\T |a(t, x')|^p \, dt$. Integrate over $x'$ and
apply \Cref{lem:fubini-cons}.
\end{proof}

The opposite direction needs a polynomial in $x_0$ that equals $1$ at $0$ and is small in
$L_p$. The normalised Fejér kernel serves, and Parseval's identity replaces the usual pointwise
kernel estimates.

\begin{lemma}[Extension by the Fejér kernel]
\label{lem:fejer-extension}
\lean{Torus.dirichletSum, Torus.fejerFace, Torus.extendFace, Torus.tailMap,
  Torus.fejerFace_cons_zero, Torus.enorm_fejerFace_le, Torus.lintegral_enorm_fejerFace,
  Torus.lintegral_enorm_dirichletSum_sq, Torus.extendFace_mem_trigPoly,
  Torus.restrictFace_extendFace, Torus.eLpNorm_extendFace_le}
\leanok
\uses{def:face-restriction, lem:fubini-cons, lem:trig-products}
Let $m \geq 1$ and $\varphi_m(x) = m^{-2} \bigl|\sum_{j<m} e^{2\pi i j x_0}\bigr|^2$ on
$\T^{n+1}$. For every $b \in \mathcal{T}_m$ on $\T^n$, the function
$E b : x \mapsto \varphi_m(x) \, b(x_1, \dots, x_n)$ lies in $\mathcal{T}_{3m}$, satisfies
$(E b)(0, \cdot) = b$, and for every $1 \leq p < \infty$,
\[
  \|E b\|_{L_p(\T^{n+1})} \leq m^{-1/p} \, \|b\|_{L_p(\T^n)} .
\]
\end{lemma}

\begin{proof}
\leanok
The Dirichlet sum has degree $< m$ and modulus at most $m$, so $\varphi_m$ has degree $< 2m$,
$0 \leq \varphi_m \leq 1$ and $\varphi_m(0, \cdot) = 1$. By Parseval,
$\int \varphi_m = m^{-2} \cdot m = 1/m$, hence $\int \varphi_m^p \leq \int \varphi_m = 1/m$,
and \Cref{lem:fubini-cons} gives $\int |E b|^p = \int \varphi_m^p \cdot \int |b|^p$.
\end{proof}

\begin{theorem}[Traces of Besov spaces on the torus]
\label{thm:trace-torus}
\lean{Torus.traceBound, Torus.traceOnApprox, Torus.traceBound_ext_toLp,
  Torus.traceBound_rightInverse, Torus.trace_besov}
\leanok
\uses{thm:trace-inequality-torus, lem:fejer-extension, thm:trace-transfer,
  thm:trace-surjective}
Let $1 \leq p < \infty$, $1 \leq q \leq \infty$ and $s > 0$. The restriction to the face
extends from trigonometric polynomials to a map $\operatorname{Tr}$ from
$\Besov{s+1/p}{p}{q}(\T^{n+1})$ into $\Besov{s}{p}{q}(\T^n)$, and every
$g \in \Besov{s}{p}{q}(\T^n)$ is $\operatorname{Tr} f$ for some
$f \in \Besov{s+1/p}{p}{q}(\T^{n+1})$.
\end{theorem}

\begin{proof}
\leanok
By \Cref{thm:trace-inequality-torus} the restriction is a trace bound of exponent $1/p$ from
the trigonometric scheme of $\T^{n+1}$ to that of $\T^n$ in $L_p$, and by
\Cref{lem:fejer-extension} with $m = 2^k$ it has a right inverse bound with $\gamma = 2$ and
$C' = 1$. Apply \Cref{thm:trace-transfer} with $s + 1/p$ in place of $s$, and
\Cref{thm:trace-surjective}.
\end{proof}

\chapter{Filtrations and martingale differences}
\label{ch:filtrations}

Throughout, $(\Omega, \mathcal{F}, \mu)$ is a finite measure space and $1 \leq p \leq \infty$.
Conditional expectations are projections of norm at most $1$ on every $L_p$, so a filtration
gives a projection scheme.

\begin{lemma}[Conditional expectation on $L_p$]
\label{lem:condexp-lp}
\lean{MeasureTheory.Lp.condExpLp, MeasureTheory.Lp.coeFn_condExpLp,
  MeasureTheory.Lp.norm_condExpLp_le, MeasureTheory.Lp.condExpLp_mem_lpMeas,
  MeasureTheory.Lp.condExpLp_of_mem}
\leanok
Let $\mathcal{G} \subseteq \mathcal{F}$ be a sub-$\sigma$-algebra. The conditional
expectation $f \mapsto E[f \mid \mathcal{G}]$ is a bounded linear map $L_p \to L_p$ of norm at
most $1$, its values are $\mathcal{G}$-measurable, and it fixes every $\mathcal{G}$-measurable
element of $L_p$.
\end{lemma}

\begin{proof}
\leanok
The bound is the conditional Jensen inequality $\|E[f \mid \mathcal{G}]\|_p \leq \|f\|_p$,
which Mathlib provides; linearity holds almost everywhere.
\end{proof}

\begin{definition}[Filtration scheme]
\label{def:filtration-scheme}
\lean{MeasureTheory.Filtration.approximants, MeasureTheory.Filtration.scheme,
  MeasureTheory.Filtration.proj, MeasureTheory.Filtration.projectionScheme,
  MeasureTheory.Filtration.proj_two_pow}
\leanok
\uses{lem:condexp-lp, def:projection-scheme, def:comap}
Let $\mathcal{F}_0 \subseteq \mathcal{F}_1 \subseteq \cdots \subseteq \mathcal{F}$ be a
filtration. Let $A_0 = \{0\}$ and, for $n \geq 1$, let $A_n$ be the set of
$\mathcal{F}_{\lfloor \log_2 n \rfloor}$-measurable elements of $L_p$. These sets form an
approximation scheme, the \emph{filtration scheme}, and $P_0 = 0$,
$P_n = E[\,\cdot \mid \mathcal{F}_{\lfloor \log_2 n \rfloor}]$ form a projection scheme for it
with bound $1$, with $P_{2^k} = E[\,\cdot \mid \mathcal{F}_k]$.
\end{definition}

For the dyadic filtration of $[0, 1)$, $\mathcal{F}_j$ is generated by the dyadic intervals of
length $2^{-j}$, and the approximants of complexity $2^j$ are the step functions on these
$2^j$ intervals: the scheme of the Haar system.

\begin{theorem}[Martingale differences]
\label{thm:martingale}
\lean{MeasureTheory.Filtration.memApprox_iff_martingale}
\leanok
\uses{def:filtration-scheme, thm:blocks}
Let $\mathcal{F}_0 \subseteq \mathcal{F}_1 \subseteq \cdots$ be a filtration whose scheme
has $\bigcup_n A_n$ dense in $L_p$, and let $s > 0$ and $1 \leq q \leq \infty$. Then
$f \in A^s_q$ if and only if
\[
  \|E[f \mid \mathcal{F}_0]\|_p + \Bigl\| \Bigl(2^{ks} \bigl\|E[f \mid \mathcal{F}_{k+1}] -
  E[f \mid \mathcal{F}_k]\bigr\|_p\Bigr)_k \Bigr\|_{\ell_q} < \infty .
\]
\end{theorem}

\begin{proof}
\leanok
\Cref{thm:blocks} with $P_{2^k} = E[\,\cdot \mid \mathcal{F}_k]$: the dyadic blocks are the
martingale differences.
\end{proof}

For the dyadic filtration of $[0, 1)$ the density holds for $p < \infty$, and
$E[f \mid \mathcal{F}_{k+1}] - E[f \mid \mathcal{F}_k]$ is the part of the Haar expansion of
$f$ at level $k$ (\Cref{sec:haar}).

\section{The Haar system}
\label{sec:haar}

On $[0, 1)$ with Lebesgue measure the dyadic filtration makes \Cref{thm:martingale} explicit.
The conditional expectations are averages over dyadic intervals, which follows from a general
formula for partitions.

\begin{lemma}[Conditional expectation onto a partition]
\label{lem:condexp-partition}
\lean{MeasureTheory.comap_top_le, MeasureTheory.StronglyMeasurable.eq_of_comap_top,
  MeasureTheory.condExp_comap_ae_eq}
\leanok
Let $\mu$ be a finite measure on $\Omega$, let $\varphi : \Omega \to I$ with $I$ countable and
every fibre $\varphi^{-1}(i)$ measurable, and let $\mathcal{G}$ be the $\sigma$-algebra of
the sets $\varphi^{-1}(T)$, $T \subseteq I$. For every integrable $f$ and almost every
$x \in \Omega$,
\[
  E[f \mid \mathcal{G}](x) = \mu(C_x)^{-1} \int_{C_x} f \, d\mu ,
  \qquad C_x = \varphi^{-1}(\varphi(x)) .
\]
\end{lemma}

\begin{proof}
\leanok
A $\mathcal{G}$-measurable function is constant on every fibre. On a fibre $C$ of positive
measure, $\int_C E[f \mid \mathcal{G}] \, d\mu = \int_C f \, d\mu$ determines the constant.
The fibres of measure $0$ are countably many, so their union is a null set.
\end{proof}

\begin{definition}[Dyadic intervals and the dyadic filtration]
\label{def:dyadic-filtration}
\lean{Haar.vol, Haar.dyadicIndex, Haar.dyadicInterval, Haar.dyadicInterval_eq_Ico,
  Haar.dyadicInterval_eq_union, Haar.vol_dyadicInterval, Haar.dyadicSigma,
  Haar.dyadicFiltration, Haar.scheme}
\leanok
\uses{def:filtration-scheme}
For $j \in \N$ and $n \in \Z$ let $I_{j,n} = [n 2^{-j}, (n + 1) 2^{-j})$, the set of $x \in \R$
with $\lfloor 2^j x \rfloor = n$. Let $\mathcal{F}_j$ be the $\sigma$-algebra generated by the
intervals $I_{j,n}$, $n \in \Z$. Then $I_{j,n} = I_{j+1,2n} \cup I_{j+1,2n+1}$, so
$\mathcal{F}_0 \subseteq \mathcal{F}_1 \subseteq \cdots$ is a filtration, and
$|I_{j,n}| = 2^{-j}$. The \emph{Haar scheme} in $L_p[0, 1)$ is its filtration scheme.
\end{definition}

\begin{lemma}[Dyadic conditional expectations]
\label{lem:dyadic-condexp}
\lean{Haar.condExp_dyadic_ae_eq, Haar.condExpLp_dyadic_ae_eq, Haar.condExpLp_zero_ae_eq,
  Haar.enorm_condExpLp_zero}
\leanok
\uses{lem:condexp-partition, def:dyadic-filtration}
For $f \in L_1[0, 1)$, $j \in \N$ and almost every $x \in [0, 1)$,
\[
  E[f \mid \mathcal{F}_j](x) = 2^j \int_{I_{j,n}} f , \qquad n = \lfloor 2^j x \rfloor .
\]
In particular $E[f \mid \mathcal{F}_0] = \int_0^1 f$.
\end{lemma}

\begin{proof}
\leanok
\Cref{lem:condexp-partition} with $\varphi(x) = \lfloor 2^j x \rfloor$; on $[0, 1)$ the
relevant fibres are the intervals $I_{j,n}$, $0 \leq n < 2^j$, of measure $2^{-j}$.
\end{proof}

\begin{definition}[Haar functions and coefficients]
\label{def:haar}
\lean{Haar.haarFunction, Haar.measurable_haarFunction, Haar.haarFunction_eq_zero,
  Haar.abs_haarFunction, Haar.enorm_haarFunction, Haar.haarFunction_mul, Haar.haarCoeff,
  Haar.haarCoeff_eq}
\leanok
\uses{def:dyadic-filtration}
For $j \in \N$ and $m \in \Z$ the Haar function is
$h_{j,m} = 2^{j/2} (\mathbf{1}_{I_{j+1,2m}} - \mathbf{1}_{I_{j+1,2m+1}})$. It vanishes outside
$I_{j,m}$ and has modulus $2^{j/2}$ on $I_{j,m}$. The Haar coefficient of $f \in L_1[0, 1)$ is
\[
  \langle f, h_{j,m} \rangle = \int_0^1 h_{j,m} f
  = 2^{j/2} \Bigl( \int_{I_{j+1,2m}} f - \int_{I_{j+1,2m+1}} f \Bigr) .
\]
\end{definition}

\begin{lemma}[Martingale differences are Haar sums]
\label{lem:haar-martingale}
\lean{Haar.sum_fin_eq_of_ne, Haar.setIntegral_dyadicInterval, Haar.condExpLp_succ_sub_ae_eq}
\leanok
\uses{lem:dyadic-condexp, def:haar}
For $f \in L_p[0, 1)$ and $k \in \N$,
\[
  E[f \mid \mathcal{F}_{k+1}] - E[f \mid \mathcal{F}_k]
  = \sum_{m=0}^{2^k - 1} \langle f, h_{k,m} \rangle \, h_{k,m}
\]
almost everywhere.
\end{lemma}

\begin{proof}
\leanok
Let $x \in I_{k,m}$ with $0 \leq m < 2^k$, and let $a$ and $b$ be the integrals of $f$ over
$I_{k+1,2m}$ and $I_{k+1,2m+1}$. By \Cref{lem:dyadic-condexp} the left side at $x$ is
$2^{k+1} a - 2^k (a + b) = 2^k (a - b)$ on $I_{k+1,2m}$ and $-2^k (a - b)$ on $I_{k+1,2m+1}$.
On the right only the term $m$ is nonzero at $x$, and it has the same value since
$\langle f, h_{k,m} \rangle = 2^{k/2} (a - b)$.
\end{proof}

\begin{lemma}[Haar sums of one level]
\label{lem:haar-level}
\lean{Haar.eLpNorm_sum_haar}
\leanok
\uses{def:haar}
Let $1 \leq p < \infty$, $j \in \N$ and $c_0, \dots, c_{2^j - 1} \in \C$. Then
\[
  \Bigl\| \sum_{m=0}^{2^j - 1} c_m h_{j,m} \Bigr\|_p = 2^{j(1/2 - 1/p)} \|c\|_{\ell_p} .
\]
\end{lemma}

\begin{proof}
\leanok
At every point of $I_{j,m}$ the sum is $c_m h_{j,m}$, of modulus $|c_m| 2^{j/2}$, so the
$p$-th power of the norm is $\sum_m |c_m|^p 2^{jp/2} 2^{-j}$.
\end{proof}

\begin{lemma}[Density of step functions]
\label{lem:haar-density}
\lean{Haar.stepFun, Haar.stronglyMeasurable_stepFun, Haar.exists_norm_sub_stepFun_le,
  Haar.toLp_mem_closure, Haar.closure_iUnion_scheme}
\leanok
\uses{def:dyadic-filtration}
For $1 \leq p < \infty$ the Haar approximants are dense in $L_p[0, 1)$.
\end{lemma}

\begin{proof}
\leanok
A bounded continuous $g$ is uniformly continuous on $[0, 1]$, so the step functions
$x \mapsto g(\lfloor 2^j x \rfloor 2^{-j})$, which are $\mathcal{F}_j$-measurable, converge to
$g$ uniformly on $[0, 1)$, hence in $L_p$. Bounded continuous functions are dense in $L_p$ for
$p < \infty$.
\end{proof}

\begin{theorem}[Haar coefficient characterisation]
\label{thm:haar}
\lean{Haar.memApprox_iff_haarCoeff}
\leanok
\uses{thm:martingale, lem:haar-martingale, lem:haar-level, lem:haar-density}
Let $1 \leq p < \infty$, $s > 0$ and $1 \leq q \leq \infty$. Then $f \in L_p[0, 1)$ lies in the
approximation space $A^s_q$ of the Haar scheme if and only if
\[
  \Bigl| \int_0^1 f \Bigr| + \Bigl\| \Bigl(2^{k(s + 1/2 - 1/p)}
  \bigl\| (\langle f, h_{k,m} \rangle)_{0 \leq m < 2^k} \bigr\|_{\ell_p} \Bigr)_k
  \Bigr\|_{\ell_q} < \infty .
\]
\end{theorem}

\begin{proof}
\leanok
\Cref{thm:martingale}, with \Cref{lem:haar-density} for the density, the formula for
$E[f \mid \mathcal{F}_0]$ from \Cref{lem:dyadic-condexp}, and
\Cref{lem:haar-martingale,lem:haar-level} for the martingale differences.
\end{proof}

For $0 < s < 1/p$ this approximation space is the classical Besov space
$\Besov{s}{p}{q}[0, 1)$; for larger $s$ it is strictly smaller, since the Haar functions are
discontinuous. The comparison with moduli of smoothness is \Cref{cor:haar-besov}.


\chapter{Classical definitions on the torus}
\label{ch:classical}

The Besov and Sobolev spaces of the previous chapters are defined through approximation. The
classical definitions use moduli of smoothness and derivatives~\cite{Triebel, DeVoreLorentz}.
This chapter shows that they coincide on $L_2(\T^d)$, with the notation of \Cref{ch:torus}.

\section{Moduli of smoothness}

\begin{definition}[Translation]
\label{def:translation}
\lean{Torus.translate, Torus.coeFn_translate}
\leanok
For $h \in \T^d$, the translation $\tau_h : L_2(\T^d) \to L_2(\T^d)$ is
$(\tau_h f)(x) = f(x + h)$.
\end{definition}

\begin{lemma}[Fourier coefficients of translates]
\label{lem:translation-fourier}
\lean{Torus.mFourierCoeff_translate, Torus.mFourier_add_apply, Torus.enorm_translate_sub_sq,
  Torus.norm_mFourier_sub_one_le, Torus.norm_prod_sub_one_le, Torus.norm_fourier_sub_one_le}
\leanok
\uses{def:translation}
For every $h \in \T^d$, $f \in L_2(\T^d)$ and $k \in \Z^d$,
$\widehat{\tau_h f}(k) = e_k(h) \hat f(k)$ and $|e_k(h) - 1| \leq 2\pi \, \#d \, \deg k \,
\|h\|$, and
\[
  \|\tau_h f - f\|_2^2 = \sum_{k \in \Z^d} |e_k(h) - 1|^2 |\hat f(k)|^2 .
\]
\end{lemma}

\begin{proof}
\leanok
The Haar measure is invariant under translation and $e_k(x + h) = e_k(x) e_k(h)$; the second
formula is Parseval's identity. For the bound, $|\prod_j a_j - 1| \leq \sum_j |a_j - 1|$ for
complex numbers of modulus $1$, and $|e^{i\theta} - 1| \leq |\theta|$.
\end{proof}

\begin{definition}[Modulus of smoothness]
\label{def:modulus}
\lean{Torus.modulus, Torus.modulus_mono, Torus.enorm_translate_sub_le_modulus}
\leanok
\uses{def:translation}
For $f \in L_2(\T^d)$ and $t \in \R$,
$\omega(f, t)_2 = \sup_{\|h\| \leq t} \|\tau_h f - f\|_2$.
\end{definition}

\begin{lemma}[Growth of the modulus]
\label{lem:modulus-linear}
\lean{Torus.enorm_translate, Torus.translate_zero, Torus.translate_add,
  Torus.enorm_translate_nsmul_sub_le, Torus.exists_coe_eq_abs_eq_norm,
  Torus.modulus_natCast_mul_le}
\leanok
\uses{def:modulus}
For $f \in L_2(\T^d)$, $n \geq 1$ and $t \geq 0$, $\omega(f, n t)_2 \leq n \, \omega(f, t)_2$.
\end{lemma}

\begin{proof}
\leanok
Every $h$ with $\|h\| \leq n t$ is $n h'$ with $\|h'\| \leq t$, and
$\tau_{n h'} f - f = \sum_{i < n} \tau_{i h'} (\tau_{h'} f - f)$, where translations are
isometries.
\end{proof}

The classical Jackson theorem bounds the error of trigonometric approximation by the modulus
of smoothness. For $p = 2$ it follows from Parseval's identity by averaging over
translations along the coordinate axes.

\begin{theorem}[Jackson's theorem on the torus]
\label{thm:torus-jackson}
\lean{Torus.approxSeq_sq_le_card_mul_modulus_sq, Torus.approxSeq_sq_le_tsum,
  Torus.integral_norm_fourier_sub_one_sq, Torus.le_integral_norm_fourier_sub_one_sq,
  Torus.approxSeq_le_modulus}
\leanok
\uses{def:modulus, lem:translation-fourier, def:torus-sobolev}
For every $n \geq 1$ and $f \in L_2(\T^d)$, the error of approximation by $\mathcal{T}_n$ in
$L_2$ satisfies
\[
  E_n(f)_2^2 \leq \#d \cdot \omega(f, 1/n)_2^2 .
\]
\end{theorem}

\begin{proof}
\leanok
First $E_n(f)_2^2 \leq \sum_{\deg k \geq n} |\hat f(k)|^2$, by the truncation. For an integer
$a$ and $t > 0$,
$\int_0^t |e^{2\pi i a u} - 1|^2 \, du = 2t - \sin(2\pi a t)/(\pi a)$, which is at least $t$
when $|a| t \geq 1$. For $\deg k \geq n$ some coordinate has $|k_j| \geq n$, so
$1 \leq n \int_0^{1/n} |e_k(u \, e_j) - 1|^2 \, du$ with $u\,e_j$ the point $u$ in coordinate
$j$. Hence, by \Cref{lem:translation-fourier},
\[
  \sum_{\deg k \geq n} |\hat f(k)|^2 \leq \sum_{j \in d} n \int_0^{1/n} \sum_k
  |e_k(u e_j) - 1|^2 |\hat f(k)|^2 \, du
\]
and the inner sum is $\|\tau_{u e_j} f - f\|_2^2$. Since $\|u e_j\| \leq u \leq 1/n$, each
integrand is at most $\omega(f, 1/n)_2^2$.
\end{proof}

In particular $E_n(f)_2 \leq (\#d + 1) \, \omega(f, 1/n)_2$, the form used below.

\begin{definition}[Besov norm through the modulus]
\label{def:modulus-norm}
\lean{Torus.modulusNorm}
\leanok
\uses{def:modulus, def:lq-norm}
For $s \in \R$, $1 \leq q \leq \infty$ and $f \in L_2(\T^d)$,
\[
  \|f\|^{\omega}_{s,q} = \|f\|_2 + \bigl\| \bigl(2^{js} \omega(f, 2^{-j})_2\bigr)_{j \in \N}
  \bigr\|_{\ell_q} .
\]
\end{definition}

\begin{theorem}[Besov spaces through moduli of smoothness]
\label{thm:moduli-coincidence}
\lean{Torus.approxNormDyadic_le_modulusNorm, Torus.modulusNorm_le_approxNormDyadic,
  Torus.coincidenceConst, Torus.coincidenceConst_ne_top, Torus.memBesov_iff_modulusNorm_lt_top,
  Torus.blockMult, Torus.modulus_le_tsum_block, Torus.blockMult_le,
  Torus.dyadicWeight_mul_modulus_le}
\leanok
\uses{def:modulus-norm, thm:torus-jackson, thm:torus-coefficients, thm:blocks, lem:hardy,
  lem:translation-fourier}
Let $s \in \R$, $1 \leq q \leq \infty$ and $f \in L_2(\T^d)$. Then
\[
  \|f\|^{\mathrm{dy}}_{\Besov{s}{2}{q}} \leq (\#d + 1) \, \|f\|^{\omega}_{s,q}
  \qquad \text{and} \qquad
  \|f\|^{\omega}_{s,q} \leq K_s \, \|f\|^{\mathrm{dy}}_{\Besov{s}{2}{q}},
\]
where $K_s = 1 + 5\bigl(2(1 - 2^{-s})^{-1} + 4\pi \, \#d \, (1 - 2^{-(1-s)})^{-1}\bigr)$,
which is finite for $0 < s < 1$. In particular, for $0 < s < 1$, $f \in
\Besov{s}{2}{q}(\T^d)$ if and only if $\|f\|^{\omega}_{s,q} < \infty$.
\end{theorem}

\begin{proof}
\leanok
The first inequality is \Cref{thm:torus-jackson} at $n = 2^j$. For the second, split $f$
into the Fourier blocks $\Delta_K f$ over $2^K \leq \deg k < 2^{K+1}$; the constant term
does not move under translation. For $\|h\| \leq t$, \Cref{lem:translation-fourier} gives
$|e_k(h) - 1| \leq \min(2, 2\pi \, \#d \, 2^{K+1} t)$ on the block $K$, so
$\|\tau_h f - f\|_2 \leq \sum_K \min(2, 2\pi \, \#d \, 2^{K+1} t) \, \|\Delta_K f\|_2$. At
$t = 2^{-j}$ the minimum is at most $2$ for $K \geq j$ and at most $4\pi \, \#d \, 2^{K-j}$
for $K \leq j$, hence
\[
  2^{js} \omega(f, 2^{-j})_2 \leq 2 \cdot 2^{js} \sum_{i \geq 0} \|\Delta_{j+i} f\|_2 +
  4\pi \, \#d \, 2^{-j(1-s)} \sum_{K \leq j} 2^K \|\Delta_K f\|_2 .
\]
The first Hardy inequality with exponent $s$ and the second with exponent $1 - s$
(\Cref{lem:hardy}) bound both sums by $\|(2^{Ks} \|\Delta_K f\|_2)_K\|_{\ell_q}$, which is
at most $5 \|f\|^{\mathrm{dy}}_{\Besov{s}{2}{q}}$ by \Cref{thm:blocks} with $M = 1$.
\end{proof}

For $s \geq 1$ the first-order modulus no longer describes $\Besov{s}{2}{q}$; higher-order
differences are needed, and they are not formalised.

\section{Weak derivatives}

For a multi-index $\alpha \in \N^d$ the derivative $\partial^\alpha$ multiplies the Fourier
coefficients by $(2\pi i k)^\alpha$. Defining weak derivatives through this rule is the
distributional definition $\int g \, \varphi = (-1)^{|\alpha|} \int f \, \partial^\alpha
\varphi$, tested against the monomials $\varphi = \overline{e_k}$, whose span is dense in
the smooth functions. Mathlib has no distributions on the torus.

\begin{definition}[Weak derivatives and $W^m_2(\T^d)$]
\label{def:weak-derivative}
\lean{Torus.derivSymbol, Torus.HasWeakDeriv, Torus.MemSobolev}
\leanok
For $\alpha \in \N^d$ put $|\alpha| = \sum_j \alpha_j$ and $(2\pi i k)^\alpha = \prod_j
(2\pi i k_j)^{\alpha_j}$ for $k \in \Z^d$. An element $g \in L_2(\T^d)$ is the \emph{weak
derivative} $\partial^\alpha f$ of $f \in L_2(\T^d)$ if $\hat g(k) = (2\pi i k)^\alpha \hat
f(k)$ for every $k \in \Z^d$. For $m \in \N$, the Sobolev space $\Sob{m}{2}(\T^d)$ consists
of the $f \in L_2(\T^d)$ that have a weak derivative $\partial^\alpha f \in L_2(\T^d)$ for
every $\alpha$ with $|\alpha| \leq m$.
\end{definition}

\begin{lemma}[Existence and uniqueness of weak derivatives]
\label{lem:weak-derivative}
\lean{Torus.HasWeakDeriv.unique, Torus.hasWeakDeriv_zero, Torus.exists_hasWeakDeriv,
  Torus.HasWeakDeriv.mem_l2}
\leanok
\uses{def:weak-derivative}
Let $f \in L_2(\T^d)$ and $\alpha \in \N^d$. The weak derivative $\partial^\alpha f$ is unique
when it exists, $\partial^0 f = f$, and $\partial^\alpha f$ exists if and only if
$\sum_k |(2\pi i k)^\alpha \hat f(k)|^2 < \infty$.
\end{lemma}

\begin{proof}
\leanok
An element of $L_2(\T^d)$ is determined by its Fourier coefficients, and every
square-summable family of coefficients is the family of some element, since the
trigonometric monomials form a Hilbert basis.
\end{proof}

\begin{lemma}[Symbols and the Sobolev weight]
\label{lem:symbol-bounds}
\lean{Torus.norm_derivSymbol, Torus.norm_derivSymbol_le, Torus.one_add_pow_le,
  Torus.one_add_deg_pow_le}
\leanok
\uses{def:weak-derivative, def:trig-poly}
Let $m \in \N$ and $k \in \Z^d$. For every $\alpha \in \N^d$ with $|\alpha| \leq m$,
\[
  |(2\pi i k)^\alpha| \leq (2\pi)^m (1 + \deg k)^m ,
\]
and, writing $m e_j$ for the multi-index with $m$ in coordinate $j$ and $0$ elsewhere,
\[
  (1 + \deg k)^m \leq 2^m \Bigl(1 + \sum_{j \in d} |(2\pi i k)^{m e_j}|\Bigr) .
\]
\end{lemma}

\begin{proof}
\leanok
For the first bound, $|k_j| \leq 1 + \deg k$ for every $j$ and $2\pi (1 + \deg k) \geq 1$.
For the second, $(1 + x)^m \leq 2^m (1 + x^m)$ for $x \geq 0$; if $\deg k > 0$, then
$\deg k = |k_j|$ for some $j$, and $|k_j|^m \leq (2\pi |k_j|)^m$.
\end{proof}

\begin{theorem}[$\Sob{m}{2}(\T^d) = H^m(\T^d) = \Besov{m}{2}{2}(\T^d)$]
\label{thm:weak-sobolev}
\lean{Torus.memSobolev_iff, Torus.memSobolev_iff_memBesov, Torus.memSobolev_iff_mem_range}
\leanok
\uses{lem:weak-derivative, lem:symbol-bounds, thm:torus-coefficients, def:torus-sobolev}
Let $m \in \N$ and $f \in L_2(\T^d)$. Then $f \in \Sob{m}{2}(\T^d)$ if and only if
$\sum_k (1 + \deg k)^{2m} |\hat f(k)|^2 < \infty$. For $m \geq 1$ this holds if and only if
$f \in \Besov{m}{2}{2}(\T^d)$, and if and only if $f \in H^m(\T^d)$.
\end{theorem}

\begin{proof}
\leanok
If $f \in \Sob{m}{2}$, the coefficient families of $f$ and of $\partial_j^m f$ are
square-summable, and by the second bound of \Cref{lem:symbol-bounds} they dominate
$((1 + \deg k)^m \hat f(k))_k$. Conversely, the first bound shows that
$((2\pi i k)^\alpha \hat f(k))_k$ is square-summable for $|\alpha| \leq m$, and
\Cref{lem:weak-derivative} gives $\partial^\alpha f$. The rest is
\Cref{thm:torus-coefficients} and \Cref{def:torus-sobolev}.
\end{proof}

That the weak derivative of a smooth function is its classical derivative is integration
by parts on the torus; it is not formalised.

\section{Lipschitz atoms}
\label{sec:lip-atoms}

The atoms of Schneider and Vyb\'{\i}ral~\cite{SchneiderVybiral} need no smoothness beyond a
Lipschitz bound. On $L_2(\T^d)$ they form an atomic system in the sense of
\Cref{def:atomic-system} for the trigonometric scheme, with $\delta = \#d/2$ and $\sigma = 1$.
The positions of level $j$ are the points of $\{0, \dots, 2^j - 1\}^d$.

\begin{definition}[Lipschitz atoms]
\label{def:lip-atom}
\lean{Torus.gridPoint, Torus.IsLipAtom, Torus.isLipAtom_zero}
\leanok
\uses{def:trig-poly}
Let $D, j \in \N$ and $m \in \{0, \dots, 2^j - 1\}^d$. The grid point of level $j$ at $m$ is
$2^{-j} m \in \T^d$. A \emph{Lipschitz atom} of level $j$ at $m$ with support radius $D$ is a
bounded continuous function $a : \T^d \to \C$ with $|a(x)| \leq 1$ and
$|a(x) - a(y)| \leq 2^j \|x - y\|$ for all $x, y \in \T^d$, and $a(x) = 0$ whenever
$\|x - 2^{-j} m\| > D \, 2^{-j}$.
\end{definition}

The balls of the norm of $\T^d$ are cubes; Schneider and Vyb\'{\i}ral use dilated dyadic cubes
of $\R^d$, which amounts to the same up to the choice of $D$.

\begin{lemma}[Bounded overlap]
\label{lem:grid-overlap}
\lean{Torus.card_filter_norm_sub_le, Torus.card_filter_norm_sub_gridPoint_le,
  Torus.volume_closedBall_circle_le, Torus.volume_closedBall_le,
  Torus.volume_closedBall_gridPoint_le}
\leanok
\uses{def:lip-atom}
Let $j, R \in \N$ and $x \in \T^d$. At most $(2R + 1)^{\#d}$ positions
$m \in \{0, \dots, 2^j - 1\}^d$ satisfy $\|x - 2^{-j} m\| \leq R \, 2^{-j}$. For every
$\rho \geq 0$, every ball of radius $\rho$ in $\T^d$ has measure at most $(2\rho)^{\#d}$.
\end{lemma}

\begin{proof}
\leanok
Both statements reduce to one coordinate. Let $N = 2^j$ and let $y \in \R$ represent $x_i$. Every
$l$ with $\dist(y - l/N, \Z) \leq R/N$ is congruent modulo $N$ to an integer in
$[Ny - R, Ny + R]$, and distinct $l < N$ give distinct integers; an interval of length $2R$
contains at most $2R + 1$ integers. A ball of radius $\rho$ in $\T$ is an arc of length at most
$2\rho$, and a ball in $\T^d$ is the product of balls in the coordinates.
\end{proof}

\begin{lemma}[$L_2$ norms of sums with bounded overlap]
\label{lem:overlap-l2}
\lean{Torus.lintegral_enorm_sum_sq_le, Torus.finLqNorm_two_eq, Torus.eLpNorm_le_of_overlap}
\leanok
\uses{lem:grid-overlap}
Let $j, R \in \N$, $\tau \geq 0$ and $c_m \in \R$, and let $F_m : \T^d \to \C$,
$m \in \{0, \dots, 2^j - 1\}^d$, satisfy $|F_m| \leq |c_m| \, \tau$ and $F_m(x) = 0$ whenever
$\|x - 2^{-j} m\| > R \, 2^{-j}$. If $G = \sum_m F_m$ is measurable, then
\[
  \|G\|_2 \leq \bigl((2R + 1)^{\#d} (2R)^{\#d}\bigr)^{1/2} \, \tau \, 2^{-j \#d/2} \,
  \|c\|_{\ell_2} .
\]
\end{lemma}

\begin{proof}
\leanok
Let $S_m$ be the closed ball of radius $R \, 2^{-j}$ around $2^{-j} m$. At every point at most
$N = (2R + 1)^{\#d}$ of the $F_m$ are nonzero, so the Cauchy--Schwarz inequality gives
$|G(x)|^2 \leq N \sum_m c_m^2 \tau^2 \mathbf{1}_{S_m}(x)$. Integrate and use
$|S_m| \leq (2R)^{\#d} 2^{-j\#d}$.
\end{proof}

\begin{proposition}[Lipschitz atoms form an atomic system]
\label{prop:lip-atomic-system}
\lean{Torus.enorm_translate_toLp_sub, Torus.eLpNorm_sum_lipAtoms_le,
  Torus.enorm_translate_sum_lipAtoms_sub_le, Torus.lipConst, Torus.lipConst_ne_top,
  Torus.trigScheme_toLp_isLinear, Torus.lipAtomicSystem}
\leanok
\uses{def:atomic-system, def:lip-atom, lem:overlap-l2, thm:torus-jackson}
Let $D \in \N$. The Lipschitz atoms of support radius $D$ form an atomic system for the
trigonometric scheme in $L_2(\T^d)$ with $p = 2$, $\delta = \#d/2$, $\sigma = 1$ and the
constant
\[
  C = \bigl((2D + 1)^{\#d} (2D)^{\#d}\bigr)^{1/2}
    + (\#d + 1) \bigl((2D + 3)^{\#d} (2D + 2)^{\#d}\bigr)^{1/2} .
\]
\end{proposition}

\begin{proof}
\leanok
Let $g = \sum_m c_m a_m$ with atoms $a_m$ of level $j$. The size bound is \Cref{lem:overlap-l2}
with $R = D$ and $\tau = 1$. For the decay let $k \geq j$ and $\|h\| \leq 2^{-k}$. The function
$g(\cdot + h) - g = \sum_m c_m (a_m(\cdot + h) - a_m)$ has terms bounded by
$|c_m| \, 2^{j-k}$ and vanishing outside the ball of radius $(D + 1) 2^{-j}$ around
$2^{-j} m$, so \Cref{lem:overlap-l2} gives
\[
  \|\tau_h g - g\|_2 \leq \bigl((2D + 3)^{\#d} (2D + 2)^{\#d}\bigr)^{1/2} \, 2^{-(k-j)} \,
  2^{-j\#d/2} \|c\|_{\ell_2} .
\]
Taking the supremum over $h$ bounds $\omega(g, 2^{-k})_2$, and Jackson's theorem
$E_{2^k}(g)_2 \leq (\#d + 1) \, \omega(g, 2^{-k})_2$ (\Cref{thm:torus-jackson}) gives the
decay bound.
\end{proof}

\begin{theorem}[Synthesis with Lipschitz atoms]
\label{thm:lip-synthesis}
\lean{Torus.approxNormDyadic_le_of_lipAtoms, Torus.memBesov_of_lipAtoms}
\leanok
\uses{prop:lip-atomic-system, thm:atomic-synthesis}
Let $D \in \N$, $0 < s < 1$ and $1 \leq q \leq \infty$. Let $a_{j,m}$ be Lipschitz atoms of
level $j$ at $m$ with support radius $D$, let $\lambda_{j,m} \in \R$, and let
$f = \sum_j \sum_m \lambda_{j,m} a_{j,m}$, the series over $j$ converging in $L_2(\T^d)$. If
$\|(2^{j(s - \#d/2)} \|\lambda_{j,\cdot}\|_{\ell_2})_j\|_{\ell_q} < \infty$, then
$f \in \Besov{s}{2}{q}(\T^d)$, and
\[
  \|f\|^{\mathrm{dy}}_{\Besov{s}{2}{q}} \leq \bigl(2 (1 - 2^{-s})^{-1} + (1 - 2^{-(1 - s)})^{-1}
  \bigr) \, C \, \|\lambda\|_{b^s_{2,q}}
\]
with $C$ from \Cref{prop:lip-atomic-system}.
\end{theorem}

\begin{proof}
\leanok
\Cref{thm:atomic-synthesis} for the atomic system of \Cref{prop:lip-atomic-system}; the
trigonometric scheme is linear.
\end{proof}

The converse decomposes $f$ through quasi-interpolants with hat functions, which split into
pieces near the grid points; no expansion of trigonometric polynomials is needed.

\begin{lemma}[Hat functions]
\label{lem:hat}
\lean{Torus.circleHat, Torus.circleHat_nonneg, Torus.circleHat_le_one, Torus.circleHat_eq_zero,
  Torus.abs_circleHat_sub_le, Torus.continuous_circleHat, Torus.coe_sub_div_eq,
  Torus.natCast_mul_norm_coe_sub_div, Torus.natCast_eq_emod, Torus.sum_circleHat,
  Torus.abs_prod_sub_prod_le, Torus.torusHat, Torus.torusHat_nonneg, Torus.torusHat_le_one,
  Torus.sum_torusHat, Torus.abs_torusHat_sub_le, Torus.norm_le_half, Torus.torusHat_eq_zero,
  Torus.continuous_torusHat}
\leanok
\uses{def:lip-atom}
For $N \geq 1$ and $0 \leq l < N$ let $\Lambda_{N,l}(u) = \max(0, 1 - N \|u - l/N\|)$ on $\T$.
For $j \geq 1$ and $m \in \{0, \dots, 2^j - 1\}^d$ let
$\varphi_{j,m}(x) = \prod_i \Lambda_{2^j, m_i}(x_i)$, and let $\varphi_{0,0} = 1$. Then
$0 \leq \varphi_{j,m} \leq 1$, $\varphi_{j,m}$ vanishes outside the open ball of radius $2^{-j}$
around $2^{-j} m$, $|\varphi_{j,m}(x) - \varphi_{j,m}(y)| \leq \#d \, 2^j \|x - y\|$, and
$\sum_m \varphi_{j,m} = 1$ for every $j$.
\end{lemma}

\begin{proof}
\leanok
For $N \geq 2$, $y \in \R$, $t = N y$ and $k = \lfloor t \rfloor$, only the hats at the classes
of $k$ and $k + 1$ modulo $N$ are nonzero at $y$, with values $1 - (t - k)$ and $t - k$. The
partition of unity of $\T^d$ is the product of those of the coordinates, and
$|\prod_i a_i - \prod_i b_i| \leq \sum_i |a_i - b_i|$ for numbers in $[0, 1]$.
\end{proof}

\begin{lemma}[Oscillation over pairs of sets]
\label{lem:pair-oscillation}
\lean{Torus.eLpNorm_two_sq, Torus.enorm_sq_eq_lintegral, Torus.enorm_translate_sub_sq_eq,
  Torus.sum_lintegral_le_lintegral_sum, Torus.sum_lintegral_pair_le}
\leanok
\uses{def:modulus}
Let $(S_i, S'_i)_{i \in I}$ be finitely many pairs of measurable subsets of $\T^d$ with
$\|z - y\| \leq \rho$ for all $y \in S_i$ and $z \in S'_i$, such that every pair $(y, z)$ lies in
at most $N$ of the products $S_i \times S'_i$. Then for every $f \in L_2(\T^d)$,
\[
  \sum_i \int_{S_i} \int_{S'_i} |f(z) - f(y)|^2 \, dz \, dy
  \leq N \, |B(0, \rho)| \, \omega(f, \rho)_2^2 .
\]
\end{lemma}

\begin{proof}
\leanok
The sum is at most $N \int \int_{\|h\| \leq \rho} |f(y + h) - f(y)|^2 \, dh \, dy$ after the
substitution $z = y + h$. Exchanging the integrals gives
$N \int_{\|h\| \leq \rho} \|\tau_h f - f\|_2^2 \, dh$.
\end{proof}

\begin{definition}[Quasi-interpolants]
\label{def:quasi-interpolant}
\lean{Torus.dyadicBall, Torus.measurableSet_dyadicBall, Torus.ballAverage, Torus.quasiInterp,
  Torus.continuous_quasiInterp, Torus.quasiInterpBCF, Torus.quasiInterpBCF_apply,
  Torus.sum_torusHat_complex, Torus.quasiInterp_sub_const}
\leanok
\uses{lem:hat}
Let $B_{j,m}$ be the closed ball of radius $2^{-j}$ around $2^{-j} m$ and
$\mu_{j,m}(f) = |B_{j,m}|^{-1} \int_{B_{j,m}} f$ for $f \in L_2(\T^d)$. The
\emph{quasi-interpolant} of level $j$ is $Q_j f = \sum_m \mu_{j,m}(f) \, \varphi_{j,m}$.
\end{definition}

\begin{lemma}[Jackson's inequality for quasi-interpolants]
\label{lem:quasi-jackson}
\lean{Torus.volume_closedBall_circle, Torus.volume_closedBall_eq,
  Torus.dyadicWeight_le_volume_dyadicBall, Torus.inv_volume_dyadicBall_le,
  Torus.volume_dyadicBall_ne_zero, Torus.enorm_average_sq_le, Torus.enorm_ballAverage_sub_sq_le,
  Torus.mem_dyadicBall_of_torusHat_ne_zero, Torus.card_filter_mem_dyadicBall_le,
  Torus.enorm_sub_quasiInterp_sq_le, Torus.enorm_sub_quasiInterp_sq_le',
  Torus.enorm_sub_toLp_quasiInterp_le}
\leanok
\uses{def:quasi-interpolant, lem:pair-oscillation, lem:modulus-linear}
For every $f \in L_2(\T^d)$ and $j \in \N$,
$\|f - Q_j f\|_2 \leq 2 \cdot 6^{\#d} \, \omega(f, 2^{-j})_2$.
\end{lemma}

\begin{proof}
\leanok
By the partition of unity, $f - Q_j f = \sum_m \varphi_{j,m} (f - \mu_{j,m})$, and at most
$3^{\#d}$ terms are nonzero at each point. Jensen's inequality gives
$|f(x) - \mu_{j,m}|^2 \leq |B_{j,m}|^{-1} \int_{B_{j,m}} |f(z) - f(x)|^2 \, dz$, and
\Cref{lem:pair-oscillation} with the pairs $(B_{j,m}, B_{j,m})$ and $\rho = 2^{1-j}$ bounds
$\|f - Q_j f\|_2^2$ by $36^{\#d} \, \omega(f, 2^{1-j})_2^2$. Finish with
\Cref{lem:modulus-linear}.
\end{proof}

\begin{lemma}[The pieces of the levels]
\label{lem:pieces}
\lean{Torus.norm_sub_gridPoint_lt, Torus.nbr, Torus.mem_nbr, Torus.localOsc,
  Torus.localOsc_nonneg, Torus.piece, Torus.continuous_piece, Torus.sum_piece,
  Torus.sum_le_sum_nbr, Torus.norm_quasiInterp_sub_le, Torus.norm_quasiInterp_sub_quasiInterp_le,
  Torus.norm_piece_le, Torus.norm_piece_sub_le, Torus.piece_eq_zero, Torus.pieceCoeff,
  Torus.pieceAtom, Torus.pieceAtom_apply, Torus.isLipAtom_pieceAtom,
  Torus.pieceCoeff_mul_pieceAtom, Torus.meanAtom, Torus.isLipAtom_meanAtom,
  Torus.norm_mul_meanAtom}
\leanok
\uses{def:quasi-interpolant, def:lip-atom}
For $k \in \N$ and a position $m'$ of level $k + 1$ let $N(m')$ be the set of positions $m$ of
level $k$ with $\|2^{-k} m - 2^{-(k+1)} m'\| < 3 \cdot 2^{-(k+1)}$, and let
$M_{k+1,m'} = \sum_{m \in N(m')} |\mu_{k,m} - \mu_{k+1,m'}|$. Then
\[
  Q_{k+1} f - Q_k f = \sum_{m'} p_{k+1,m'}, \qquad
  p_{k+1,m'} = \varphi_{k+1,m'} \, (\mu_{k+1,m'} - Q_k f),
\]
and $p_{k+1,m'} = \lambda_{k+1,m'} \, a_{k+1,m'}$ with a Lipschitz atom $a_{k+1,m'}$ of support
radius $1$ and $\lambda_{k+1,m'} = (2 \#d + 1) M_{k+1,m'}$. Moreover $Q_0 f = \mu_{0,0}$ is
$|\mu_{0,0}|$ times a Lipschitz atom of level $0$.
\end{lemma}

\begin{proof}
\leanok
On the support of $\varphi_{k+1,m'}$ only the hats $\varphi_{k,m}$ with $m \in N(m')$ are
nonzero, and $Q_k f - \mu_{k+1,m'} = \sum_m \varphi_{k,m} (\mu_{k,m} - \mu_{k+1,m'})$. So
$|p_{k+1,m'}| \leq M_{k+1,m'}$, and $p_{k+1,m'}$ is Lipschitz with constant
$2 \#d \, 2^{k+1} M_{k+1,m'}$.
\end{proof}

\begin{lemma}[Coefficients of the levels]
\label{lem:piece-coefficients}
\lean{Torus.card_nbr_le, Torus.enorm_pieceCoeff_sq_le, Torus.enorm_ballAverage_sub_sq_le',
  Torus.sum_nbr_lintegral_le, Torus.sum_enorm_pieceCoeff_sq_le, Torus.sum_enorm_mean_sq_le}
\leanok
\uses{lem:pieces, lem:pair-oscillation, lem:modulus-linear}
For every $f \in L_2(\T^d)$ and $k \in \N$,
\[
  \Bigl( \sum_{m'} \lambda_{k+1,m'}^2 \Bigr)^{1/2}
  \leq (2 \#d + 1) \cdot 3 \cdot 24^{\#d} \, 2^{k \#d / 2} \, \omega(f, 2^{-k})_2 ,
\]
and $|\mu_{0,0}| \leq \|f\|_2$.
\end{lemma}

\begin{proof}
\leanok
The set $N(m')$ has at most $5^{\#d}$ elements, so by the Cauchy--Schwarz inequality and
Jensen's inequality $\lambda_{k+1,m'}^2$ is at most $(2 \#d + 1)^2 \, 5^{\#d}$ times
\[
  \sum_{m \in N(m')} |B_{k,m}|^{-1} |B_{k+1,m'}|^{-1}
  \int_{B_{k+1,m'}} \int_{B_{k,m}} |f(y) - f(z)|^2 \, dy \, dz .
\]
\Cref{lem:pair-oscillation} with $\rho = 3 \cdot 2^{-k}$ and overlap $9^{\#d}$ bounds the sum over
$m'$, and $5 \cdot 2 \cdot 9 \cdot 6 \leq 24^2$.
\end{proof}

\begin{theorem}[Analysis with Lipschitz atoms]
\label{thm:lip-analysis}
\lean{Torus.lipAtomOf, Torus.lipCoeffOf, Torus.isLipAtom_lipAtomOf, Torus.levelBCF,
  Torus.torusHat_zero, Torus.sum_lipCoeff_smul_lipAtom, Torus.sum_lipCoeff_smul_toLp,
  Torus.sum_range_levelBCF, Torus.dyadicWeight_mul_modulus_le_modulusNorm,
  Torus.modulus_le_dyadicWeight_mul, Torus.enorm_toLp_levelBCF_succ_le, Torus.hasSum_levels,
  Torus.finLqNorm_two_le, Torus.analysisConst, Torus.analysisConst_ne_top,
  Torus.bNorm_lipCoeffOf_le, Torus.exists_lipAtoms}
\leanok
\uses{lem:quasi-jackson, lem:piece-coefficients, def:modulus-norm}
Let $s > 0$, $1 \leq q \leq \infty$ and $f \in L_2(\T^d)$ with $\|f\|^{\omega}_{s,q} < \infty$.
Then $f = \sum_j \sum_m \lambda_{j,m} a_{j,m}$ in $L_2(\T^d)$ with Lipschitz atoms $a_{j,m}$ of
level $j$ at $m$ with support radius $1$ and
\[
  \|\lambda\|_{b^s_{2,q}} \leq \bigl(1 + (2 \#d + 1) \cdot 3 \cdot 24^{\#d} \, 2^{s - \#d/2}\bigr)
  \|f\|^{\omega}_{s,q} .
\]
\end{theorem}

\begin{proof}
\leanok
The levels $Q_0 f$ and $Q_{k+1} f - Q_k f$ add up to $Q_J f$, which tends to $f$ by
\Cref{lem:quasi-jackson}; their norms are summable since
$\omega(f, 2^{-k})_2 \leq 2^{-ks} \|f\|^{\omega}_{s,q}$. By \Cref{lem:piece-coefficients} the
level $k + 1$ contributes
$2^{(k+1)(s - \#d/2)} \|\lambda_{k+1,\cdot}\|_{\ell_2} \leq (2 \#d + 1) \cdot 3 \cdot 24^{\#d} \,
2^{s - \#d/2} \, 2^{ks} \omega(f, 2^{-k})_2$, and the level $0$ at most $\|f\|_2$.
\end{proof}

\begin{theorem}[Lipschitz-atomic characterisation]
\label{thm:lip-characterisation}
\lean{Torus.memBesov_iff_exists_lipAtoms}
\leanok
\uses{thm:lip-analysis, thm:lip-synthesis, thm:moduli-coincidence}
Let $0 < s < 1$ and $1 \leq q \leq \infty$. Then $f \in \Besov{s}{2}{q}(\T^d)$ if and only if
$f = \sum_j \sum_m \lambda_{j,m} a_{j,m}$ in $L_2(\T^d)$ with Lipschitz atoms of support radius
$1$ and $\|\lambda\|_{b^s_{2,q}} < \infty$.
\end{theorem}

\begin{proof}
\leanok
\Cref{thm:lip-analysis} with \Cref{thm:moduli-coincidence}, and \Cref{thm:lip-synthesis} for
the converse.
\end{proof}

This is the characterisation of Schneider and Vyb\'{\i}ral~\cite{SchneiderVybiral} on $\T^d$
for $p = 2$.

\chapter{Polynomial approximation on intervals}
\label{ch:interval}

This chapter treats the approximation by piecewise polynomials on an interval without
Fourier analysis and identifies their approximation spaces with Besov spaces defined by
moduli of smoothness. Functions are complex-valued and measurable on $\R$; norms are taken
on subintervals with Lebesgue measure.

\section{Differences and moduli}

\begin{definition}[Differences of higher order]
\label{def:iter-diff}
\lean{Interval.iterDiff, Interval.iterDiff_apply, Interval.iterDiff_zero, Interval.iterDiff_succ,
  Interval.iterDiff_succ', Interval.iterDiff_add, Interval.iterDiff_sub,
  Interval.iterDiff_const_mul, Interval.iterDiff_comp_add, Interval.iterDiff_reflect}
\leanok
For $f : \R \to \C$, $r \in \N$ and $h \in \R$ the difference of order $r$ is
$\Delta_h^r f(x) = \sum_{k=0}^r (-1)^{r-k} \binom{r}{k} f(x + k h)$, the $r$-fold iterate of
$\Delta_h f(x) = f(x + h) - f(x)$.
\end{definition}

\begin{lemma}[Differences annihilate polynomials]
\label{lem:iter-diff-poly}
\lean{Interval.iterDiff_polynomial}
\leanok
\uses{def:iter-diff}
For every polynomial $P$ of degree $< r$ and every $h \in \R$, $\Delta_h^r P = 0$.
\end{lemma}

\begin{proof}
\leanok
Mathlib proves this for the step $1$; the polynomial $y \mapsto P(x + h y)$ has degree $< r$ and
its differences with step $1$ at $0$ are those of $P$ with step $h$ at $x$.
\end{proof}

\begin{definition}[Modulus of smoothness on an interval]
\label{def:interval-modulus}
\lean{Interval.modulus, Interval.eLpNorm_iterDiff_le_modulus, Interval.modulus_mono,
  Interval.eLpNorm_comp_add_Icc, Interval.eLpNorm_comp_sub_Icc, Interval.eLpNorm_Icc_mono,
  Interval.measurable_iterDiff}
\leanok
\uses{def:iter-diff}
For $1 \leq p \leq \infty$, $r \in \N$ and $t \geq 0$ the modulus of smoothness of $f$ on
$[a, b]$ is
\[
  \omega_r(f, t, [a, b])_p = \sup_{0 \leq h \leq t} \|\Delta_h^r f\|_{L_p[a, b - r h]} .
\]
\end{definition}

\begin{lemma}[Properties of the modulus]
\label{lem:interval-modulus}
\lean{Interval.eLpNorm_iterDiff_le, Interval.modulus_le, Interval.modulus_add_le,
  Interval.modulus_const_mul, Interval.modulus_polynomial, Interval.modulus_congr_ae,
  Interval.modulus_reflect}
\leanok
\uses{def:interval-modulus, lem:iter-diff-poly}
The modulus $\omega_r(f, t, [a, b])_p$ increases in $t$, is subadditive and homogeneous in $f$,
satisfies $\omega_r(f, t, [a, b])_p \leq 2^r \|f\|_{L_p[a, b]}$, vanishes for polynomials of
degree $< r$, depends only on the restriction of $f$ to $[a, b]$ up to null sets, and is
invariant under $f \mapsto f(a + b - \cdot)$.
\end{lemma}

\begin{proof}
\leanok
Each term $f(\cdot + k h)$ is a translate of $f$ restricted to a subinterval of $[a, b]$, and
$\sum_k \binom{r}{k} = 2^r$. Reflection maps $\Delta_h^r$ to $\pm \Delta_h^r$ at the reflected
point.
\end{proof}

\section{Polynomials on intervals}

\begin{lemma}[Nikolskii inequality for polynomials on intervals]
\label{lem:interval-nikolskii}
\lean{Interval.polyFun, Interval.continuous_polyFun, Interval.measurable_polyFun,
  Interval.comp_affine_mem_degreeLT, Interval.polyFun_comp_affine, Interval.eq_zero_of_ae_eq_zero,
  Interval.memLp_polyFun, Interval.polyFun_degreeLTEquiv_symm, Interval.coeffToLp,
  Interval.enorm_coeffToLp, Interval.coeffToLp_injective, Interval.exists_enorm_le_eLpNorm,
  Interval.eLpNorm_comp_affine, Interval.enorm_polyFun_le}
\leanok
Let $r \in \N$ and $1 \leq p < \infty$. There is $K < \infty$ such that for every polynomial $P$ of
degree $< r$, every interval $[u, u + L]$ with $L > 0$ and every $x \in [u, u + L]$,
\[
  |P(x)| \leq K L^{-1/p} \|P\|_{L_p[u, u + L]} .
\]
\end{lemma}

\begin{proof}
\leanok
On $[0, 1]$ the map from the $r$ coefficients to $L_p[0, 1]$ is injective, since a nonzero
polynomial has finitely many zeros, hence bounded below in finite dimension; this bounds the
coefficients and so $|P(x)|$ by $\|P\|_{L_p[0,1]}$. The general interval follows by the affine
change of variables $x = u + L y$, which multiplies the $L_p$ norm by $L^{-1/p}$.
\end{proof}

\begin{lemma}[Polynomial approximation of smooth functions]
\label{lem:taylor-lp}
\lean{Interval.exists_derivative_eq, Interval.hasDerivAt_polyFun,
  Interval.exists_eLpNorm_sub_polyFun_le}
\leanok
Let $1 \leq p \leq \infty$, $r \in \N$, $g \in C^r(\R)$ and $u \leq v$. There is a polynomial $Q$
of degree $< r$ with
$\|g - Q\|_{L_p[u, v]} \leq (v - u)^r \|g^{(r)}\|_{L_p[u, v]}$.
\end{lemma}

\begin{proof}
\leanok
Induction on $r$. Approximate $g'$ by $Q'$ of degree $< r - 1$ and let $Q$ be an antiderivative of
$Q'$ with $Q(u) = g(u)$. By the fundamental theorem of calculus and Hölder's inequality,
$|(g - Q)(x)| \leq \int_u^v |g' - Q'| \leq (v - u)^{1 - 1/p} \|g' - Q'\|_{L_p[u, v]}$, and the
$L_p$ norm of a function bounded by $M$ is at most $(v - u)^{1/p} M$.
\end{proof}

\section{Steklov smoothing}

\begin{definition}[Steklov averages]
\label{def:steklov}
\lean{Interval.steklov, Interval.steklov_eq, Interval.steklovIter, Interval.steklovIter_succ}
\leanok
For $g : \R \to \C$ and $\eta > 0$ the Steklov average is
$M_\eta g(x) = \eta^{-1} \int_0^\eta g(x + \tau) \, d\tau$, and $M_\eta^m$ is its $m$-fold
iterate.
\end{definition}

\begin{lemma}[Averages gain derivatives]
\label{lem:steklov-deriv}
\lean{Interval.hasDerivAt_steklov, Interval.differentiable_steklov, Interval.deriv_steklov,
  Interval.steklov_comp_add, Interval.steklov_sub, Interval.steklov_fwdDiff,
  Interval.continuous_steklov, Interval.continuous_fwdDiff, Interval.contDiff_fwdDiff,
  Interval.contDiff_steklov, Interval.contDiff_steklovIter, Interval.continuous_steklovIter,
  Interval.steklovIter_fwdDiff, Interval.iteratedDeriv_steklovIter_add,
  Interval.iteratedDeriv_steklovIter}
\leanok
\uses{def:steklov, def:iter-diff}
For continuous $g$, $(M_\eta g)' = \eta^{-1} \Delta_\eta g$. Hence $M_\eta^m g \in C^m$ and
$(M_\eta^r g)^{(r)} = \eta^{-r} \Delta_\eta^r g$.
\end{lemma}

\begin{proof}
\leanok
$M_\eta g(x) = \eta^{-1} (G(x + \eta) - G(x))$ with a primitive $G$ of $g$. Averages commute with
translations and differences, so each further average adds one derivative.
\end{proof}

\begin{lemma}[Minkowski's inequality for averages]
\label{lem:minkowski-avg}
\lean{Interval.eLpNorm_rpow_eq, Interval.enorm_avg_rpow_le, Interval.eLpNorm_avg_le,
  Interval.avgStep, Interval.avgStepIter, Interval.avgStepIter_succ, Interval.continuous_avgStep,
  Interval.continuous_avgStepIter, Interval.eLpNorm_avgStepIter_le}
\leanok
Let $K : \R \times \R \to \C$ be continuous, $\delta > 0$, $1 \leq p < \infty$ and $S \subseteq \R$.
Then
\[
  \Bigl\| \delta^{-1} \int_0^\delta K(t, \cdot) \, dt \Bigr\|_{L_p(S)}
  \leq \sup_{0 \leq t \leq \delta} \|K(t, \cdot)\|_{L_p(S)} .
\]
Iterating $m$ times, an average over $v \in [0, \delta]^m$ of $G(\sigma + v_1 + \cdots + v_m, \cdot)$
has $L_p(S)$ norm at most $\sup_{\sigma \leq h \leq \sigma + m \delta} \|G(h, \cdot)\|_{L_p(S)}$.
\end{lemma}

\begin{proof}
\leanok
Hölder's inequality gives
$|\delta^{-1} \int_0^\delta K(t, x) \, dt|^p \leq \delta^{-1} \int_0^\delta |K(t, x)|^p \, dt$;
integrate over $x \in S$ and exchange the integrals by Tonelli's theorem.
\end{proof}

\begin{theorem}[Steklov smoothing]
\label{thm:steklov-smoothing}
\lean{Interval.avgStepIter_const, Interval.avgStepIter_shift, Interval.avgStepIter_sum,
  Interval.smooth, Interval.sub_smooth_eq, Interval.contDiff_smooth, Interval.iteratedDeriv_smooth,
  Interval.continuous_iterDiff_uncurry, Interval.eLpNorm_sub_smooth_le,
  Interval.eLpNorm_iteratedDeriv_smooth_le}
\leanok
\uses{lem:steklov-deriv, lem:minkowski-avg, def:interval-modulus}
Let $f$ be continuous, $r \in \N$, $\delta > 0$ and
$S_\delta f = \sum_{k=1}^r (-1)^{k+1} \binom{r}{k} M_{k\delta}^r f$. Then $S_\delta f \in C^r$, and for
$1 \leq p < \infty$ and $[u, w]$ with $a \leq u$ and $w + r^2 \delta \leq b$,
\[
  \|f - S_\delta f\|_{L_p[u, w]} \leq \omega_r(f, r\delta, [a, b])_p, \qquad
  \|(S_\delta f)^{(r)}\|_{L_p[u, w]} \leq 2^r \delta^{-r} \omega_r(f, r\delta, [a, b])_p .
\]
\end{theorem}

\begin{proof}
\leanok
Averaging $f(x + k h)$ over $h = v_1 + \cdots + v_r$, $v \in [0, \delta]^r$, gives
$M_{k\delta}^r f(x)$, so $f - S_\delta f = (-1)^r$ times the average of $\Delta_h^r f$, which
\Cref{lem:minkowski-avg} bounds. The derivative is
$(S_\delta f)^{(r)} = \sum_{k=1}^r (-1)^{k+1} \binom{r}{k} (k\delta)^{-r} \Delta_{k\delta}^r f$ by
\Cref{lem:steklov-deriv}.
\end{proof}

\section{Dyadic piecewise polynomials}

\begin{definition}[Dyadic piecewise polynomials]
\label{def:pp-scheme}
\lean{Interval.unitVol, Interval.ae_mem_Ico_unitVol, Interval.dyadicIco,
  Interval.measurableSet_dyadicIco, Interval.mem_dyadicIco, Interval.eq_of_mem_dyadicIco,
  Interval.mem_dyadicIco_floor, Interval.floor_two_pow_succ_div, Interval.ppFun,
  Interval.ppFun_eq_of_mem, Interval.measurable_ppFun, Interval.memLp_ppFun, Interval.ppFun_add,
  Interval.ppFun_const_mul, Interval.polyFun_zero, Interval.ppFun_zero, Interval.ppFun_refine,
  Interval.ppSpace, Interval.ppSpace_mono, Interval.ppApproximants, Interval.ppApproximants_mono,
  Interval.ppScheme, Interval.mem_ppScheme_two_pow, Interval.ppScheme_isLinear}
\leanok
Let $r \geq 1$ and $1 \leq p < \infty$. For $k \in \N$ the dyadic intervals of level $k$ are
$I_{k,n} = [n 2^{-k}, (n+1) 2^{-k})$, $n < 2^k$. The space $V_k \subseteq L_p[0,1]$ consists of
the classes of functions that agree on every $I_{k,n}$ with a polynomial of degree $< r$. The
scheme of dyadic piecewise polynomials has $A_0 = \{0\}$ and $A_n = V_{\lfloor \log_2 n \rfloor}$
for $n \geq 1$.
\end{definition}

\begin{lemma}[Bookkeeping for dyadic intervals]
\label{lem:pp-bookkeeping}
\lean{Interval.floor_eq_of_mem_dyadicIco, Interval.dyadicIco_subset,
  Interval.pairwise_disjoint_dyadicIco, Interval.ppFun_eq_zero, Interval.ppFun_eq_polyFun,
  Interval.ite_mem_degreeLT, Interval.lintegral_Icc_le_sum, Interval.sum_le_lintegral_Icc,
  Interval.eLpNorm_le_of_lintegral_le}
\leanok
\uses{def:pp-scheme}
For $x \in I_{k,n}$, $\lfloor 2^k x \rfloor = n$. A piecewise polynomial of level $k$ vanishes
outside $[0, 1)$. For $u \in \R$, $L \geq 0$, $m \in \N$ and every $G : \R \to [0, \infty]$,
$\int_{[u, u + mL]} G = \sum_{n < m} \int_{[u + nL, u + (n+1)L]} G$.
\end{lemma}

\begin{theorem}[Jackson's inequality for piecewise polynomials]
\label{thm:pp-jackson}
\lean{Interval.jacksonConst, Interval.jacksonConst_ne_top, Interval.exists_left_approx,
  Interval.exists_right_approx, Interval.eLpNorm_Icc_split, Interval.exists_ppFun_approx,
  Interval.lpModulus, Interval.lpModulus_add_le, Interval.lpModulus_le,
  Interval.approxSeq_ppScheme_le}
\leanok
\uses{def:pp-scheme, lem:pp-bookkeeping, thm:steklov-smoothing, lem:taylor-lp,
  lem:interval-modulus}
Let $1 \leq p < \infty$ and $r \geq 1$. For every $f \in L_p[0,1]$ and $k \geq 1$,
\[
  E_{2^k}(f)_p \leq 2 (1 + 2^r r^{2r}) \, \omega_r(f, 2^{-k}, [0,1])_p .
\]
\end{theorem}

\begin{proof}
\leanok
First let $f$ be continuous, $L = 2^{-k}$ and $\delta = L / r^2$. Since
$\tfrac12 + r^2 \delta \leq 1$, the smoothing $g = S_\delta f$ of \Cref{thm:steklov-smoothing}
satisfies $\|f - g\|_{L_p[0,1/2]} \leq \omega_r(f, L, [0,1])_p$ and
$\|g^{(r)}\|_{L_p[0,1/2]} \leq 2^r \delta^{-r} \omega_r(f, L, [0,1])_p$. By \Cref{lem:taylor-lp}
there is on each $I_{k,n} \subseteq [0, \tfrac12]$ a polynomial $Q_n$ of degree $< r$ with
$\|g - Q_n\|_{L_p(I_{k,n})} \leq L^r \|g^{(r)}\|_{L_p(I_{k,n})}$. Summing $p$-th powers, the
piecewise polynomial $S$ made of the $Q_n$ satisfies
$\|g - S\|_{L_p[0,1/2]} \leq L^r 2^r \delta^{-r} \omega_r(f, L, [0,1])_p
= 2^r r^{2r} \omega_r(f, L, [0,1])_p$. On $[\tfrac12, 1]$ apply the same to $f(1 - \cdot)$, which
has the same modulus by \Cref{lem:interval-modulus}; $\tfrac12$ is a knot since $k \geq 1$.
For general $f$ take a bounded continuous $g$ close to $f$ in $L_p$: then
$E_{2^k}(f)_p \leq E_{2^k}(g)_p + \|f - g\|_p$ and
$\omega_r(g, t)_p \leq \omega_r(f, t)_p + 2^r \|f - g\|_p$.
\end{proof}

\begin{lemma}[Local estimates for piecewise polynomials]
\label{lem:pp-local}
\lean{Interval.localMass, Interval.tsum_localMass_ppFun_le, Interval.IsPiecewisePoly,
  Interval.isPiecewisePoly_ppFun, Interval.enorm_le_localMass,
  Interval.iterDiff_eq_zero_of_isPiecewisePoly, Interval.enorm_iterDiff_le_of_isPiecewisePoly,
  Interval.stripBound, Interval.lintegral_stripBound_le, Interval.sum_sum_localMass_le}
\leanok
\uses{def:pp-scheme, lem:pp-bookkeeping, lem:interval-nikolskii, lem:iter-diff-poly}
Let $1 \leq p < \infty$, and let $a$ be a piecewise polynomial of level $j$ and degree $< r$
on $[0, \infty)$: on every $I_{j,n}$, $n \in \N$, it agrees with a polynomial of degree $< r$.
Let $m_n = \int_{I_{j,n}} |a|^p$ for $n \in \N$, $0 \leq h \leq 2^{-j}$, $x \geq 0$ and
$n = \lfloor 2^j x \rfloor$. If $x + rh < (n+1) 2^{-j}$, then $\Delta_h^r a(x) = 0$. In every
case
\[
  |\Delta_h^r a(x)| \leq 2^r K 2^{j/p} \Bigl( \sum_{l=0}^r m_{n+l} \Bigr)^{1/p}
\]
with the constant $K$ of \Cref{lem:interval-nikolskii}. If $a$ vanishes outside $[0, 1)$, then
$\sum_{n < 2^j} \sum_{l = 0}^r m_{n+l} \leq (r+1) \|a\|_p^p$.
\end{lemma}

\begin{proof}
\leanok
In the first case $x, x + h, \dots, x + rh$ lie in $I_{j,n}$, where $a$ is a polynomial of
degree $< r$, so \Cref{lem:iter-diff-poly} applies. In general $x + ih \in I_{j, n+l}$ with
$0 \leq l \leq r$, where $|a| \leq K 2^{j/p} m_{n+l}^{1/p}$ by \Cref{lem:interval-nikolskii}, and
$\sum_i \binom{r}{i} = 2^r$. In the double sum every $m_n$ occurs at most $r + 1$ times.
\end{proof}

\begin{theorem}[Inverse inequality for piecewise polynomials]
\label{thm:pp-inverse}
\lean{Interval.eLpNorm_iterDiff_le_of_isPiecewisePoly, Interval.eLpNorm_iterDiff_ppFun_le,
  Interval.ofReal_two_pow_mul_rpow_le, Interval.ppInverseConst, Interval.ppInverseConst_ne_top,
  Interval.two_pow_le_ppInverseConst, Interval.le_ppInverseConst_mul,
  Interval.exists_inverse_ppScheme}
\leanok
\uses{lem:pp-local}
Let $1 \leq p < \infty$ and $r \in \N$. There is $C_B < \infty$ such that for all $j, k \in \N$
and every $a \in V_j$,
\[
  \omega_r(a, 2^{-k}, [0, 1])_p \leq C_B \, 2^{-\max(k - j, 0)/p} \|a\|_p .
\]
\end{theorem}

\begin{proof}
\leanok
For $k < j$ this is $\omega_r(a, t, [0,1])_p \leq 2^r \|a\|_p$. Let $k \geq j$ and
$0 \leq h \leq 2^{-k}$. By \Cref{lem:pp-local}, $|\Delta_h^r a|^p$ vanishes on $I_{j,n}$ outside
the strip $[(n+1)2^{-j} - rh, (n+1)2^{-j})$ and is bounded on it by
$(2^r K)^p 2^j \sum_{l=0}^r m_{n+l}$. The strips have width $rh$, so for every piecewise
polynomial $a$ of level $j$ on $[0, \infty)$ with
$\sum_{n < 2^j} \sum_{l=0}^r m_{n+l} \leq (r+1) B^p$,
$\|\Delta_h^r a\|_{L_p[0, 1]}^p \leq (2^r K)^p \, r (r+1) \, 2^j h \, B^p$. Here $B = \|a\|_p$,
and $2^j h \leq 2^{-(k-j)}$.
\end{proof}

\begin{theorem}[Approximation spaces of dyadic piecewise polynomials]
\label{thm:pp-besov}
\lean{Interval.ppModulusData, Interval.memApprox_ppScheme_iff}
\leanok
\uses{thm:pp-jackson, thm:pp-inverse, def:modulus-data, thm:modulus-coincidence}
Let $1 \leq p < \infty$, $1 \leq q \leq \infty$, $r \geq 1$ and $0 \leq s < 1/p$. A function
$f \in L_p[0,1]$ lies in the approximation space $A^s_q$ of the scheme of dyadic piecewise
polynomials of degree $< r$ if and only if
\[
  \|f\|_p + \bigl\| \bigl( 2^{ks} \omega_r(f, 2^{-k}, [0,1])_p \bigr)_k \bigr\|_{\ell_q} < \infty .
\]
\end{theorem}

\begin{proof}
\leanok
By \Cref{thm:pp-jackson}, \Cref{thm:pp-inverse} and $\omega_r(f, t, [0,1])_p \leq 2^r \|f\|_p$
(\Cref{lem:interval-modulus}), $\omega_r(\cdot, 2^{-k}, [0,1])_p$ is a modulus compatible with
the scheme of inverse order $\theta = 1/p$ in the sense of \Cref{def:modulus-data}. Apply
\Cref{thm:modulus-coincidence}.
\end{proof}

For $0 < s < 1/p$ the condition says that $f$ lies in the Besov space $\Besov{s}{p}{q}[0,1]$
defined by moduli of smoothness. The bound $s < 1/p$ comes from the jumps at the knots: the
indicator of $[0, \tfrac12)$ lies in $V_1$, and its modulus is of order $t^{1/p}$. Higher
smoothness needs piecewise polynomials that are smooth across the knots, that is, splines.

\section{Piecewise polynomials on the circle}

\begin{definition}[Periodic modulus of smoothness]
\label{def:per-modulus}
\lean{Interval.perExt, Interval.perExt_periodic, Interval.perExt_of_mem,
  Interval.measurable_perExt, Interval.perExt_add, Interval.perModulus,
  Interval.eLpNorm_iterDiff_le_perModulus, Interval.perModulus_mono}
\leanok
\uses{def:interval-modulus}
For $f : \R \to \C$ the periodic extension of $f|_{[0,1)}$ is
$\tilde f(x) = f(x - \lfloor x \rfloor)$. For $1 \leq p \leq \infty$, $r \in \N$ and $t \geq 0$
the periodic modulus of smoothness is
\[
  \omega_r^{\mathrm{per}}(f, t)_p = \sup_{0 \leq h \leq t} \|\Delta_h^r \tilde f\|_{L_p[0,1]} .
\]
\end{definition}

Through $\tilde f$, functions on $[0, 1)$ are the functions on the circle $\T = \R / \Z$, and
$L_p[0,1]$ is $L_p(\T)$. Unlike $\omega_r(f, t, [0,1])_p$, the periodic modulus compares values
near $1$ with values near $0$.

\begin{lemma}[Properties of the periodic modulus]
\label{lem:per-modulus}
\lean{Interval.perExt_ae_eq, Interval.iterDiff_perExt_ae_eq, Interval.lintegral_Icc_periodic,
  Interval.perModulus_congr_ae, Interval.perModulus_add_le, Interval.eLpNorm_perExt_Icc,
  Interval.perModulus_le, Interval.modulus_le_perModulus}
\leanok
\uses{def:per-modulus, lem:interval-modulus}
Let $1 \leq p < \infty$, $r \in \N$ and $t \geq 0$. The periodic modulus depends only on the class
of $f$ in $L_p[0,1]$, is subadditive in $f$, and for measurable $f$
\[
  \omega_r(f, t, [0,1])_p \leq \omega_r^{\mathrm{per}}(f, t)_p \leq 2^r \|f\|_{L_p[0,1]} .
\]
\end{lemma}

\begin{proof}
\leanok
If $f = g$ almost everywhere on $[0, 1]$, then $\{\tilde f \neq \tilde g\}$ is contained in the
union of the integer translates of a null set. The integral of a $1$-periodic function over
$[c, c + 1]$ is its integral over the circle, so it does not depend on $c$; hence each term
$\tilde f(\cdot + kh)$ of $\Delta_h^r \tilde f$ has $L_p[0,1]$ norm $\|f\|_{L_p[0,1]}$, and
$\sum_k \binom{r}{k} = 2^r$. For $x \in [0, 1 - rh)$ the points $x, \dots, x + rh$ lie in
$[0, 1)$, where $\tilde f = f$.
\end{proof}

\begin{theorem}[Approximation spaces on the circle]
\label{thm:pp-besov-per}
\lean{Interval.isPiecewisePoly_perExt_ppFun, Interval.sum_localMass_perExt_le,
  Interval.sum_sum_localMass_perExt_le, Interval.lpPerModulus, Interval.lpPerModulus_add_le,
  Interval.lpPerModulus_le, Interval.lpModulus_le_lpPerModulus,
  Interval.approxSeq_ppScheme_le_per, Interval.exists_inverse_ppScheme_per,
  Interval.ppPerModulusData, Interval.memApprox_ppScheme_iff_per}
\leanok
\uses{thm:pp-jackson, thm:pp-inverse, lem:pp-local, lem:per-modulus, def:modulus-data,
  thm:modulus-coincidence}
Let $1 \leq p < \infty$, $1 \leq q \leq \infty$, $r \geq 1$ and $0 \leq s < 1/p$. A function
$f \in L_p[0,1]$ lies in the approximation space $A^s_q$ of the scheme of dyadic piecewise
polynomials of degree $< r$ if and only if
\[
  \|f\|_p + \bigl\| \bigl( 2^{ks} \omega_r^{\mathrm{per}}(f, 2^{-k})_p \bigr)_k \bigr\|_{\ell_q}
  < \infty .
\]
\end{theorem}

\begin{proof}
\leanok
Jackson's inequality for $\omega_r^{\mathrm{per}}$ follows from \Cref{thm:pp-jackson} and
$\omega_r(f, t, [0,1])_p \leq \omega_r^{\mathrm{per}}(f, t)_p$. For the inverse inequality, the
periodic extension $\tilde a$ of $a \in V_j$ is a piecewise polynomial of level $j$ on
$[0, \infty)$: on $I_{j, m' + 2^j c}$ with $m' < 2^j$ it is the piece of $a$ on $I_{j, m'}$
shifted by $c$. Any $2^j$ consecutive pieces of $\tilde a$ make up one period, so
$\sum_{n < 2^j} \sum_{l = 0}^r m_{n+l} \leq (r + 1) \|a\|_p^p$, and the estimate in the proof of
\Cref{thm:pp-inverse} gives
$\omega_r^{\mathrm{per}}(a, 2^{-k})_p \leq C_B 2^{-\max(k-j, 0)/p} \|a\|_p$. Apply
\Cref{thm:modulus-coincidence} with $\theta = 1/p$.
\end{proof}

\begin{corollary}[Interval and circle]
\label{cor:interval-circle}
\lean{Interval.lqNorm_modulus_lt_top_iff_perModulus}
\leanok
\uses{thm:pp-besov, thm:pp-besov-per}
Let $1 \leq p < \infty$, $1 \leq q \leq \infty$, $r \geq 1$ and $0 \leq s < 1/p$. For every
$f \in L_p[0,1]$,
\[
  \|f\|_p + \bigl\| \bigl( 2^{ks} \omega_r(f, 2^{-k}, [0,1])_p \bigr)_k \bigr\|_{\ell_q} < \infty
  \iff
  \|f\|_p + \bigl\| \bigl( 2^{ks} \omega_r^{\mathrm{per}}(f, 2^{-k})_p \bigr)_k \bigr\|_{\ell_q}
  < \infty .
\]
\end{corollary}

\begin{proof}
\leanok
By \Cref{thm:pp-besov} and \Cref{thm:pp-besov-per} both conditions say $f \in A^s_q$.
\end{proof}

So $\Besov{s}{p}{q}[0,1] = \Besov{s}{p}{q}(\T)$ for $0 < s < 1/p$: below smoothness $1/p$ a jump
between the ends of the interval costs nothing.

\begin{corollary}[Besov spaces through Haar coefficients]
\label{cor:haar-besov}
\lean{Haar.vol_eq_unitVol, Haar.measurePreserving_id_vol, Haar.measurePreserving_id_unitVol,
  Haar.ae_eq_unitVol_of_vol, Haar.ae_eq_vol_of_unitVol, Haar.toHaar, Haar.ofHaar,
  Haar.coeFn_toHaar, Haar.coeFn_ofHaar, Haar.ofHaar_toHaar, Haar.polyFun_of_mem_degreeLT_one,
  Haar.dyadicIndex_eq_floor, Haar.dyadicIndex_natCast_div, Haar.ppFun_ae_eq_stepFun,
  Haar.ae_eq_ppFun_of_stronglyMeasurable, Haar.toHaar_mem_scheme, Haar.ofHaar_mem_ppScheme,
  Haar.memApprox_ppScheme_one_iff, Haar.haarCoeff_toHaar, Haar.integral_toHaar,
  Haar.lqNorm_modulus_lt_top_iff_haarCoeff, Haar.lqNorm_perModulus_lt_top_iff_haarCoeff}
\leanok
\uses{thm:haar, thm:pp-besov, thm:pp-besov-per}
Let $1 \leq p < \infty$, $1 \leq q \leq \infty$ and $0 < s < 1/p$. For $f \in L_p[0,1]$ the
following are equivalent.
\begin{enumerate}
  \item $\|f\|_p + \bigl\| \bigl( 2^{ks} \omega_1(f, 2^{-k}, [0,1])_p \bigr)_k \bigr\|_{\ell_q}
    < \infty$.
  \item $\|f\|_p + \bigl\| \bigl( 2^{ks} \omega_1^{\mathrm{per}}(f, 2^{-k})_p \bigr)_k
    \bigr\|_{\ell_q} < \infty$.
  \item $\bigl| \int_0^1 f \bigr| + \Bigl\| \Bigl(2^{k(s + 1/2 - 1/p)}
    \bigl\| (\langle f, h_{k,m} \rangle)_{0 \leq m < 2^k} \bigr\|_{\ell_p} \Bigr)_k
    \Bigr\|_{\ell_q} < \infty$.
\end{enumerate}
\end{corollary}

\begin{proof}
\leanok
Lebesgue measure on $[0, 1)$ and on $[0, 1]$ is the same measure, and a function measurable
for the dyadic $\sigma$-algebra $\mathcal{F}_j$ is constant on every $I_{j,n}$. So the Haar
scheme is the scheme of dyadic piecewise polynomials of degree $< 1$, and each of the three
conditions says $f \in A^s_q$, by \Cref{thm:pp-besov}, \Cref{thm:pp-besov-per} and
\Cref{thm:haar}.
\end{proof}

\section{Local projections and multiscale details}

\begin{lemma}[Dual basis for polynomials]
\label{lem:poly-dual-basis}
\lean{Interval.integrableOn_polyFun_mul, Interval.polyFun_add, Interval.polyFun_smul,
  Interval.polyPairing, Interval.polyPairing_apply, Interval.polyPairing_comm,
  Interval.polyFun_map_conj, Interval.polyPairing_separatingLeft,
  Interval.polyPairing_nondegenerate, Interval.monoBasis, Interval.dualPoly,
  Interval.sum_polyPairing_smul_dualPoly, Interval.polyFun_eq_sum_dualPoly}
\leanok
Let $r \in \N$ and $b_i(t) = t^i$, $i < r$. There are polynomials $d_0, \dots, d_{r-1}$ of degree
$< r$ such that every polynomial $P$ of degree $< r$ satisfies
\[
  P = \sum_{i < r} \Bigl( \int_0^1 P \, b_i \Bigr) d_i .
\]
\end{lemma}

\begin{proof}
\leanok
The pairing $\langle P, Q \rangle = \int_0^1 P Q$ on polynomials of degree $< r$ is
nondegenerate: $\langle P, \bar P \rangle = \int_0^1 |P|^2$, where $\bar P$ has the conjugate
coefficients, vanishes only for $P = 0$. So $(b_i)$ has a dual basis $(d_i)$ with
$\langle d_i, b_k \rangle = \delta_{ik}$, and the coordinates of $P$ in it are
$\langle P, b_i \rangle$.
\end{proof}

\begin{definition}[Local projections]
\label{def:loc-proj}
\lean{Interval.exists_enorm_polyFun_le_of_fintype, Interval.locCoeff, Interval.locProj,
  Interval.locProj_mem, Interval.polyFun_locProj, Interval.integrableOn_mul_monoBasis,
  Interval.locCoeff_add, Interval.locCoeff_const_mul, Interval.locCoeff_congr,
  Interval.locProj_add, Interval.locProj_const_mul, Interval.locProj_congr}
\leanok
\uses{lem:poly-dual-basis, def:pp-scheme}
For $g \in L_1(I_{j,m})$ the local projection of $g$ on $I_{j,m}$ is the polynomial
\[
  \Pi_{j,m} g(x) = \sum_{i < r} c_i \, d_i(2^j x - m), \qquad
  c_i = 2^j \int_{I_{j,m}} g(y) \, b_i(2^j y - m) \, dy .
\]
It is linear in $g$ and depends only on the class of $g$ in $L_1(I_{j,m})$.
\end{definition}

\begin{lemma}[Properties of the local projections]
\label{lem:loc-proj}
\lean{Interval.integral_dyadicIco_eq, Interval.polyFun_locProj_of_eqOn,
  Interval.ofReal_two_pow_mul_inv_rpow, Interval.volume_dyadicIco, Interval.enorm_locCoeff_le,
  Interval.enorm_polyFun_locProj_le, Interval.eLpNorm_locProj_le}
\leanok
\uses{def:loc-proj}
Let $1 \leq p < \infty$ and let $M$ bound all $|b_i|$ and $|d_i|$ on $[0, 1]$. If $g$ agrees on
$I_{j,m}$ with a polynomial $Q$ of degree $< r$, then $\Pi_{j,m} g = Q$. For every
$g \in L_p(I_{j,m})$,
\[
  \|\Pi_{j,m} g\|_{L_p(I_{j,m})} \leq r M^2 \|g\|_{L_p(I_{j,m})} .
\]
\end{lemma}

\begin{proof}
\leanok
With $y = (m + t) 2^{-j}$ the coefficients of $Q$ are $c_i = \int_0^1 \tilde Q \, b_i$ for
$\tilde Q(t) = Q((m + t) 2^{-j})$, so $\Pi_{j,m} Q(x) = \tilde Q(2^j x - m) = Q(x)$ by
\Cref{lem:poly-dual-basis}. Hölder's inequality on an interval of length $2^{-j}$ gives
$|c_i| \leq M 2^{j/p} \|g\|_{L_p(I_{j,m})}$, hence
$|\Pi_{j,m} g| \leq r M^2 2^{j/p} \|g\|_{L_p(I_{j,m})}$ on $I_{j,m}$, and the $L_p$ norm of a
function bounded by $B$ on $I_{j,m}$ is at most $2^{-j/p} B$.
\end{proof}

\begin{theorem}[Projections onto piecewise polynomials]
\label{thm:pp-proj}
\lean{Interval.ppFun_congr, Interval.ae_dyadicIco_of_unitVol, Interval.integrableOn_dyadicIco,
  Interval.basisBound, Interval.basisBound_ne_top, Interval.enorm_monoBasis_le,
  Interval.enorm_dualPoly_le, Interval.ppProjConst, Interval.ppProjConst_ne_top,
  Interval.eLpNorm_ppFun_locProj_le, Interval.ppProjLinear, Interval.ppProj,
  Interval.coeFn_ppProj, Interval.norm_ppProj_le, Interval.ppProj_mem, Interval.ppProj_of_mem,
  Interval.ppProjectionScheme, Interval.ppProjectionScheme_two_pow}
\leanok
\uses{lem:loc-proj, lem:pp-bookkeeping}
Let $1 \leq p < \infty$ and $r \in \N$. For $j \in \N$ let $P_j f = \Pi_{j,m} f$ on every
$I_{j,m}$, $m < 2^j$. Then $P_j$ is a bounded projection of $L_p[0,1]$ onto $V_j$, and
$\|P_j\| \leq r M^2$ for all $j$.
\end{theorem}

\begin{proof}
\leanok
$P_j f \in V_j$ by construction, and $P_j a = a$ for $a \in V_j$ by \Cref{lem:loc-proj}. Summing
the $p$-th powers of the local bounds of \Cref{lem:loc-proj} over the intervals of level $j$
gives $\|P_j f\|_p \leq r M^2 \|f\|_p$.
\end{proof}

\begin{theorem}[Multiscale characterisation]
\label{thm:pp-multiscale}
\lean{Interval.ppScheme_mono_degree, Interval.closure_iUnion_ppScheme,
  Interval.memApprox_ppScheme_iff_block, Interval.lqNorm_modulus_lt_top_iff_block,
  Interval.lqNorm_perModulus_lt_top_iff_block}
\leanok
\uses{thm:pp-proj, lem:haar-density, cor:haar-besov, thm:pp-besov, thm:pp-besov-per}
Let $1 \leq p < \infty$, $r \geq 1$, $s > 0$ and $1 \leq q \leq \infty$. A function
$f \in L_p[0,1]$ lies in the approximation space $A^s_q$ of the scheme of dyadic piecewise
polynomials of degree $< r$ if and only if
\[
  \|P_0 f\|_p + \bigl\| \bigl( 2^{ks} \|P_{k+1} f - P_k f\|_p \bigr)_k \bigr\|_{\ell_q} < \infty .
\]
For $0 < s < 1/p$ this condition is equivalent to $f \in \Besov{s}{p}{q}[0,1]$ and to
$f \in \Besov{s}{p}{q}(\T)$, defined by $\omega_r$ and $\omega_r^{\mathrm{per}}$.
\end{theorem}

\begin{proof}
\leanok
The piecewise polynomials are dense in $L_p[0,1]$, since they contain the step functions of
the Haar system (\Cref{lem:haar-density}, transported by \Cref{cor:haar-besov}). The
projections of \Cref{thm:pp-proj} form a projection scheme, and the abstract block
characterisation applies. The last statement combines this with \Cref{thm:pp-besov} and
\Cref{thm:pp-besov-per}.
\end{proof}

The details $P_{k+1} f - P_k f$ are what multiwavelets of degree $< r$ expand at level $k$, and
the local projections $P_j$ are those of discontinuous Galerkin methods.

\section{Cardinal B-splines}

\begin{definition}[Cardinal B-splines]
\label{def:bspline}
\lean{Interval.bsplineCoeff, Interval.bsplinePiece, Interval.bspline, Interval.bspline_eq_piece,
  Interval.bspline_eq_zero_of_lt, Interval.bspline_eq_zero_of_ge}
\leanok
Let $r \geq 1$. For $0 \leq j \leq r$ let
\[
  B_j(x) = \frac{1}{(r-1)!} \sum_{l = 0}^{j} (-1)^l \binom{r}{l} (x - l)^{r-1} .
\]
The cardinal B-spline of order $r$ is $N_r(x) = B_j(x)$ for $x \in [j, j+1)$, $0 \leq j < r$, and
$N_r(x) = 0$ for $x \notin [0, r)$.
\end{definition}

\begin{lemma}[The pieces of the B-spline]
\label{lem:bspline-pieces}
\lean{Interval.bsplinePiece_succ, Interval.pow_mem_degreeLT, Interval.natDegree_X_sub_C_le_one,
  Interval.natDegree_X_add_C_le_one, Interval.bsplinePiece_mem, Interval.neg_one_pow_eq_of_le,
  Interval.bsplinePiece_top}
\leanok
\uses{def:bspline, lem:iter-diff-poly}
Every $B_j$ is a polynomial of degree $< r$, $B_{j+1} - B_j$ is a multiple of $(x - j - 1)^{r-1}$, and
$B_r = 0$.
\end{lemma}

\begin{proof}
\leanok
The first two statements are immediate. $B_r(z)$ is $\pm 1/(r-1)!$ times the $r$-th difference
of the polynomial $t \mapsto (z - t)^{r-1}$ of degree $< r$, which vanishes by
\Cref{lem:iter-diff-poly}.
\end{proof}

So consecutive pieces of $N_r$, including the zero pieces outside $[0, r)$, agree to order $r - 2$
at the knots: $N_r$ is a spline of order $r$.

\begin{lemma}[B-splines on a unit interval]
\label{lem:bspline-basis}
\lean{Interval.bsplineLocal, Interval.bsplineLocal_eq, Interval.bsplineLocal_mem,
  Interval.finiteDimensional_degreeLT, Interval.finrank_degreeLT,
  Interval.linearIndependent_X_add_C_pow, Interval.shiftedPow, Interval.top_le_span_shiftedPow,
  Interval.bsplineVec, Interval.bsplineVec_eq, Interval.top_le_span_bsplineVec,
  Interval.bsplineBasis, Interval.bsplineBasis_apply}
\leanok
\uses{lem:bspline-pieces}
The $r$ B-splines $N_r(\cdot + j)$, $0 \leq j < r$, restricted to $[0, 1)$ are the polynomials
$\beta_j(x) = B_j(x + j)$, and these form a basis of the polynomials of degree $< r$.
\end{lemma}

\begin{proof}
\leanok
The shifted powers $(x + m)^{r-1}$, $0 \leq m < r$, are linearly independent: the coefficients
of a vanishing combination $\sum_m g_m (x + m)^{r-1}$ give $\sum_m g_m m^e = 0$ for $e < r$, a
Vandermonde system. They form a basis, since there are $r$ of them. In this basis
$\beta_j = \frac{1}{(r-1)!} (x + j)^{r-1} + \sum_{m < j} (\ldots) (x + m)^{r-1}$ is triangular,
so the $\beta_j$ span as well.
\end{proof}

\begin{theorem}[Marsden's identity]
\label{thm:marsden}
\lean{Interval.sum_range_eq_add_last, Interval.sum_range_eq_first_add, Interval.bcoord,
  Interval.sum_bcoord, Interval.bcoord_eq_of_sum, Interval.bsplineLocal_comp_add_one,
  Interval.bsplineLocal_zero, Interval.bsplineLocal_top, Interval.bcoord_comp_add_one,
  Interval.comp_X_add_C_mem, Interval.bsplineCoef, Interval.bcoord_comp_eq_bsplineCoef,
  Interval.bspline_sub_eq, Interval.polyFun_eq_sum_bsplineCoef}
\leanok
\uses{lem:bspline-basis}
Let $r \geq 1$. For every polynomial $P$ of degree $< r$ there are coefficients
$c_n(P) \in \C$, $n \in \Z$, such that for all $i \in \Z$ and $x \in [i, i+1)$
\[
  P(x) = \sum_{j < r} c_{i - j}(P) \, N_r(x - (i - j)) ,
\]
that is, $P = \sum_{n \in \Z} c_n(P) N_r(\cdot - n)$ on $\R$.
\end{theorem}

\begin{proof}
\leanok
On $[i, i+1)$ expand $P(\cdot + i)$ in the basis of \Cref{lem:bspline-basis}; let $c_i(P)$ be the
coordinate of $\beta_0$ there. Since $\beta_j(x + 1) = \beta_{j+1}(x) - c\, x^{r-1}$ with
$\beta_r = 0$ and $x^{r-1} = (r-1)! \beta_0$, the coordinate $j + 1$ of $P(\cdot + i + 1)$ is the
coordinate $j$ of $P(\cdot + i)$. By induction the coordinate $j$ on $[i, i+1)$ is
$c_{i-j}(P)$, and $\beta_j(x - i) = N_r(x - (i - j))$ there.
\end{proof}
